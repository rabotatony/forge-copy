     1→// ============================================================
     2→// Forge — Experiments Lab engine
     3→// ============================================================
     4→// Runs hard, sandboxed experiments on Forge itself to discover
     5→// development breakthroughs. Each experiment:
     6→//   1. Has a hypothesis (what we think will happen)
     7→//   2. Runs a procedure (multi-step, AI-generated where useful)
     8→//   3. Captures metrics (duration, output size, exit codes)
     9→//   4. Computes a verdict: BREAKTHROUGH | NO_CHANGE | REGRESSION
    10→//   5. Can be promoted to a permanent Forge workflow
    11→//
    12→// Safety rails (hard-enforced, non-negotiable):
    13→//   - Per-script timeout (default 10s, max 30s) — SIGKILL on overrun
    14→//   - Output cap (200 KB stdout + 200 KB stderr) — truncated, not fatal
    15→//   - AI generation timeout (30s)
    16→//   - Working directory is a fresh temp dir per experiment run
    17→//   - No network access from generated scripts (they run locally)
    18→//   - All steps logged to the ExperimentRun.evidence JSON
    19→//
    20→// The engine is deliberately self-contained: it does NOT touch the
    21→// workflow execution engine or the SSE bus, so a runaway experiment
    22→// cannot poison the main run pipeline.
    23→// ============================================================

   103→// ---------------------------------------------------------------------------
   104→// JSON extraction helper — strips markdown fences, finds the first JSON
   105→// object/array, matches nesting respecting string literals, strips trailing
   106→// commas. Returns null on parse failure.
   107→// ---------------------------------------------------------------------------
   108→function extractJson<T = unknown>(text: string): T | null {
   109→  if (typeof text !== `string`) return null;
   110→  // 1. Strip markdown fences anywhere.
   111→  let s = text.replace(/```[a-z]*\n?/gi, ``).replace(/```/gi, ``).trim();
   112→  // 2. Strip shebang lines (bash scripts wrapping JSON).
   113→  s = s.replace(/^#!.*$/gm, ``).trim();
   114→  // 3. Find the first opening brace or bracket.
   115→  const firstObj = s.indexOf(`{`);
   116→  const firstArr = s.indexOf(`[`);
   117→  let start = -1;
   118→  let open = ``;
   119→  let close = ``;
   120→  if (firstObj !== -1 && (firstArr === -1 || firstObj < firstArr)) {
   121→    start = firstObj; open = `{`; close = `}`;
   122→  } else if (firstArr !== -1) {
   123→    start = firstArr; open = `[`; close = `]`;
   124→  } else {
   125→    return null;
   126→  }
   127→  // 4. Match nesting, respecting string literals + escapes.
   128→  let depth = 0;
   129→  let inStr = false;
   130→  let end = -1;
   131→  for (let i = start; i < s.length; i++) {
   132→    const ch = s[i];
   133→    if (inStr) {
   134→      if (ch === `\\`) { i++; continue; }
   135→      if (ch === `"`) inStr = false;
   136→      continue;
   137→    }
   138→    if (ch === `"`) { inStr = true; continue; }
   139→    if (ch === open) depth++;
   140→    else if (ch === close) {
   141→      depth--;
   142→      if (depth === 0) { end = i; break; }
   143→    }
   144→  }
   145→  if (end === -1) return null;
   146→  let jsonStr = s.slice(start, end + 1);
   147→  // 5. Convert single quotes to double quotes (common AI mistake).
   148→  jsonStr = jsonStr.replace(/'/g, `"`);
   149→  // 6. Strip trailing commas before } or ].
   150→  jsonStr = jsonStr.replace(/,\s*([}\]])/g, `$1`);
   151→  try {
   152→    return JSON.parse(jsonStr) as T;
   153→  } catch {
   154→    return null;
   155→  }
   156→}

   182→/** Call the LLM to generate a script. Hard timeout via Promise.race.
   183→ *  Retries up to 2 times on rate-limit (429) errors with a 10s backoff. */
   184→async function generateScript(
   185→  prompt: string,
   186→  language: 'bash' | 'python' | 'node',
   187→): Promise<GeneratedScript> {
   188→  const zai = await ZAI.create();
   189→  const shebang = SHEBANG_BY_LANG[language];
   190→
   191→  const systemPrompt = `You are a scripting expert inside the Forge Experiments Lab.
   192→Generate a single executable ${language} script that accomplishes the user's request.
   193→Output ONLY the script code — no markdown fences, no explanation.
   194→The script MUST start with the shebang line \`${shebang}\`.
   195→The script must be self-contained, deterministic, and finish quickly (under 5 seconds).
   196→Do NOT access the network. Do NOT delete files outside the current directory.
   197→After the script, on a new line by itself, output exactly \`---DESCRIPTION---\`,
   198→then on the next line a single one-line description (max 120 chars).`;
   199→
   200→  let completion;
   201→  for (let attempt = 0; attempt < 4; attempt++) {
   202→    try {
   203→      completion = await zai.chat.completions.create({
   204→        messages: [
   205→          { role: 'system', content: systemPrompt },
   206→          { role: 'user', content: prompt },
   207→        ],
   208→        thinking: { type: 'disabled' },
   209→      });
   210→      break;
   211→    } catch (err: unknown) {
   212→      const errMsg = String(err);
   213→      if (errMsg.includes('429') && attempt < 3) {
   214→        // Rate limited — exponential backoff: 15s, 30s, 60s.
   215→        const waitMs = 15_000 * Math.pow(2, attempt);
   216→        await new Promise((r) => setTimeout(r, waitMs));
   217→        continue;
   218→      }
   219→      throw err;
   220→    }
   221→  }
   222→  if (!completion) throw new Error('LLM completion failed after retries');
   223→
   224→  const raw = completion.choices[0]?.message?.content ?? '';
   225→
   226→  let code = raw;
   227→  let description = `${language} script`;
   228→  const delim = raw.indexOf('---DESCRIPTION---');
   229→  if (delim !== -1) {
   230→    code = raw.slice(0, delim).trim();
   231→    const after = raw.slice(delim + '---DESCRIPTION---'.length).trim();
   232→    description = after.split('\n')[0]?.trim() || description;
   233→  }
   234→
   235→  // Strip markdown fences if the model added them anyway.
   236→  code = code.replace(/^```[a-z]*\n?/i, '').replace(/\n?```$/i, '').trim();
   237→
   238→  return {
   239→    language,
   240→    filename: FILENAME_BY_LANG[language],
   241→    code,
   242→    description,
   243→  };
   244→}

   246→/** Spawn a script with a hard timeout + output cap. SIGKILL on overrun. */
   247→function executeScript(
   248→  script: GeneratedScript,
   249→  workDir: string,
   250→  opts: ExecOpts = {},
   251→): Promise<ExecResult> {
   252→  return new Promise((resolve) => {
   253→    const timeoutMs = Math.min(opts.timeoutMs ?? DEFAULT_SCRIPT_TIMEOUT, MAX_SCRIPT_TIMEOUT);
   254→    const filename = script.filename;
   255→    const filePath = path.join(workDir, filename);
   256→    try {
   257→      fs.writeFileSync(filePath, script.code, { mode: 0o755 });
   258→    } catch (err) {
   259→      resolve({
   260→        exitCode: -1,
   261→        stdout: '',
   262→        stderr: `Failed to write script: ${err instanceof Error ? err.message : String(err)}`,
   263→        durationMs: 0,
   264→        timedOut: false,
   265→        truncated: false,
   266→      });
   267→      return;
   268→    }
   269→
   270→    const start = Date.now();
   271→    let cmd: string;
   272→    let args: string[];
   273→    if (script.language === 'bash') {
   274→      cmd = 'bash';
   275→      args = [filePath];
   276→    } else if (script.language === 'python') {
   277→      cmd = 'python3';
   278→      args = [filePath];
   279→    } else {
   280→      cmd = 'node';
   281→      args = [filePath];
   282→    }
   283→
   284→    const child = spawn(cmd, args, {
   285→      cwd: workDir,
   286→      env: { ...process.env, ...opts.env, FORGE_EXPERIMENT: '1' },
   287→      stdio: ['pipe', 'pipe', 'pipe'],
   288→      timeout: 0, // we manage the timeout ourselves for SIGKILL reliability
   289→    });
   290→
   291→    let stdout = '';
   292→    let stderr = '';
   293→    let truncated = false;
   294→    let timedOut = false;
   295→    let settled = false;
   296→
   297→    const finish = (exitCode: number) => {
   298→      if (settled) return;
   299→      settled = true;
   300→      resolve({
   301→        exitCode,
   302→        stdout,
   303→        stderr,
   304→        durationMs: Date.now() - start,
   305→        timedOut,
   306→        truncated,
   307→      });
   308→    };
   309→
   310→    child.stdout?.on('data', (chunk: Buffer) => {
   311→      if (stdout.length < MAX_OUTPUT_BYTES) {
   312→        stdout += chunk.toString('utf8').slice(0, MAX_OUTPUT_BYTES - stdout.length);
   313→        if (stdout.length >= MAX_OUTPUT_BYTES) truncated = true;
   314→      }
   315→    });
   316→    child.stderr?.on('data', (chunk: Buffer) => {
   317→      if (stderr.length < MAX_OUTPUT_BYTES) {
   318→        stderr += chunk.toString('utf8').slice(0, MAX_OUTPUT_BYTES - stderr.length);
   319→        if (stderr.length >= MAX_OUTPUT_BYTES) truncated = true;
   320→      }
   321→    });
   322→
   323→    if (opts.stdin) {
   324→      try {
   325→        child.stdin?.write(opts.stdin);
   326→      } catch {
   327→        /* ignore */
   328→      }
   329→    }
   330→    child.stdin?.end();
   331→
   332→    const timer = setTimeout(() => {
   333→      timedOut = true;
   334→      try {
   335→        child.kill('SIGKILL');
   336→      } catch {
   337→        /* ignore */
   338→      }
   339→    }, timeoutMs);
   340→
   341→    child.on('error', () => {
   342→      clearTimeout(timer);
   343→      finish(-1);
   344→    });
   345→    child.on('close', (code) => {
   346→      clearTimeout(timer);
   347→      finish(code ?? -1);
   348→    });
   349→  });
   350→}
