    56→export interface ValidationResult {
    57→  valid: boolean;
    58→  errors: string[];
    59→}
    60→
    61→/**
    62→ * Validate a PipelineDefinition:
    63→ *   - stages must be non-empty
    64→ *   - all stage IDs unique
    65→ *   - all `needs` reference existing stages
    66→ *   - no cycles
    67→ */
    68→export function validatePipelineDefinition(def: PipelineDefinition): ValidationResult {
    69→  const errors: string[] = [];
    70→  if (!def.stages || def.stages.length === 0) {
    71→    errors.push('Pipeline must have at least one stage.');
    72→    return { valid: false, errors };
    73→  }
    74→  const ids = new Set<string>();
    75→  for (const stage of def.stages) {
    76→    if (!stage.id) {
    77→      errors.push(`Stage "${stage.name ?? '<unnamed>'}" is missing an id.`);
    78→      continue;
    79→    }
    80→    if (ids.has(stage.id)) {
    81→      errors.push(`Duplicate stage id: "${stage.id}".`);
    82→    }
    83→    ids.add(stage.id);
    84→  }
    85→  // Check `needs` references.
    86→  for (const stage of def.stages) {
    87→    if (!stage.needs) continue;
    88→    for (const need of stage.needs) {
    89→      if (!ids.has(need)) {
    90→        errors.push(`Stage "${stage.id}" needs unknown stage "${need}".`);
    91→      }
    92→      if (need === stage.id) {
    93→        errors.push(`Stage "${stage.id}" cannot need itself.`);
    94→      }
    95→    }
    96→  }
    97→  // Cycle detection (DFS).
    98→  const cycle = detectCycle(def.stages);
    99→  if (cycle) {
   100→    errors.push(`Pipeline has a cycle: ${cycle.join(' → ')}.`);
   101→  }
   102→  return { valid: errors.length === 0, errors };
   103→}
   104→
   105→function detectCycle(stages: PipelineStage[]): string[] | null {
   106→  const adj = new Map<string, string[]>();
   107→  for (const s of stages) adj.set(s.id, s.needs ?? []);
   108→  const visited = new Set<string>();
   109→  const path = new Set<string>();
   110→  const pathList: string[] = [];
   111→
   112→  function dfs(node: string): string[] | null {
   113→    if (path.has(node)) {
   114→      // Found a cycle — return the cycle path from the first occurrence.
   115→      const start = pathList.indexOf(node);
   116→      return [...pathList.slice(start), node];
   117→    }
   118→    if (visited.has(node)) return null;
   119→    visited.add(node);
   120→    path.add(node);
   121→    pathList.push(node);
   122→    for (const dep of adj.get(node) ?? []) {
   123→      const result = dfs(dep);
   124→      if (result) return result;
   125→    }
   126→    path.delete(node);
   127→    pathList.pop();
   128→    return null;
   129→  }
   130→
   131→  for (const s of stages) {
   132→    const result = dfs(s.id);
   133→    if (result) return result;
   134→  }
   135→  return null;
   136→}

   138→// ---------------------------------------------------------------------------
   139→// Topological ordering (Kahn's algorithm — level by level)
   140→// ---------------------------------------------------------------------------
   141→
   142→/**
   143→ * Group stages into "levels" — sets of stages that can run in parallel
   144→ * because all their `needs` are in earlier levels.
   145→ */
   146→function topologicalLevels(stages: PipelineStage[]): PipelineStage[][] {
   147→  const byId = new Map(stages.map(s => [s.id, s]));
   148→  const remaining = new Set(stages.map(s => s.id));
   149→  const levels: PipelineStage[][] = [];
   150→  while (remaining.size > 0) {
   151→    const ready: PipelineStage[] = [];
   152→    for (const id of remaining) {
   153→      const stage = byId.get(id)!;
   154→      const needs = stage.needs ?? [];
   155→      if (needs.every(n => !remaining.has(n))) {
   156→        ready.push(stage);
   157→      }
   158→    }
   159→    if (ready.length === 0) {
   160→      // Shouldn't happen — cycle detection should have caught this.
   161→      break;
   162→    }
   163→    for (const s of ready) remaining.delete(s.id);
   164→    levels.push(ready);
   165→  }
   166→  return levels;
   167→}

   173→interface ConditionContext {
   174→  matrix: MatrixRow;
   175→  // Status of all needed stages: { stageId: 'success' | 'failed' | 'skipped' | 'canceled' }
   176→  neededResults: Record<string, string>;
   177→}
   178→
   179→/**
   180→ * Evaluate a stage `if` condition.
   181→ *
   182→ * Supported syntax:
   183→ *   • `always()`                    — always true
   184→ *   • `success()`                   — true if all needed stages succeeded
   185→ *   • `failure()`                   — true if any needed stage failed/canceled
   186→ *   • `matrix.KEY == 'value'`       — string equality
   187→ *   • `matrix.KEY != 'value'`       — string inequality
   188→ *   • `matrix.KEY == "value"`       — string equality (double quotes)
   189→ *   • `matrix.KEY != "value"`       — string inequality (double quotes)
   190→ *   • Combinations via `&&`, `||`, `!`, and parentheses.
   191→ *
   192→ * Returns false on any parse error (conservative — skip the stage).
   193→ */
   194→export function evaluateCondition(expr: string, ctx: ConditionContext): boolean {
   195→  if (!expr || expr.trim() === '') return true;
   196→  const allSuccess = Object.values(ctx.neededResults).every(s => s === 'success');
   197→  const anyFailed = Object.values(ctx.neededResults).some(s => s === 'failed' || s === 'canceled');
   198→
   199→  let s = expr.trim();
   200→  // Function-call replacements.
   201→  s = s.replace(/\balways\s*\(\s*\)/g, 'true');
   202→  s = s.replace(/\bsuccess\s*\(\s*\)/g, allSuccess ? 'true' : 'false');
   203→  s = s.replace(/\bfailure\s*\(\s*\)/g, anyFailed ? 'true' : 'false');
   204→  // matrix.KEY == 'value' / != 'value' (single or double quotes).
   205→  s = s.replace(
   206→    /\bmatrix\.(\w+)\s*(==|!=)\s*'([^']*)'/g,
   207→    (_, key, op, val) => (op === '==' ? ((ctx.matrix[key] ?? '') === val) : ((ctx.matrix[key] ?? '') !== val)) ? 'true' : 'false',
   208→  );
   209→  s = s.replace(
   210→    /\bmatrix\.(\w+)\s*(==|!=)\s*"([^"]*)"/g,
   211→    (_, key, op, val) => (op === '==' ? ((ctx.matrix[key] ?? '') === val) : ((ctx.matrix[key] ?? '') !== val)) ? 'true' : 'false',
   212→  );
   213→  // Any remaining matrix.KEY reference → replace with its value (quoted) so
   214→  // bare references work too (treated as truthy non-empty string).
   215→  s = s.replace(/\bmatrix\.(\w+)/g, (_, key) => (ctx.matrix[key] ?? '') ? 'true' : 'false');
   216→
   217→  return evalBoolean(s);
   218→}
   219→
   220→/**
   221→ * Evaluate a reduced boolean expression containing only `true`, `false`,
   222→ * `&&`, `||`, `!`, `(`, `)`, and whitespace. Hand-written recursive descent
   223→ * parser — no `eval` / `Function` constructor, so no code injection risk.
   224→ */
   225→function evalBoolean(s: string): boolean {
   226→  const tokens: string[] = [];
   227→  let i = 0;
   228→  while (i < s.length) {
   229→    const c = s[i];
   230→    if (c === ' ' || c === '\t' || c === '\n' || c === '\r') { i++; continue; }
   231→    if (c === '(' || c === ')') { tokens.push(c); i++; continue; }
   232→    if (c === '!') { tokens.push('!'); i++; continue; }
   233→    if (c === '&' && s[i + 1] === '&') { tokens.push('&&'); i += 2; continue; }
   234→    if (c === '|' && s[i + 1] === '|') { tokens.push('||'); i += 2; continue; }
   235→    if (s.slice(i, i + 4) === 'true') { tokens.push('true'); i += 4; continue; }
   236→    if (s.slice(i, i + 5) === 'false') { tokens.push('false'); i += 5; continue; }
   237→    // Unknown token — bail.
   238→    return false;
   239→  }
   240→  let pos = 0;
   241→  const peek = (): string | undefined => tokens[pos];
   242→  const consume = (): string | undefined => tokens[pos++];
   243→  function parseOr(): boolean {
   244→    let v = parseAnd();
   245→    while (peek() === '||') { consume(); v = parseAnd() || v; }
   246→    return v;
   247→  }
   248→  function parseAnd(): boolean {
   249→    let v = parseNot();
   250→    while (peek() === '&&') { consume(); v = parseNot() && v; }
   251→    return v;
   252→  }
   253→  function parseNot(): boolean {
   254→    if (peek() === '!') { consume(); return !parseNot(); }
   255→    return parsePrimary();
   256→  }
   257→  function parsePrimary(): boolean {
   258→    const t = consume();
   259→    if (t === 'true') return true;
   260→    if (t === 'false') return false;
   261→    if (t === '(') {
   262→      const v = parseOr();
   263→      if (peek() === ')') consume();
   264→      return v;
   265→    }
   266→    return false;
   267→  }
   268→  return parseOr();
   269→}

   277→function substituteMatrixInRecord(rec: Record<string, string> | undefined, matrix: MatrixRow): Record<string, string> {

   376→// ============================================================
   377→// Custom workflows (moved from `./custom-workflow` in R-4)
   378→// ============================================================
   379→// A CustomWorkflow is a JSON document authored by the project owner
   380→// describing a list of shell steps (with per-step env, retry, timeout,
   381→// cache, working dir, test-report capture, and continueOnError). It is
   382→// the lightweight alternative to defining a multi-stage Pipeline — in
   383→// fact, a custom workflow IS a single-stage pipeline. Storage: a custom
   384→// workflow is stored as a Pipeline with a single 'custom' stage and the
   385→// full CustomWorkflow JSON in the pipeline's `config.customWorkflow`
   386→// field. The pipeline executor detects this and calls `runCustomWorkflow`
   387→// directly (it's now in the same file) instead of `startRunExtended`.
   388→//
   389→// Custom workflows share the engine's SSE event bus (via `subscribe`,
   390→// `emit`, `appendLog`, `finishRun`) so the existing
   391→// `/api/forge/runs/[id]/events` SSE route picks them up automatically.
   392→// ------------------------------------------------------------
   393→
   394→// Allowed interpreter languages for custom workflow steps.
   395→const STEP_LANGUAGES: readonly CustomWorkflowStepLanguage[] = ['bash', 'node', 'python', 'ruby'] as const;
   396→
   397→/**
   398→ * Coerce an unknown value into a valid CustomWorkflowStepLanguage, or throw.
   399→ * Returns 'bash' when the value is absent (undefined/null). Throws a clear
   400→ * error when the value is present but not one of the allowed languages.
   401→ */
   402→function coerceStepLanguage(raw: unknown, stepIndex: number): CustomWorkflowStepLanguage {
   403→  if (raw === undefined || raw === null) return 'bash';
   404→  if (typeof raw !== 'string') {
   405→    throw new Error(`Step ${stepIndex} has invalid "language" (expected string, got ${typeof raw}).`);
   406→  }
   407→  if (!(STEP_LANGUAGES as readonly string[]).includes(raw)) {
   408→    throw new Error(`Step ${stepIndex} has invalid "language" "${raw}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
   409→  }
   410→  return raw as CustomWorkflowStepLanguage;
   411→}

   417→/**
   418→ * Parse a JSON string into a CustomWorkflow. Throws on invalid input.
   419→ */
   420→export function parseCustomWorkflow(json: string): CustomWorkflow {
   421→  let raw: unknown;
   422→  try {
   423→    raw = JSON.parse(json);
   424→  } catch (err) {
   425→    throw new Error(`Custom workflow is not valid JSON: ${err instanceof Error ? err.message : String(err)}`);
   426→  }
   427→  if (typeof raw !== 'object' || raw === null || Array.isArray(raw)) {
   428→    throw new Error('Custom workflow must be a JSON object.');
   429→  }
   430→  const obj = raw as Record<string, unknown>;
   431→  if (typeof obj.name !== 'string' || obj.name.trim() === '') {
   432→    throw new Error('Custom workflow must have a non-empty "name" string.');
   433→  }
   434→  if (!Array.isArray(obj.steps) || obj.steps.length === 0) {
   435→    throw new Error('Custom workflow must have a non-empty "steps" array.');
   436→  }
   437→  const steps: CustomWorkflowStep[] = [];
   438→  for (let i = 0; i < obj.steps.length; i++) {
   439→    const s = obj.steps[i] as Record<string, unknown>;
   440→    if (typeof s !== 'object' || s === null) {
   441→      throw new Error(`Step ${i} must be an object.`);
   442→    }
   443→    if (typeof s.name !== 'string' || s.name.trim() === '') {
   444→      throw new Error(`Step ${i} must have a non-empty "name".`);
   445→    }
   446→    if (typeof s.run !== 'string' || s.run.trim() === '') {
   447→      throw new Error(`Step ${i} must have a non-empty "run" command.`);
   448→    }
   449→    const step: CustomWorkflowStep = { name: s.name, run: s.run };
   450→    // Per-step interpreter language (defaults to 'bash' when absent).
   451→    const language = coerceStepLanguage(s.language, i);
   452→    if (language !== 'bash') step.language = language;
   453→    if (typeof s.workingDir === 'string') step.workingDir = s.workingDir;
   454→    if (s.env && typeof s.env === 'object') {
   455→      step.env = Object.fromEntries(
   456→        Object.entries(s.env as Record<string, unknown>).filter(([, v]) => typeof v === 'string'),
   457→      ) as Record<string, string>;
   458→    }
   459→    if (typeof s.retry === 'number') step.retry = s.retry;
   460→    if (typeof s.timeoutMs === 'number') step.timeoutMs = s.timeoutMs;
   461→    if (s.cache && typeof s.cache === 'object') {
   462→      const c = s.cache as Record<string, unknown>;
   463→      if (typeof c.key === 'string' && Array.isArray(c.paths)) {
   464→        step.cache = {
   465→          key: c.key,
   466→          paths: (c.paths as unknown[]).filter((p): p is string => typeof p === 'string'),
   467→          restore: c.restore !== false,
   468→          save: c.save !== false,
   469→        };
   470→      }
   471→    }
   472→    if (s.testReport && typeof s.testReport === 'object') {
   473→      const tr = s.testReport as Record<string, unknown>;
   474→      if (typeof tr.format === 'string' && typeof tr.path === 'string') {
   475→        step.testReport = {
   476→          format: tr.format as 'junit' | 'json' | 'tap',
   477→          path: tr.path,
   478→        };
   479→      }
   480→    }
   481→    if (typeof s.continueOnError === 'boolean') step.continueOnError = s.continueOnError;
   482→    steps.push(step);
   483→  }
   484→  const workflow: CustomWorkflow = { name: obj.name, steps };
   485→  if (typeof obj.description === 'string') workflow.description = obj.description;
   486→  if (obj.matrix && typeof obj.matrix === 'object') {
   487→    const m = obj.matrix as Record<string, unknown>;
   488→    if (Array.isArray(m.dimensions)) {
   489→      workflow.matrix = {
   490→        dimensions: (m.dimensions as unknown[])
   491→          .filter((d): d is Record<string, unknown> => typeof d === 'object' && d !== null)
   492→          .map(d => ({
   493→            key: String(d.key ?? ''),
   494→            values: Array.isArray(d.values)
   495→              ? (d.values as unknown[]).filter((v): v is string => typeof v === 'string')
   496→              : [],
   497→          }))
   498→          .filter(d => d.key !== ''),
   499→        exclude: Array.isArray(m.exclude) ? (m.exclude as MatrixRow[]) : undefined,
   500→        include: Array.isArray(m.include) ? (m.include as MatrixRow[]) : undefined,
   501→      };
   502→    }
   503→  }
   504→  if (typeof obj.retry === 'number') workflow.retry = obj.retry;
   505→  if (typeof obj.timeoutMs === 'number') workflow.timeoutMs = obj.timeoutMs;
   506→  if (typeof obj.requiresApproval === 'boolean') workflow.requiresApproval = obj.requiresApproval;
   507→  if (Array.isArray(obj.secrets)) {
   508→    workflow.secrets = (obj.secrets as unknown[]).filter((s): s is string => typeof s === 'string');
   509→  }
   510→  if (obj.env && typeof obj.env === 'object') {
   511→    workflow.env = Object.fromEntries(
   512→      Object.entries(obj.env as Record<string, unknown>).filter(([, v]) => typeof v === 'string'),
   513→    ) as Record<string, string>;
   514→  }
   515→  return workflow;
   516→}

   518→/**
   519→ * Validate a CustomWorkflow object. Returns `{ valid, errors }`.
   520→ */
   521→export function validateCustomWorkflow(workflow: CustomWorkflow): { valid: boolean; errors: string[] } {
   522→  const errors: string[] = [];
   523→  if (!workflow.name || workflow.name.trim() === '') {
   524→    errors.push('Custom workflow must have a non-empty "name".');
   525→  }
   526→  if (!workflow.steps || workflow.steps.length === 0) {
   527→    errors.push('Custom workflow must have at least one step.');
   528→  } else {
   529→    const names = new Set<string>();
   530→    for (let i = 0; i < workflow.steps.length; i++) {
   531→      const s = workflow.steps[i]!;
   532→      if (!s.name || s.name.trim() === '') {
   533→        errors.push(`Step ${i} must have a non-empty "name".`);
   534→        continue;
   535→      }
   536→      if (names.has(s.name)) {
   537→        errors.push(`Duplicate step name: "${s.name}".`);
   538→      }
   539→      names.add(s.name);
   540→      if (!s.run || s.run.trim() === '') {
   541→        errors.push(`Step "${s.name}" must have a non-empty "run" command.`);
   542→      }
   543→      if (s.language !== undefined && !(STEP_LANGUAGES as readonly string[]).includes(s.language)) {
   544→        errors.push(`Step "${s.name}" has invalid "language" "${s.language}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
   545→      }
   546→    }
   547→  }
   548→  if (workflow.matrix?.dimensions) {
   549→    for (const dim of workflow.matrix.dimensions) {
   550→      if (!dim.key || dim.key.trim() === '') {
   551→        errors.push('Matrix dimension must have a non-empty "key".');
   552→      }
   553→      if (!dim.values || dim.values.length === 0) {
   554→        errors.push(`Matrix dimension "${dim.key}" must have at least one value.`);
   555→      }
   556→    }
   557→  }
   558→  return { valid: errors.length === 0, errors };
   559→}

   637→  const run = await db.run.create({

   645→      pipelineRunId: options.pipelineRunId,
   646→      stageId: options.stageId,
   647→      label: options.label ?? workflow.name,
   648→    },
   649→  });
   650→
   651→  emit({ type: 'status', runId: run.id, status: 'running' });
   652→
   653→  // Fire-and-forget execution.
   654→  void executeCustomSteps(run.id, project.extractedPath, workflow, projectId, options);
   655→
   656→  return { runId: run.id };

   659→async function executeCustomSteps(
   660→  runId: string,
   661→  projectRoot: string,
   662→  workflow: CustomWorkflow,
   663→  projectId: string,
   664→  options: RunCustomWorkflowOptions,
   665→): Promise<void> {
   666→  const startedAt = Date.now();
   667→  try {
   668→    // Resolve secrets (workflow-level).
   669→    const secretKeys = new Set<string>(workflow.secrets ?? []);
   670→    const allSecrets = secretKeys.size > 0
   671→      ? await getSecrets(projectId, Array.from(secretKeys))
   672→      : {};
   673→
   674→    // Merge workflow-level env + matrix-derived env.
   675→    const workflowEnv: Record<string, string> = { ...(workflow.env ?? {}) };
   676→    if (options.matrixValues) {
   677→      for (const [k, v] of Object.entries(options.matrixValues)) {
   678→        workflowEnv[`MATRIX_${k.toUpperCase()}`] = v;
   679→      }
   680→    }
   681→
   682→    // Run each step.
   683→    let lastExitCode = 0;
   684→    for (const step of workflow.steps) {
   685→      // Matrix substitution in step command + env.
   686→      const matrix = options.matrixValues ?? {};
   687→      const stepCommand = substituteMatrix(step.run, matrix);
   688→      const stepEnv: Record<string, string> = {
   689→        ...workflowEnv,
   690→        ...substituteMatrixInRecord(step.env ?? {}, matrix),
   691→      };
   692→
   693→      // Cache restore (per-step).
   694→      if (step.cache?.restore) {
   695→        const hit = await hasCache(projectId, step.cache.key);
   696→        if (hit) {
   697→          const restored = await restoreCache(projectId, step.cache.key);
   698→          if (restored.hit) {
   699→            await appendLog(runId, 'system', `💾 Cache hit: ${step.cache.key} (${formatBytes(restored.size ?? 0)})`);
   700→          }
   701→        } else {
   702→          await appendLog(runId, 'system', `💾 Cache miss: ${step.cache.key}`);
   703→        }
   704→      }
   705→
   706→      // Run with per-step retry.
   707→      const maxAttempts = (step.retry ?? 0) + 1;
   708→      let attempt = 0;
   709→      let stepExit = 1;
   710→      while (attempt < maxAttempts) {
   711→        if (attempt > 0) {
   712→          emit({ type: 'retry', runId, retryAttempt: attempt });
   713→          await appendLog(runId, 'system', `↻ Retry ${attempt}/${maxAttempts - 1} for: ${step.name}`);
   714→        }
   715→        await appendLog(runId, 'system', `▶ ${step.name}`);
   716→        // Pass the step with `run` already matrix-substituted so the
   717→        // executeStepCommand helper can write the script body to a temp
   718→        // file when language is node/python/ruby.
   719→        stepExit = await executeStepCommand(
   720→          runId,
   721→          { ...step, run: stepCommand },
   722→          step.workingDir ? path.join(projectRoot, step.workingDir) : projectRoot,
   723→          { secrets: allSecrets, extraEnv: stepEnv, timeoutMs: step.timeoutMs, projectId },
   724→        );
   725→        if (stepExit === 0) break;
   726→        attempt++;
   727→      }
   728→
   729→      // Cache save (per-step) — even on failure (partial cache is useful).
   730→      if (step.cache?.save) {
   731→        try {
   732→          const { size } = await saveCache(projectId, step.cache.key, step.cache.key, step.cache.paths);
   733→          await appendLog(runId, 'system', `💾 Cache saved: ${step.cache.key} (${formatBytes(size)})`);
   734→        } catch (err) {
   735→          await appendLog(runId, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   736→        }
   737→      }
   738→
   739→      // Test report capture (per-step).
   740→      if (step.testReport) {
   741→        try {
   742→          const reportPath = path.join(projectRoot, step.testReport.path);
   743→          if (fs.existsSync(reportPath)) {
   744→            const content = fs.readFileSync(reportPath, 'utf-8');
   745→            const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   746→            const report =
   747→              step.testReport.format === 'junit' ? parseJUnit(content)
   748→              : step.testReport.format === 'json' ? parseJSONReport(content)
   749→              : parseTAP(content);
   750→            await storeTestReport(runId, report);
   751→            await appendLog(runId, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   752→          }
   753→        } catch (err) {
   754→          await appendLog(runId, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   755→        }
   756→      }
   757→
   758→      if (stepExit !== 0) {
   759→        lastExitCode = stepExit;
   760→        if (step.continueOnError) {
   761→          await appendLog(runId, 'system', `⚠ Step "${step.name}" failed with exit code ${stepExit} (continueOnError=true).`);
   762→          continue;
   763→        }
   764→        await appendLog(runId, 'system', `✗ Step "${step.name}" failed with exit code ${stepExit} after ${attempt} attempt(s).`);
   765→        await finishRun(runId, 'failed', stepExit, Date.now() - startedAt);
   766→        return;
   767→      }
   768→      await appendLog(runId, 'system', `✓ ${step.name}`);
   769→    }
   770→
   771→    // Workflow-level test report capture (after all steps complete).
   772→    if (workflow.testReport) {
   773→      try {
   774→        const reportPath = path.join(projectRoot, workflow.testReport.path);
   775→        if (fs.existsSync(reportPath)) {
   776→          const content = fs.readFileSync(reportPath, 'utf-8');
   777→          const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   778→          const report =
   779→            workflow.testReport.format === 'junit' ? parseJUnit(content)
   780→            : workflow.testReport.format === 'json' ? parseJSONReport(content)
   781→            : parseTAP(content);
   782→          await storeTestReport(runId, report);
   783→          await appendLog(runId, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   784→        }
   785→      } catch (err) {
   786→        await appendLog(runId, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   787→      }
   788→    }
   789→
   790→    await finishRun(runId, 'success', lastExitCode, Date.now() - startedAt);
   791→  } catch (err) {
   792→    await appendLog(runId, 'system', `Custom workflow crashed: ${err instanceof Error ? err.message : String(err)}`);
   793→    await finishRun(runId, 'failed', -1, Date.now() - startedAt);
   794→  }
   795→}

   801→interface StepEnvOptions {
   802→  projectId: string;
   803→  secrets: Record<string, string>;
   804→  extraEnv: Record<string, string>;
   805→  timeoutMs?: number;
   806→}
   807→
   808→const LANGUAGE_MAP: Record<CustomWorkflowStepLanguage, StepLanguage> = {
   809→  bash: 'bash',
   810→  node: 'node',
   811→  python: 'python',
   812→  ruby: 'ruby',
   813→};
   814→
   815→/**
   816→ * Execute a single custom-workflow step. Delegates the actual spawning,
   817→ * timeout, secret-masking, and SIGTERM/SIGKILL dance to the shared
   818→ * `runChildStep` primitive so behaviour is identical to the engine's
   819→ * `runShellStep`.
   820→ */
   821→async function executeStepCommand(
   822→  runId: string,
   823→  step: CustomWorkflowStep,
   824→  cwd: string,
   825→  envOptions: StepEnvOptions,
   826→): Promise<number> {
   827→  const rawLanguage = step.language ?? 'bash';
   828→  if (!(STEP_LANGUAGES as readonly string[]).includes(rawLanguage)) {
   829→    await appendLog(runId, 'system', `✗ Unknown step language "${String(step.language)}" for step "${step.name}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
   830→    return 1;
   831→  }
   832→  const language = LANGUAGE_MAP[rawLanguage];
   833→  const command = step.run;
   834→
   835→  if (isBlockedCommand(command)) {
   836→    await appendLog(runId, 'system', `🚫 Command blocked by security policy: ${command.slice(0, 80)}`);
   837→    return 126;
   838→  }
   839→
   840→  await appendLog(runId, 'system', `▶ Running ${language} step: ${step.name}`);
   841→
   842→  const baseEnv = await buildProcessEnv(envOptions.projectId, {
   843→    extraEnv: envOptions.extraEnv,
   844→    projectRoot: cwd,
   845→  });
   846→  const fullEnv = { ...baseEnv, ...envOptions.secrets };
   847→
   848→  const result = await runChildStep({
   849→    cwd,
   850→    command,
   851→    language,
   852→    env: fullEnv,
   853→    secrets: envOptions.secrets,
   854→    timeoutMs: envOptions.timeoutMs ?? null,
   855→    onLine: (stream, text) => { void appendLog(runId, stream, text); },
   856→  });
   857→
   858→  if (result.timedOut) {
   859→    await appendLog(runId, 'system', `⏱ Step timed out after ${envOptions.timeoutMs}ms.`);
   860→  }
   861→  return result.exitCode;
   862→}

   904→/** Alias used by `triggers.ts`'s lazy importer. */

   906→
   907→async function runPipelineInBackground(
   908→  pipelineRunId: string,
   909→  projectId: string,
   910→  stages: PipelineStage[],
   911→  config: PipelineDefinition['config'],
   912→  trigger: string,
   913→): Promise<void> {
   914→  try {
   915→    // Check if this is a custom-workflow pipeline (single 'custom' stage with
   916→    // the workflow JSON stored in config.customWorkflow).
   917→    const customWorkflow = (config as { customWorkflow?: unknown }).customWorkflow;
   918→    if (customWorkflow && stages.length === 1 && stages[0]?.workflow === 'custom') {
   919→      const stage = stages[0];
   920→      const stageRun = await db.stageRun.create({
   921→        data: {
   922→          pipelineRunId,
   923→          stageId: stage.id,
   924→          stageName: stage.name,
   925→          status: 'running',
   926→          startedAt: new Date(),
   927→          runIds: '[]',
   928→        },
   929→      });
   930→      try {
   931→        const { runId } = await runCustomWorkflow(projectId, customWorkflow as Parameters<typeof runCustomWorkflow>[1], {
   932→          trigger: trigger as 'manual' | 'auto' | 'webhook' | 'cron' | 'pipeline',
   933→          pipelineRunId,
   934→          stageId: stage.id,
   935→          label: stage.name,
   936→        });
   937→        await db.stageRun.update({
   938→          where: { id: stageRun.id },
   939→          data: { runIds: JSON.stringify([runId]) },
   940→        });
   941→        const finalStatus = await waitForRunCompletion(runId);
   942→        await db.stageRun.update({
   943→          where: { id: stageRun.id },
   944→          data: {
   945→            status: finalStatus,
   946→            finishedAt: new Date(),
   947→          },
   948→        });
   949→        await finishPipelineRun(pipelineRunId, finalStatus === 'success' ? 'success' : 'failed');
   950→        return;
   951→      } catch (err) {
   952→        await db.stageRun.update({
   953→          where: { id: stageRun.id },
   954→          data: {
   955→            status: 'failed',
   956→            finishedAt: new Date(),
   957→            error: err instanceof Error ? err.message : String(err),
   958→          },
   959→        });
   960→        await finishPipelineRun(pipelineRunId, 'failed');
   961→        return;
   962→      }
   963→    }
   964→
   965→    // Standard DAG execution: topologically-grouped levels, run in parallel.
   966→    const levels = topologicalLevels(stages);
   967→    const stageResults: Record<string, string> = {}; // stageId → 'success' | 'failed' | 'skipped' | 'canceled'
   968→
   969→    for (const level of levels) {
   970→      // Run all stages in this level in parallel.
   971→      await Promise.all(level.map(stage => runStage(stage, pipelineRunId, projectId, trigger, stageResults, config)));
   972→      // If any stage in this level failed/canceled and the pipeline doesn't
   973→      // allow continuation, we still proceed to the next level — individual
   974→      // stages will skip themselves based on their `needs` results.
   975→    }
   976→
   977→    // Final status: success if no stage failed/canceled; failed otherwise.
   978→    const anyFailed = Object.values(stageResults).some(s => s === 'failed' || s === 'canceled');
   979→    await finishPipelineRun(pipelineRunId, anyFailed ? 'failed' : 'success');
   980→  } catch (err) {
   981→    await db.pipelineRun.update({
   982→      where: { id: pipelineRunId },
   983→      data: {
   984→        status: 'failed',
   985→        finishedAt: new Date(),
   986→      },
   987→    });
   988→    console.error('[forge:pipeline] background execution failed:', err);
   989→  }
   990→}
   991→
   992→async function runStage(
   993→  stage: PipelineStage,
   994→  pipelineRunId: string,
   995→  projectId: string,
   996→  trigger: string,
   997→  stageResults: Record<string, string>,
   998→  config: PipelineDefinition['config'],
   999→): Promise<void> {
  1000→  const neededResults: Record<string, string> = {};
  1001→  for (const need of stage.needs ?? []) {
  1002→    neededResults[need] = stageResults[need] ?? 'skipped';
  1003→  }
  1004→
  1005→  // Create a StageRun row up-front (status: pending).
  1006→  const stageRun = await db.stageRun.create({
  1007→    data: {
  1008→      pipelineRunId,
  1009→      stageId: stage.id,
  1010→      stageName: stage.name,
  1011→      status: 'pending',
  1012→      runIds: '[]',
  1013→    },
  1014→  });
  1015→
  1016→  // 1. `if` condition.
  1017→  try {
  1018→    if (stage.if && !evaluateCondition(stage.if, { matrix: {}, neededResults })) {
  1019→      await db.stageRun.update({
  1020→        where: { id: stageRun.id },
  1021→        data: { status: 'skipped', finishedAt: new Date() },
  1022→      });
  1023→      stageResults[stage.id] = 'skipped';
  1024→      return;
  1025→    }
  1026→  } catch {
  1027→    // Parse error → skip conservatively.
  1028→    await db.stageRun.update({
  1029→      where: { id: stageRun.id },
  1030→      data: { status: 'skipped', finishedAt: new Date() },
  1031→    });
  1032→    stageResults[stage.id] = 'skipped';
  1033→    return;
  1034→  }
  1035→
  1036→  // 2. If any `needs` stage failed/canceled/skipped and this stage doesn't
  1037→  //    have `if: failure()` (or `if: always()`), skip.
  1038→  const needsFailed = Object.values(neededResults).some(s => s === 'failed' || s === 'canceled');
  1039→  const needsSkipped = Object.values(neededResults).some(s => s === 'skipped');
  1040→  const hasFailureCondition = stage.if?.includes('failure()') ?? false;
  1041→  const hasAlwaysCondition = stage.if?.includes('always()') ?? false;
  1042→  if ((needsFailed || needsSkipped) && !hasFailureCondition && !hasAlwaysCondition) {
  1043→    await db.stageRun.update({
  1044→      where: { id: stageRun.id },
  1045→      data: { status: 'skipped', finishedAt: new Date() },
  1046→    });
  1047→    stageResults[stage.id] = 'skipped';
  1048→    return;
  1049→  }
  1050→
  1051→  await db.stageRun.update({
  1052→    where: { id: stageRun.id },
  1053→    data: { status: 'running', startedAt: new Date() },
  1054→  });
  1055→
  1056→  // 3. Expand matrix (if any).
  1057→  let matrixRows: MatrixRow[] = [{}];
  1058→  if (stage.matrix?.dimensions && stage.matrix.dimensions.length > 0) {
  1059→    matrixRows = expandMatrix(
  1060→      stage.matrix.dimensions,
  1061→      stage.matrix.exclude ?? [],
  1062→      stage.matrix.include ?? [],
  1063→    );
  1064→  }
  1065→
  1066→  const runIds: string[] = [];
  1067→  let stageFailed = false;
  1068→  let stageCanceled = false;
  1069→
  1070→  // 4. Run each matrix row. Use limited concurrency (Promise.all is fine
  1071→  //    for typical matrix sizes — they're usually ≤ 10 rows).
  1072→  const results = await Promise.all(matrixRows.map(async (matrixValues, idx) => {
  1073→    // Per-row env / cache.key / if substitution.
  1074→    const rowEnv = substituteMatrixInRecord(stage.env, matrixValues);
  1075→    const rowCacheKey = stage.cache ? substituteMatrix(stage.cache.key, matrixValues) : undefined;
  1076→
  1077→    // Re-evaluate `if` with this matrix row (per-row skip).
  1078→    if (stage.if) {
  1079→      try {
  1080→        if (!evaluateCondition(stage.if, { matrix: matrixValues, neededResults })) {
  1081→          return { matrixValues, status: 'skipped' as const, runId: null };
  1082→        }
  1083→      } catch {
  1084→        return { matrixValues, status: 'skipped' as const, runId: null };
  1085→      }
  1086→    }
  1087→
  1088→    // 5. Start the run via the extended runner. Always use 'pipeline' as
  1089→    //    the trigger so matrix siblings don't cancel each other via the
  1090→    //    engine's concurrent-cancellation logic. The user-facing trigger
  1091→    //    (manual/webhook/cron) is preserved on the PipelineRun row.
  1092→    //
  1093→    //    RETRY POLICY: we delegate per-step retry to `startRunExtended`
  1094→    //    (which retries only the failing step, not the whole stage).
  1095→    //    The previous whole-stage retry loop here was removed because
  1096→    //    it caused a quadratic (retry+1)² explosion — a stage with
  1097→    //    retry=2 would run each step up to 9 times.
  1098→    const retry = stage.retry ?? config?.defaultRetry ?? 0;
  1099→    const timeoutMs = stage.timeoutMs ?? config?.defaultTimeoutMs ?? undefined;
  1100→
  1101→    const { runId } = await startRunExtended({
  1102→      projectId,
  1103→      workflow: stage.workflow,
  1104→      trigger: 'pipeline',
  1105→      secrets: stage.secrets,
  1106→      env: rowEnv,
  1107→      cache: stage.cache
  1108→        ? {
  1109→            key: rowCacheKey ?? stage.cache.key,
  1110→            label: stage.name,
  1111→            paths: stage.cache.paths,
  1112→            restore: true,
  1113→            save: true,
  1114→          }
  1115→        : undefined,
  1116→      retry,
  1117→      timeoutMs,
  1118→      matrixValues,
  1119→      matrixIndex: matrixRows.length > 1 ? idx : undefined,
  1120→      matrixTotal: matrixRows.length > 1 ? matrixRows.length : undefined,
  1121→      pipelineRunId,
  1122→      stageId: stage.id,
  1123→      label: matrixRows.length > 1
  1124→        ? `${stage.name} (${Object.entries(matrixValues).map(([k, v]) => `${k}=${v}`).join(', ')})`
  1125→        : stage.name,
  1126→      requiresApproval: stage.requiresApproval,
  1127→    });
  1128→
  1129→    return { matrixValues, status: 'running' as const, runId };
  1130→  }));
  1131→
  1132→  // 6. Wait for all started runs to complete.
  1133→  for (const r of results) {
  1134→    if (r.runId) {
  1135→      runIds.push(r.runId);
  1136→      const finalStatus = await waitForRunCompletion(r.runId);
  1137→      if (finalStatus === 'failed') stageFailed = true;
  1138→      if (finalStatus === 'canceled') stageCanceled = true;
  1139→    } else if (r.status === 'skipped') {
  1140→      // Per-row skip — counts as not-failed.
  1141→    }
  1142→  }
  1143→
  1144→  // 7. Retry policy: per-step retry is handled by `startRunExtended`
  1145→    //    (which retries only the failing step). The previous whole-stage
  1146→    //    retry loop here was removed because it caused a quadratic
  1147→    //    (retry+1)² explosion — see the comment in step 5 above.
  1148→
  1149→  // 8. Update the StageRun row.
  1150→  const finalStageStatus = stageCanceled
  1151→    ? 'canceled'
  1152→    : stageFailed
  1153→      ? 'failed'
  1154→      : 'success';
  1155→  await db.stageRun.update({
  1156→    where: { id: stageRun.id },
  1157→    data: {
  1158→      status: finalStageStatus,
  1159→      finishedAt: new Date(),
  1160→      runIds: JSON.stringify(runIds),
  1161→      matrixValues: matrixRows.length > 1 ? JSON.stringify(matrixRows) : null,
  1162→    },
  1163→  });
  1164→  stageResults[stage.id] = finalStageStatus;
  1165→}
  1166→
  1167→// `mapRunStatusToStageStatus` was removed — the previous identity
  1168→// function was dead code. Run status strings and stage status strings
  1169→// use the same vocabulary (running/success/failed/canceled).
  1170→
  1171→/**
  1172→ * Poll the DB every 1s until the run reaches a terminal status.
  1173→ * Returns the terminal status. Also handles approval waiting — the run
  1174→ * is in `waiting_approval` until approved/rejected, which we treat as a
  1175→ * non-terminal status here (we keep waiting).
  1176→ */
  1177→async function waitForRunCompletion(runId: string): Promise<RunStatus> {
  1178→  // Safety cap: 24 hours.
  1179→  const deadline = Date.now() + 24 * 60 * 60 * 1000;
  1180→  while (Date.now() < deadline) {
  1181→    const run = await db.run.findUnique({ where: { id: runId }, select: { status: true } });
  1182→    if (!run) return 'failed';
  1183→    const status = run.status as RunStatus;
  1184→    if (status === 'running' || status === 'queued' || status === 'waiting_approval') {
  1185→      await sleep(1000);
  1186→      continue;
  1187→    }
  1188→    return status;
  1189→  }
  1190→  return 'failed';
  1191→}
  1192→
  1193→function sleep(ms: number): Promise<void> {
  1194→  return new Promise(resolve => setTimeout(resolve, ms));
  1195→}

  1197→async function finishPipelineRun(pipelineRunId: string, status: 'success' | 'failed' | 'canceled'): Promise<void> {
  1198→  await db.pipelineRun.update({
  1199→    where: { id: pipelineRunId },
  1200→    data: { status, finishedAt: new Date() },
  1201→  });
  1202→  // Best-effort pipeline-level notification (uses the first run's project).
  1203→  try {
  1204→    const pipelineRun = await db.pipelineRun.findUnique({
  1205→      where: { id: pipelineRunId },
  1206→      include: { pipeline: true },
  1207→    });
  1208→    if (pipelineRun) {
  1209→      const { notifyRunEvent } = await import('./notifications');
  1210→      // Find any run associated with this pipeline run to use as the
  1211→      // notification subject. If none, skip.
  1212→      const firstRun = await db.run.findFirst({
  1213→        where: { pipelineRunId },
  1214→        orderBy: { startedAt: 'asc' },
  1215→      });
  1216→      if (firstRun) {
  1217→        // Update the run's status if it doesn't match (rare), then fire.
  1218→        await notifyRunEvent(firstRun.id, status);
  1219→      }
  1220→    }
  1221→  } catch {
  1222→    // Notifications are best-effort.
  1223→  }
  1224→}

  1311→export async function cancelPipelineRun(pipelineRunId: string): Promise<void> {
  1312→  const pipelineRun = await db.pipelineRun.findUnique({
  1313→    where: { id: pipelineRunId },
  1314→    include: { stageRuns: true },
  1315→  });
  1316→  if (!pipelineRun) throw new Error(`Pipeline run ${pipelineRunId} not found`);
  1317→
  1318→  // Cancel all in-progress runs in this pipeline.
  1319→  const runs = await db.run.findMany({
  1320→    where: {
  1321→      pipelineRunId,
  1322→      status: { in: ['running', 'queued', 'waiting_approval'] },
  1323→    },
  1324→  });
  1325→  for (const run of runs) {
  1326→    try {
  1327→      await cancelRun(run.id);
  1328→    } catch {
  1329→      // Best-effort — continue canceling others.
  1330→    }
  1331→  }
  1332→
  1333→  // Mark all pending/running stages as canceled.
  1334→  for (const stage of pipelineRun.stageRuns) {
  1335→    if (stage.status === 'pending' || stage.status === 'running') {
  1336→      await db.stageRun.update({
  1337→        where: { id: stage.id },
  1338→        data: { status: 'canceled', finishedAt: new Date() },
  1339→      });
  1340→    }
  1341→  }
  1342→
  1343→  // Set pipeline run status to canceled.
  1344→  await db.pipelineRun.update({
  1345→    where: { id: pipelineRunId },
  1346→    data: { status: 'canceled', finishedAt: new Date() },
  1347→  });
  1348→}
