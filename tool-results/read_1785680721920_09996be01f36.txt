     1→// ============================================================
     2→// Forge — first-class GitHub client
     3→// ============================================================
     4→// Wraps the GitHub REST API via `octokit`. Replaces the hand-rolled
     5→// `fetch('https://api.github.com/...')` helpers that previously lived
     6→// only inside the Experiments Lab (ghFetch / createFixPR / etc.) and
     7→// were unreachable from the main app.
     8→//
     9→// Credentials are read from the global Forge settings store
    10→// (`.forge-settings.json`, AES-256-GCM) — the same store the
    11→// `github-settings.tsx` panel writes to. Owner/Repo come from the
    12→// project's `repoUrl` (parsed) OR from the global GITHUB_OWNER /
    13→// GITHUB_REPO plain settings (backwards-compat with experiments).
    14→//
    15→// Every function is project-aware: given a projectId it resolves the
    16→// GitHub token + owner + repo once and returns a typed result.
    17→// Failures throw a `GitHubError` with the upstream message.
    18→// ============================================================
    19→
    20→import * as fs from "node:fs";
    21→import * as path from "node:path";
    22→import * as crypto from "node:crypto";
    23→import { db } from "@/lib/db";

    25→
    26→// Octokit is loaded lazily (only when getOctokit() is actually called) so
    27→// that simply importing this module — which the engine does on every run
    28→// via github-feedback — doesn't pull in the (heavy) Octokit bundle. This
    29→// keeps the run path lightweight when GitHub isn't configured.
    30→type OctokitInstance = InstanceType<typeof import("octokit").Octokit>;
    31→let _OctokitCtor: (new (opts: { auth: string; request: { fetch: typeof fetch } }) => OctokitInstance) | null = null;
    32→async function loadOctokit(): Promise<new (opts: { auth: string; request: { fetch: typeof fetch } }) => OctokitInstance> {
    33→  if (_OctokitCtor) return _OctokitCtor;
    34→  const mod = await import("octokit");
    35→  _OctokitCtor = mod.Octokit;
    36→  return _OctokitCtor;
    37→}

    49→export class GitHubError extends Error {
    50→  status: number;
    51→  constructor(message: string, status: number) {
    52→    super(message);
    53→    this.name = "GitHubError";
    54→    this.status = status;
    55→  }
    56→}
    57→
    58→/**
    59→ * Map any error thrown by the GitHub client (our GitHubError OR Octokit's
    60→ * RequestError, which has `name === "HttpError"` and a `.status` number)
    61→ * to a { message, status } pair that API routes can turn into an HTTP
    62→ * response. Without this, every Octokit 404/422/403 collapses to 500.
    63→ *
    64→ * Octokit RequestError also exposes `.response.data` (the GitHub error
    65→ * body) — we extract the human-readable message from it when present.
    66→ */
    67→export function mapGitHubError(err: unknown): { message: string; status: number } {
    68→  // Our own typed error.
    69→  if (err instanceof GitHubError) {
    70→    return { message: err.message, status: err.status };
    71→  }
    72→  // Octokit RequestError (name "HttpError", has .status, may have .response.data).
    73→  if (err instanceof Error && "status" in err && typeof (err as { status: unknown }).status === "number") {
    74→    const status = (err as { status: number }).status;
    75→    const resp = (err as { response?: { data?: unknown } }).response;
    76→    let message = err.message;
    77→    if (resp?.data && typeof resp.data === "object") {
    78→      const data = resp.data as { message?: string; errors?: Array<{ message?: string }> };
    79→      if (data.message) message = data.message;
    80→      else if (Array.isArray(data.errors) && data.errors[0]?.message) message = data.errors[0].message;
    81→    }
    82→    return { message, status: status || 500 };
    83→  }
    84→  // Unknown error.
    85→  return {
    86→    message: err instanceof Error ? err.message : String(err),
    87→    status: 500,
    88→  };
    89→}

   133→// ---------------------------------------------------------------------------
   134→// Settings decryption (mirrors settings/route.ts + experiments/definitions.ts)
   135→// ---------------------------------------------------------------------------
   136→
   137→// Use the shared key helper so the production guard (throw if unset)
   138→// applies here too — not just in settings/route.ts.
   139→const ENCRYPTION_KEY = getSettingsEncryptionKeyString();
   140→const SETTINGS_FILE = path.join(process.cwd(), ".forge-settings.json");
   141→
   142→interface StoredSecret {
   143→  ciphertext: string;
   144→  iv: string;
   145→  tag: string;
   146→}
   147→
   148→function decryptSecret(s: StoredSecret): string {
   149→  const decipher = crypto.createDecipheriv(
   150→    "aes-256-gcm",
   151→    Buffer.from(ENCRYPTION_KEY, "base64"),
   152→    Buffer.from(s.iv, "base64"),
   153→  );
   154→  decipher.setAuthTag(Buffer.from(s.tag, "base64"));
   155→  return (
   156→    decipher.update(s.ciphertext, "base64", "utf8") + decipher.final("utf8")
   157→  );
   158→}
   159→
   160→// Cache the parsed settings file (with mtime check) so we don't hit the
   161→// filesystem + decrypt on every single GitHub API call. The settings file
   162→// changes rarely; re-reading only when its mtime moves.
   163→let _settingsCache: { mtime: number; data: { secrets: Record<string, StoredSecret>; plain: Record<string, string> } } | null = null;
   164→
   165→function loadSettings(): { secrets: Record<string, StoredSecret>; plain: Record<string, string> } {
   166→  try {
   167→    if (!fs.existsSync(SETTINGS_FILE)) {
   168→      _settingsCache = null;
   169→      return { secrets: {}, plain: {} };
   170→    }
   171→    const stat = fs.statSync(SETTINGS_FILE);
   172→    if (_settingsCache && _settingsCache.mtime === stat.mtimeMs) {
   173→      return _settingsCache.data;
   174→    }
   175→    const data = JSON.parse(fs.readFileSync(SETTINGS_FILE, "utf-8"));
   176→    _settingsCache = { mtime: stat.mtimeMs, data };
   177→    return data;
   178→  } catch {
   179→    _settingsCache = null;
   180→    return { secrets: {}, plain: {} };
   181→  }
   182→}
   183→
   184→function getSetting(key: string): string | null {
   185→  const s = loadSettings();
   186→  if (s.secrets?.[key]) {
   187→    try {
   188→      return decryptSecret(s.secrets[key]);
   189→    } catch {
   190→      return null;
   191→    }
   192→  }
   193→  if (s.plain?.[key]) return s.plain[key];
   194→  return process.env[key] ?? null;
   195→}

   197→/**
   198→ * Validate a git branch/ref name against GitHub's rules.
   199→ * Rejects: empty, >255 chars, leading '-', '..', '~', ':', '?', '*',
   200→ * '[', '\', spaces, control chars. Returns an error message or null.
   201→ *
   202→ * (Pre-empts GitHub 422s with cryptic messages by giving the user a
   203→ * clear validation error upfront.)
   204→ */
   205→export function validateBranchName(name: string): string | null {
   206→  if (!name || name.length === 0) return "Branch name is required";
   207→  if (name.length > 255) return "Branch name too long (max 255 chars)";
   208→  if (name.startsWith("-")) return "Branch name cannot start with '-'";
   209→  if (name.includes("..")) return "Branch name cannot contain '..'";
   210→  if (name.includes("~")) return "Branch name cannot contain '~'";
   211→  if (name.includes(":")) return "Branch name cannot contain ':'";
   212→  if (name.includes(" ")) return "Branch name cannot contain spaces";
   213→  if (/[?*\[\\]/.test(name)) return "Branch name cannot contain '?', '*', '[', or '\\'";
   214→  if (/[\x00-\x1f]/.test(name)) return "Branch name cannot contain control characters";
   215→  return null;
   216→}

   222→/**
   223→ * Parse owner + repo from a GitHub URL.
   224→ *   https://github.com/owner/repo.git          -> {owner,repo}
   225→ *   git@github.com:owner/repo.git              -> {owner,repo}
   226→ *   https://github.com/owner/repo              -> {owner,repo}
   227→ * Returns null if not a github URL or malformed.
   228→ */
   229→export function parseOwnerRepoFromUrl(repoUrl: string): { owner: string; repo: string } | null {
   230→  if (!repoUrl) return null;
   231→  // SSH form: git@github.com:owner/repo.git
   232→  const ssh = repoUrl.match(/git@github\.com:([^/]+)\/([^/]+?)(?:\.git)?$/i);
   233→  if (ssh) return { owner: ssh[1], repo: ssh[2] };
   234→  // HTTPS form
   235→  const https = repoUrl.match(/github\.com[/:]([^/]+)\/([^/]+?)(?:\.git)?(?:\/|$)/i);
   236→  if (https) return { owner: https[1], repo: https[2] };
   237→  return null;
   238→}

   240→// ---------------------------------------------------------------------------
   241→// Credentials resolution
   242→// ---------------------------------------------------------------------------
   243→
   244→/**
   245→ * Resolve GitHub credentials for a given project. Reads:
   246→ *   • token  — global GITHUB_TOKEN setting (or env)
   247→ *   • owner/repo — from the project's repoUrl (parsed), falling back to
   248→ *     the global GITHUB_OWNER / GITHUB_REPO settings (backwards-compat
   249→ *     with the experiments lab).
   250→ *
   251→ * Returns null if no token is configured, or if owner/repo can't be
   252→ * resolved from either the project's repoUrl or the global settings.
   253→ */
   254→export async function getProjectCreds(projectId: string): Promise<GitHubCreds | null> {
   255→  const token = getSetting("GITHUB_TOKEN");
   256→  if (!token) return null;
   257→
   258→  const project = await db.project.findUnique({
   259→    where: { id: projectId },
   260→    select: { repoUrl: true, repoProvider: true },
   261→  });
   262→
   263→  let owner: string | null = null;
   264→  let repo: string | null = null;
   265→
   266→  if (project?.repoUrl) {
   267→    const parsed = parseOwnerRepoFromUrl(project.repoUrl);
   268→    if (parsed) {
   269→      owner = parsed.owner;
   270→      repo = parsed.repo;
   271→    }
   272→  }
   273→  if (!owner) owner = getSetting("GITHUB_OWNER");
   274→  if (!repo) repo = getSetting("GITHUB_REPO");
   275→
   276→  if (!owner || !repo) return null;
   277→  return { token, owner, repo };
   278→}

   280→/**
   281→ * Get a cached Octokit instance for a project. Returns null if creds
   282→ * are missing (so callers can no-op gracefully).
   283→ */
   284→export async function getOctokit(projectId: string): Promise<{ octokit: OctokitInstance; creds: GitHubCreds } | null> {
   285→  const creds = await getProjectCreds(projectId);
   286→  if (!creds) return null;
   287→  const OctokitCtor = await loadOctokit();
   288→  const octokit = new OctokitCtor({
   289→    auth: creds.token,
   290→    request: { fetch },
   291→  });
   292→  return { octokit, creds };
   293→}

   337→/**
   338→ * Create a new branch off the HEAD of `base` (defaults to the repo's
   339→ * default branch — read dynamically, NOT hardcoded to 'main').
   340→ */
   341→export async function createBranch(
   342→  octokit: OctokitInstance,
   343→  creds: GitHubCreds,
   344→  name: string,
   345→  base?: string,
   346→): Promise<{ ref: string; sha: string }> {
   347→  const baseBranch = base ?? (await getDefaultBranch(octokit, creds));
   348→  const { data: ref } = await octokit.rest.git.getRef({
   349→    owner: creds.owner,
   350→    repo: creds.repo,
   351→    ref: `heads/${baseBranch}`,
   352→  });
   353→  const sha = ref.object.sha;
   354→  const { data: newRef } = await octokit.rest.git.createRef({
   355→    owner: creds.owner,
   356→    repo: creds.repo,
   357→    ref: `refs/heads/${name}`,
   358→    sha,
   359→  });
   360→  return { ref: newRef.ref, sha };
   361→}

   367→export async function commitFileViaContentsApi(
   368→  octokit: OctokitInstance,
   369→  creds: GitHubCreds,
   370→  args: { path: string; content: string; branch: string; message: string },
   371→): Promise<{ commitSha: string; commitUrl: string }> {
   372→  const { data } = await octokit.rest.repos.createOrUpdateFileContents({
   373→    owner: creds.owner,
   374→    repo: creds.repo,
   375→    path: args.path,
   376→    message: args.message,
   377→    content: Buffer.from(args.content).toString("base64"),
   378→    branch: args.branch,
   379→  });
   380→  return {
   381→    commitSha: data.commit?.sha ?? "",
   382→    commitUrl: data.commit?.html_url ?? "",
   383→  };
   384→}

   386→/**
   387→ * Commit multiple files to a branch (each via Contents API) and return
   388→ * the last commit's sha. For large diffs prefer `git.ts.pushBranch`.
   389→ */
   390→export async function commitFilesViaContentsApi(
   391→  octokit: OctokitInstance,
   392→  creds: GitHubCreds,
   393→  files: Array<{ path: string; content: string }>,
   394→  branch: string,
   395→  message: string,
   396→): Promise<{ lastCommitSha: string | null; commitCount: number }> {
   397→  let lastSha: string | null = null;
   398→  let count = 0;
   399→  for (const file of files) {
   400→    const r = await commitFileViaContentsApi(octokit, creds, {
   401→      path: file.path,
   402→      content: file.content,
   403→      branch,
   404→      message: count === 0 ? message : `${message} (${count + 1}/${files.length})`,
   405→    });
   406→    lastSha = r.commitSha;
   407→    count++;
   408→  }
   409→  return { lastCommitSha: lastSha, commitCount: count };
   410→}

   599→/**
   600→ * Download a workflow run's logs. Returns a ReadableStream<Uint8Array>
   601→ * that the caller can pipe straight to an HTTP response — no buffering
   602→ * into RAM. Throws GitHubError if GitHub returns a non-2xx status.
   603→ *
   604→ * (Streaming matters: log ZIPs can be 100+ MB, and buffering them
   605→ * would OOM the server under concurrent downloads.)
   606→ */
   607→export async function downloadWorkflowRunLogs(
   608→  octokit: OctokitInstance,
   609→  creds: GitHubCreds,
   610→  runId: number,
   611→): Promise<{ stream: ReadableStream<Uint8Array>; size: number | null }> {
   612→  void octokit; // creds+token used directly; octokit kept for API consistency
   613→  const url = `https://api.github.com/repos/${creds.owner}/${creds.repo}/actions/runs/${runId}/logs`;
   614→  const resp = await fetch(url, {
   615→    headers: {
   616→      Authorization: `Bearer ${creds.token}`,
   617→      Accept: "application/vnd.github.v3+json",
   618→    },
   619→  });
   620→  if (!resp.ok) {
   621→    throw new GitHubError(`Failed to download logs (${resp.status})`, resp.status);
   622→  }
   623→  if (!resp.body) {
   624→    throw new GitHubError("GitHub returned no log body", 502);
   625→  }
   626→  const size = resp.headers.get("content-length");
   627→  return {
   628→    stream: resp.body as ReadableStream<Uint8Array>,
   629→    size: size ? parseInt(size, 10) : null,
   630→  };
   631→}

   709→// Webhook signature verification (GitHub standard: x-hub-signature-256)
   710→// ---------------------------------------------------------------------------

   712→/**
   713→ * Verify a GitHub webhook signature. GitHub sends `x-hub-signature-256`
   714→ * as `sha256=<hex>`. Constant-time comparison.
   715→ */
   716→export function verifyGitHubWebhookSignature(
   717→  payload: string | Buffer,
   718→  signature: string,
   719→  secret: string,
   720→): boolean {
   721→  const payloadBuf = typeof payload === "string" ? Buffer.from(payload, "utf8") : payload;
   722→  const expected = crypto.createHmac("sha256", secret).update(payloadBuf).digest("hex");
   723→  const sig = signature.startsWith("sha256=") ? signature.slice("sha256=".length) : signature;
   724→  const expectedBuf = Buffer.from(expected, "hex");
   725→  const sigBuf = Buffer.from(sig, "hex");
   726→  if (expectedBuf.length !== sigBuf.length || sigBuf.length === 0) return false;
   727→  try {
   728→    return crypto.timingSafeEqual(expectedBuf, sigBuf);
   729→  } catch {
   730→    return false;
   731→  }
   732→}

   734→/**
   735→ * Parse a GitHub webhook payload into a normalized event descriptor.
   736→ * Returns null for events we don't handle.
   737→ */
   738→export interface GitHubWebhookEvent {
   739→  type: string; // push | pull_request | check_run | workflow_run | ...
   740→  ref?: string; // branch ref (push)
   741→  prNumber?: number;
   742→  prAction?: string; // opened | synchronize | closed | reopened
   743→  headSha?: string;
   744→  repoFullName?: string;
   745→  sender?: string;
   746→}
   747→
   748→export function parseGitHubWebhookEvent(
   749→  eventType: string,
   750→  body: unknown,
   751→): GitHubWebhookEvent | null {
   752→  const b = body as Record<string, unknown>;
   753→  const repo = b.repository as { full_name?: string } | undefined;
   754→  const sender = b.sender as { login?: string } | undefined;
   755→  const base: GitHubWebhookEvent = {
   756→    type: eventType,
   757→    repoFullName: repo?.full_name,
   758→    sender: sender?.login,
   759→  };
   760→  switch (eventType) {
   761→    case "push":
   762→      return { ...base, ref: b.ref as string | undefined };
   763→    case "pull_request": {
   764→      const pr = b.pull_request as { number?: number; head?: { sha?: string }; action?: string } | undefined;
   765→      return {
   766→        ...base,
   767→        prNumber: pr?.number,
   768→        prAction: pr?.action,
   769→        headSha: pr?.head?.sha,
   770→      };
   771→    }
   772→    case "check_run":
   773→    case "workflow_run": {
   774→      const cr = b[eventType] as { head_sha?: string } | undefined;
   775→      return { ...base, headSha: cr?.head_sha };
   776→    }
   777→    case "ping":
   778→      return { ...base, type: "ping" };
   779→    default:
   780→      return base;
   781→  }
   782→}
