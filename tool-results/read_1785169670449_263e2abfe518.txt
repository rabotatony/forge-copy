     3→     3→// ============================================================
     4→     4→// Forge — Experiments Lab
     5→     5→// ============================================================
     6→     6→// A sandboxed page where hard, monitored experiments run on
     7→     7→// Forge itself to discover development breakthroughs. Each
     8→     8→// experiment has a hypothesis, runs in isolation with hard
     9→     9→// timeouts + output caps, and produces a verdict:
    10→    10→//   BREAKTHROUGH → can be promoted to a permanent workflow
    11→    11→//   NO_CHANGE    → ran fine, no breakthrough
    12→    12→//   REGRESSION   → made things worse
    13→    13→//
    14→    14→// Safety rails are enforced in the engine (engine.ts). The UI
    15→    15→// only triggers runs and displays results — it cannot bypass
    16→    16→// the sandbox.
    17→    17→// ============================================================
    18→    18→

    20→    20→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";

    43→    43→import {

    57→    57→// ---------------------------------------------------------------------------
    58→    58→// Types — mirror the API response shapes
    59→    59→// ---------------------------------------------------------------------------
    60→    60→
    61→    61→type Verdict = "BREAKTHROUGH" | "NO_CHANGE" | "REGRESSION";
    62→    62→type Category = "self-improvement" | "tournament" | "synthesis" | "adversarial" | "recursive";
    63→    63→
    64→    64→interface LatestRun {
    65→    65→  id: string;
    66→    66→  status: string;
    67→    67→  verdict: Verdict | null;
    68→    68→  verdictReason: string | null;
    69→    69→  promoted: boolean;
    70→    70→  startedAt: string;
    71→    71→  completedAt: string | null;
    72→    72→  metrics: string | null;
    73→    73→}
    74→    74→
    75→    75→interface ExperimentListItem {
    76→    76→  slug: string;
    77→    77→  name: string;
    78→    78→  category: Category;
    79→    79→  hypothesis: string;
    80→    80→  procedure: string;
    81→    81→  dangerLevel: "safe" | "moderate" | "aggressive";
    82→    82→  dbId: string | null;
    83→    83→  totalRuns: number;
    84→    84→  latestRun: LatestRun | null;
    85→    85→}
    86→    86→
    87→    87→interface ExperimentsResponse {
    88→    88→  experiments: ExperimentListItem[];
    89→    89→  stats: { totalExperiments: number; byCategory: Record<string, number> };
    90→    90→  verdictCounts: Record<string, number>;
    91→    91→  promotedCount: number;
    92→    92→}
    93→    93→
    94→    94→interface RunDetail {
    95→    95→  run: {
    96→    96→    id: string;
    97→    97→    status: string;
    98→    98→    verdict: Verdict | null;
    99→    99→    verdictReason: string | null;
   100→   100→    promoted: boolean;
   101→   101→    promotedWorkflowId: string | null;
   102→   102→    promotedPresetId: string | null;
   103→   103→    startedAt: string;
   104→   104→    completedAt: string | null;
   105→   105→    metrics: Record<string, number | string | boolean> | null;
   106→   106→    evidence: { summary: string; steps: Array<{ step: string; detail: unknown; t?: number }> } | null;
   107→   107→  };
   108→   108→  experiment: {
   109→   109→    slug: string;
   110→   110→    name: string;
   111→   111→    category: Category;
   112→   112→    hypothesis: string;
   113→   113→    procedure: string;
   114→   114→    dangerLevel: string;
   115→   115→  };
   116→   116→}
   117→   117→

   119→   119→// Category metadata

   121→   121→

   123→   123→  "self-improvement": {
   124→   124→    icon: TrendingUp,
   125→   125→    label: "Self-Improvement",
   126→   126→    color: "text-emerald-600 dark:text-emerald-400",
   127→   127→    desc: "Can Forge's AI generate a faster version of an existing script?",
   128→   128→  },
   129→   129→  tournament: {
   130→   130→    icon: Swords,
   131→   131→    label: "Tournament",
   132→   132→    color: "text-amber-600 dark:text-amber-400",
   133→   133→    desc: "Generate the same tool in 3 languages, benchmark, pick the winner.",
   134→   134→  },
   135→   135→  synthesis: {
   136→   136→    icon: Sparkles,
   137→   137→    label: "Synthesis",
   138→   138→    color: "text-violet-600 dark:text-violet-400",
   139→   139→    desc: "Fill a capability gap by generating a tool from a description.",
   140→   140→  },
   141→   141→  adversarial: {
   142→   142→    icon: Shield,
   143→   143→    label: "Adversarial",
   144→   144→    color: "text-rose-600 dark:text-rose-400",
   145→   145→    desc: "Generate malformed inputs, find crashes, synthesize a hardened version.",
   146→   146→  },
   147→   147→  recursive: {
   148→   148→    icon: GitBranch,
   149→   149→    label: "Recursive",
   150→   150→    color: "text-sky-600 dark:text-sky-400",
   151→   151→    desc: "Chain AI-generated scripts N levels deep, feed output → input.",
   152→   152→  },
   153→   153→};
   154→   154→
   155→   155→const VERDICT_META: Record<Verdict, { icon: typeof CheckCircle2; label: string; color: string; bg: string }> = {
   156→   156→  BREAKTHROUGH: {
   157→   157→    icon: Trophy,
   158→   158→    label: "Breakthrough",
   159→   159→    color: "text-emerald-700 dark:text-emerald-300",
   160→   160→    bg: "bg-emerald-500/10 border-emerald-500/30",

   168→   168→  REGRESSION: {
   169→   169→    icon: XCircle,
   170→   170→    label: "Regression",
   171→   171→    color: "text-rose-700 dark:text-rose-300",
   172→   172→    bg: "bg-rose-500/10 border-rose-500/30",
   173→   173→  },
   174→   174→};
   175→   175→
   176→   176→const DANGER_META: Record<string, { label: string; color: string }> = {
   177→   177→  safe: { label: "Safe", color: "text-emerald-600 dark:text-emerald-400" },
   178→   178→  moderate: { label: "Moderate", color: "text-amber-600 dark:text-amber-400" },
   179→   179→  aggressive: { label: "Aggressive", color: "text-rose-600 dark:text-rose-400" },
   180→   180→};
   181→   181→
   182→   182→// ---------------------------------------------------------------------------
   183→   183→// Main component
   184→   184→// ---------------------------------------------------------------------------
   185→   185→
   186→   186→export function ExperimentsLab() {
   187→   187→  const queryClient = useQueryClient();
   188→   188→  const [expandedSlug, setExpandedSlug] = useState<string | null>(null);
   189→   189→  const [selectedRunId, setSelectedRunId] = useState<string | null>(null);
   190→   190→
   191→   191→  const { data, isLoading } = useQuery<ExperimentsResponse>({
   192→   192→    queryKey: ["experiments"],
   193→   193→    queryFn: async () => {
   194→   194→      const r = await fetch("/api/forge/experiments");
   195→   195→      if (!r.ok) throw new Error("Failed to load experiments");
   196→   196→      return r.json();
   197→   197→    },
   198→   198→    refetchInterval: 5000,
   199→   199→  });
   200→   200→
   201→   201→  const runMutation = useMutation({
   202→   202→    mutationFn: async (slug: string) => {
   203→   203→      const r = await fetch(`/api/forge/experiments/${slug}/run`, { method: "POST" });
   204→   204→      if (!r.ok) {
   205→   205→        const e = await r.json().catch(() => ({}));
   206→   206→        throw new Error(e.error ?? "Run failed");
   207→   207→      }
   208→   208→      return r.json() as Promise<{ runId: string; verdict: Verdict | null; status: string }>;
   209→   209→    },
   210→   210→    onSuccess: (result, slug) => {
   211→   211→      queryClient.invalidateQueries({ queryKey: ["experiments"] });
   212→   212→      setSelectedRunId(result.runId);
   213→   213→      setExpandedSlug(slug);
   214→   214→      if (result.verdict === "BREAKTHROUGH") {
   215→   215→        toast.success("Breakthrough discovered!", {
   216→   216→          description: "The experiment found a real improvement. You can promote it.",
   217→   217→        });
   218→   218→      } else if (result.verdict === "REGRESSION") {
   219→   219→        toast.error("Experiment regressed", {
   220→   220→          description: result.status === "timeout" ? "The experiment timed out." : "See the run details.",
   221→   221→        });
   222→   222→      } else {
   223→   223→        toast.info("Experiment completed", { description: "No breakthrough this time." });
   224→   224→      }
   225→   225→    },
   226→   226→    onError: (err: Error) => {
   227→   227→      toast.error("Experiment failed", { description: err.message });
   228→   228→    },
   229→   229→  });
   230→   230→
   231→   231→  const promoteMutation = useMutation({
   232→   232→    mutationFn: async (runId: string) => {
   233→   233→      const r = await fetch(`/api/forge/experiments/runs/${runId}?action=promote`, { method: "POST" });
   234→   234→      if (!r.ok) {
   235→   235→        const e = await r.json().catch(() => ({}));
   236→   236→        throw new Error(e.error ?? "Promotion failed");
   237→   237→      }
   238→   238→      return r.json();
   239→   239→    },
   240→   240→    onSuccess: () => {
   241→   241→      queryClient.invalidateQueries({ queryKey: ["experiments"] });
   242→   242→      queryClient.invalidateQueries({ queryKey: ["experiment-run"] });
   243→   243→      toast.success("Promoted to permanent workflow", {
   244→   244→        description: "The breakthrough is now a reusable Forge workflow + preset.",
   245→   245→      });
   246→   246→    },
   247→   247→    onError: (err: Error) => toast.error("Promotion failed", { description: err.message }),
   248→   248→  });
   249→   249→
   250→   250→  const totalRuns = data
   251→   251→    ? data.experiments.reduce((sum, e) => sum + e.totalRuns, 0)
   252→   252→    : 0;
   253→   253→  const breakthroughs = data?.verdictCounts.BREAKTHROUGH ?? 0;
   254→   254→
   255→   255→  return (
   256→   256→    <div className="space-y-6">
   257→   257→      {/* Header */}
   258→   258→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   259→   259→        <div className="flex items-center gap-3">
   260→   260→          <span className="flex size-10 items-center justify-center rounded-lg bg-violet-500/10 text-violet-600 dark:text-violet-400">
   261→   261→            <FlaskConical className="size-5" aria-hidden />
   262→   262→          </span>
   263→   263→          <div>
   264→   264→            <h1 className="text-xl font-semibold tracking-tight">Experiments Lab</h1>
   265→   265→            <p className="text-sm text-muted-foreground">
   266→   266→              Sandboxed breakthrough discovery — Forge runs hard experiments on itself.
   267→   267→            </p>
   268→   268→          </div>
   269→   269→        </div>
   270→   270→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   271→   271→          <Shield className="size-3.5 text-emerald-500" aria-hidden />
   272→   272→          <span>Sandboxed · time-capped · output-limited</span>
   273→   273→        </div>
   274→   274→      </div>
   275→   275→
   276→   276→      {/* Stats row */}
   277→   277→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   278→   278→        <StatCard
   279→   279→          icon={FlaskConical}
   280→   280→          label="Experiments"
   281→   281→          value={data?.stats.totalExperiments ?? 5}
   282→   282→          color="text-violet-600 dark:text-violet-400"
   283→   283→        />
   284→   284→        <StatCard
   285→   285→          icon={Activity}
   286→   286→          label="Total runs"
   287→   287→          value={totalRuns}
   288→   288→          color="text-sky-600 dark:text-sky-400"
   289→   289→        />
   290→   290→        <StatCard
   291→   291→          icon={Trophy}
   292→   292→          label="Breakthroughs"
   293→   293→          value={breakthroughs}
   294→   294→          color="text-emerald-600 dark:text-emerald-400"
   295→   295→        />
   296→   296→        <StatCard
   297→   297→          icon={Rocket}
   298→   298→          label="Promoted"
   299→   299→          value={data?.promotedCount ?? 0}
   300→   300→          color="text-amber-600 dark:text-amber-400"
   301→   301→        />
   302→   302→      </div>
   303→   303→
   304→   304→      {/* Breakthrough rate bar */}
   305→   305→      {data && totalRuns > 0 && (
   306→   306→        <Card>
   307→   307→          <CardContent className="p-4">
   308→   308→            <div className="flex items-center justify-between text-xs text-muted-foreground">
   309→   309→              <span className="flex items-center gap-1.5">
   310→   310→                <TrendingUp className="size-3.5" aria-hidden />
   311→   311→                Breakthrough rate
   312→   312→              </span>
   313→   313→              <span className="font-mono">
   314→   314→                {breakthroughs}/{totalRuns} runs ({Math.round((breakthroughs / totalRuns) * 100)}%)
   315→   315→              </span>
   316→   316→            </div>
   317→   317→            <Progress
   318→   318→              value={(breakthroughs / totalRuns) * 100}
   319→   319→              className="mt-2 h-2"
   320→   320→            />
   321→   321→            <div className="mt-2 flex flex-wrap gap-2 text-[11px]">
   322→   322→              {(["BREAKTHROUGH", "NO_CHANGE", "REGRESSION"] as Verdict[]).map((v) => (
   323→   323→                <Badge
   324→   324→                  key={v}
   325→   325→                  variant="outline"
   326→   326→                  className={cn("gap-1 font-mono", VERDICT_META[v].color)}
   327→   327→                >
   328→   328→                  {VERDICT_META[v].label}: {data.verdictCounts[v] ?? 0}
   329→   329→                </Badge>
   330→   330→              ))}
   331→   331→            </div>
   332→   332→          </CardContent>
   333→   333→        </Card>
   334→   334→      )}
   335→   335→
   336→   336→      {/* Experiment cards */}
   337→   337→      <div className="space-y-3">
   338→   338→        {isLoading ? (
   339→   339→          <Card>
   340→   340→            <CardContent className="flex items-center justify-center p-12 text-sm text-muted-foreground">
   341→   341→              <Loader2 className="mr-2 size-4 animate-spin" aria-hidden />
   342→   342→              Loading experiments…
   343→   343→            </CardContent>
   344→   344→          </Card>
   345→   345→        ) : (
   346→   346→          data?.experiments.map((exp) => {
   347→   347→            const meta = CATEGORY_META[exp.category];
   348→   348→            const Icon = meta.icon;
   349→   349→            const isExpanded = expandedSlug === exp.slug;
   350→   350→            const isRunning = runMutation.isPending && runMutation.variables === exp.slug;
   351→   351→            const latest = exp.latestRun;
   352→   352→            const verdict = latest?.verdict;
   353→   353→            const vMeta = verdict ? VERDICT_META[verdict] : null;
   354→   354→            const danger = DANGER_META[exp.dangerLevel];
   355→   355→
   356→   356→            return (
   357→   357→              <Card key={exp.slug} className={cn("overflow-hidden transition-shadow", isExpanded && "shadow-md")}>
   358→   358→                <CardHeader className="pb-3">
   359→   359→                  <div className="flex items-start justify-between gap-3">
   360→   360→                    <div className="flex min-w-0 items-start gap-3">
   361→   361→                      <span className={cn("mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-md bg-muted", meta.color)}>
   362→   362→                        <Icon className="size-4" aria-hidden />
   363→   363→                      </span>
   364→   364→                      <div className="min-w-0">
   365→   365→                        <CardTitle className="flex flex-wrap items-center gap-2 text-base">
   366→   366→                          <span className="truncate">{exp.name}</span>
   367→   367→                          <Badge variant="outline" className="gap-1 text-[10px] font-normal">
   368→   368→                            <Icon className="size-2.5" aria-hidden />
   369→   369→                            {meta.label}
   370→   370→                          </Badge>
   371→   371→                          <Tooltip>
   372→   372→                            <TooltipTrigger asChild>
   373→   373→                              <Badge variant="outline" className={cn("gap-1 text-[10px] font-normal", danger.color)}>
   374→   374→                                <AlertTriangle className="size-2.5" aria-hidden />
   375→   375→                                {danger.label}
   376→   376→                              </Badge>
   377→   377→                            </TooltipTrigger>
   378→   378→                            <TooltipContent>Danger level: {exp.dangerLevel}</TooltipContent>
   379→   379→                          </Tooltip>
   380→   380→                        </CardTitle>
   381→   381→                        <CardDescription className="mt-1 line-clamp-2 text-xs">
   382→   382→                          {exp.hypothesis}
   383→   383→                        </CardDescription>
   384→   384→                      </div>
   385→   385→                    </div>
   386→   386→                    <div className="flex shrink-0 items-center gap-2">
   387→   387→                      <Button
   388→   388→                        size="sm"
   389→   389→                        variant="default"
   390→   390→                        disabled={isRunning}
   391→   391→                        onClick={() => runMutation.mutate(exp.slug)}
   392→   392→                        className="gap-1.5"
   393→   393→                      >
   394→   394→                        {isRunning ? (
   395→   395→                          <Loader2 className="size-3.5 animate-spin" aria-hidden />
   396→   396→                        ) : (
   397→   397→                          <Play className="size-3.5" aria-hidden />
   398→   398→                        )}
   399→   399→                        {isRunning ? "Running" : "Run"}
   400→   400→                      </Button>
   401→   401→                      <Button
   402→   402→                        size="sm"
   403→   403→                        variant="ghost"
   404→   404→                        onClick={() => setExpandedSlug(isExpanded ? null : exp.slug)}
   405→   405→                        aria-label={isExpanded ? "Collapse" : "Expand"}
   406→   406→                      >
   407→   407→                        {isExpanded ? <ChevronDown className="size-4" /> : <ChevronRight className="size-4" />}
   408→   408→                      </Button>
   409→   409→                    </div>
   410→   410→                  </div>
   411→   411→                </CardHeader>
   412→   412→
   413→   413→                {/* Latest verdict strip */}
   414→   414→                {latest && (
   415→   415→                  <CardContent className="pb-3 pt-0">
   416→   416→                    <div className={cn("flex flex-wrap items-center gap-2 rounded-md border px-3 py-2 text-xs", vMeta?.bg ?? "bg-muted/30 border-border")}>
   417→   417→                      {verdict && (
   418→   418→                        <span className={cn("flex items-center gap-1 font-medium", vMeta?.color)}>
   419→   419→                          {(() => {
   420→   420→                            const VIcon = vMeta!.icon;
   421→   421→                            return <VIcon className="size-3.5" aria-hidden />;
   422→   422→                          })()}
   423→   423→                          {vMeta?.label}
   424→   424→                        </span>
   425→   425→                      )}
   426→   426→                      <span className="text-muted-foreground">
   427→   427→                        {latest.verdictReason ?? `Status: ${latest.status}`}
   428→   428→                      </span>
   429→   429→                      <span className="ml-auto flex items-center gap-2 font-mono text-[10px] text-muted-foreground">
   430→   430→                        {latest.promoted && (
   431→   431→                          <Badge variant="outline" className="gap-1 text-[10px] text-emerald-600 dark:text-emerald-400">
   432→   432→                            <Rocket className="size-2.5" aria-hidden />
   433→   433→                            Promoted
   434→   434→                          </Badge>
   435→   435→                        )}
   436→   436→                        <Clock className="size-3" aria-hidden />
   437→   437→                        {new Date(latest.startedAt).toLocaleString()}
   438→   438→                      </span>
   439→   439→                    </div>
   440→   440→                  </CardContent>
   441→   441→                )}
   442→   442→
   443→   443→                {/* Expanded: procedure + runs */}
   444→   444→                {isExpanded && (
   445→   445→                  <CardContent className="border-t border-border pt-4">
   446→   446→                    <div className="grid gap-4 md:grid-cols-2">
   447→   447→                      <div>
   448→   448→                        <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   449→   449→                          Procedure
   450→   450→                        </h4>
   451→   451→                        <p className="text-xs leading-relaxed text-foreground/90">{exp.procedure}</p>
   452→   452→                      </div>
   453→   453→                      <div>
   454→   454→                        <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   455→   455→                          Recent runs
   456→   456→                        </h4>
   457→   457→                        <RecentRunsList slug={exp.slug} onSelect={setSelectedRunId} selectedRunId={selectedRunId} />
   458→   458→                      </div>
   459→   459→                    </div>
   460→   460→                  </CardContent>
   461→   461→                )}
   462→   462→              </Card>
   463→   463→            );
   464→   464→          })
   465→   465→        )}
   466→   466→      </div>
   467→   467→

   505→   505→function RecentRunsList({ slug, onSelect, selectedRunId }: { slug: string; onSelect: (id: string) => void; selectedRunId: string | null }) {
   506→   506→  const { data, isLoading } = useQuery<{ runs: LatestRun[] }>({

   514→   514→  });

   531→   531→          const vMeta = run.verdict ? VERDICT_META[run.verdict] : null;
   532→   532→          return (
   533→   533→            <button
   534→   534→              key={run.id}
   535→   535→              type="button"
   536→   536→              onClick={() => onSelect(run.id)}
   537→   537→              className={cn(
   538→   538→                "flex w-full items-center gap-2 rounded-md border px-2.5 py-1.5 text-left text-xs transition-colors hover:bg-accent",

   540→   540→              )}
   541→   541→            >
   542→   542→              {run.verdict ? (
   543→   543→                <span className={cn("flex size-4 shrink-0 items-center justify-center", vMeta?.color)}>
   544→   544→                  {(() => {
   545→   545→                    const VIcon = vMeta!.icon;
