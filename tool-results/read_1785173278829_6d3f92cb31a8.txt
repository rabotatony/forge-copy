    92→type Listener = (event: RunEvent) => void;
    93→const listeners = new Map<string, Set<Listener>>();
    94→const active = new Map<
    95→  string,
    96→  { child: import("node:child_process").ChildProcess; canceled: boolean }
    97→>();
    98→const seqCounter = new Map<string, number>();
    99→
   100→export function subscribe(runId: string, fn: Listener): () => void {
   101→  let set = listeners.get(runId);
   102→  if (!set) {
   103→    set = new Set();
   104→    listeners.set(runId, set);
   105→  }
   106→  set.add(fn);
   107→  return () => {
   108→    set!.delete(fn);
   109→    if (set!.size === 0) listeners.delete(runId);
   110→  };
   111→}
   112→
   113→export function emit(event: RunEvent): void {
   114→  const set = listeners.get(event.runId);
   115→  if (set) for (const fn of set) fn(event);
   116→}

   118→// Maximum log lines per run (prevents unbounded DB growth).
   119→const MAX_LOG_LINES_PER_RUN = 10_000;
   120→const logLineCounts = new Map<string, number>();
   121→const logTruncationWarned = new Set<string>();
   122→
   123→export async function appendLog(
   124→  runId: string,
   125→  stream: "stdout" | "stderr" | "system",
   126→  text: string,
   127→): Promise<void> {
   128→  const count = logLineCounts.get(runId) ?? 0;
   129→  // Once the limit is reached, only system messages continue (so the user can
   130→  // still see errors / status updates), and only one truncation warning is
   131→  // ever emitted per run.
   132→  if (count >= MAX_LOG_LINES_PER_RUN && stream !== "system") return;
   133→
   134→  // Truncate very long lines (prevents memory issues).
   135→  const truncated =
   136→    text.length > 10_000 ? text.slice(0, 10_000) + "…[truncated]" : text;
   137→
   138→  const seq = seqCounter.get(runId) ?? 0;
   139→  seqCounter.set(runId, seq + 1);
   140→  logLineCounts.set(runId, count + 1);
   141→  const ts = Date.now();
   142→  await db.logLine.create({ data: { runId, seq, stream, text: truncated } });
   143→  emit({ type: "log", runId, log: { seq, stream, text: truncated, ts } });
   144→
   145→  if (count === MAX_LOG_LINES_PER_RUN - 1 && !logTruncationWarned.has(runId)) {
   146→    logTruncationWarned.add(runId);
   147→    const warnSeq = seq + 1;
   148→    seqCounter.set(runId, warnSeq + 1);
   149→    logLineCounts.set(runId, count + 2);
   150→    const warnText = `⚠ Log limit reached (${MAX_LOG_LINES_PER_RUN} lines). Further stdout/stderr output suppressed.`;
   151→    await db.logLine.create({
   152→      data: { runId, seq: warnSeq, stream: "system", text: warnText },
   153→    });
   154→    emit({
   155→      type: "log",
   156→      runId,
   157→      log: { seq: warnSeq, stream: "system", text: warnText, ts: Date.now() },
   158→    });
   159→  }
   160→}

   198→async function cancelInprogressRuns(
   199→  projectId: string,
   200→  group?: string | null,
   201→): Promise<string[]> {
   202→  const where = group
   203→    ? { projectId, concurrencyGroup: group, status: "running" as const }
   204→    : { projectId, status: "running" as const };
   205→  const inProgress = await db.run.findMany({ where });
   206→  const canceled: string[] = [];
   207→  for (const run of inProgress) {
   208→    const entry = active.get(run.id);
   209→    if (entry) {
   210→      entry.canceled = true;
   211→      try {
   212→        entry.child.kill("SIGTERM");
   213→      } catch {
   214→        /* ignore */
   215→      }
   216→    }
   217→    await appendLog(
   218→      run.id,
   219→      "system",
   220→      group
   221→        ? `Run canceled: a newer run in group "${group}" was triggered.`
   222→        : "Run canceled: a newer run was triggered (concurrent cancellation).",
   223→    );
   224→    await finishRun(run.id, "canceled", 130);
   225→    canceled.push(run.id);
   226→  }
   227→  return canceled;
   228→}

   234→async function waitForApproval(runId: string): Promise<boolean> {
   235→  await db.approval.create({
   236→    data: {
   237→      runId,
   238→      status: "pending",
   239→      expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000),
   240→    },
   241→  });
   242→  emit({
   243→    type: "approval-required",
   244→    runId,
   245→    message: "Manual approval required to start this run.",
   246→  });
   247→  await appendLog(
   248→    runId,
   249→    "system",
   250→    "⏸ Manual approval required. Waiting for approval...",
   251→  );
   252→
   253→  // Poll for decision (every 2 seconds, up to 24 hours).
   254→  const deadline = Date.now() + 24 * 60 * 60 * 1000;
   255→  while (Date.now() < deadline) {
   256→    await sleep(2000);
   257→    const approval = await db.approval.findUnique({ where: { runId } });
   258→    if (!approval) return false;
   259→    if (approval.status === "approved") {
   260→      await appendLog(
   261→        runId,
   262→        "system",
   263→        `✓ Approved by ${approval.decidedBy ?? "unknown"}. Starting run...`,
   264→      );
   265→      return true;
   266→    }
   267→    if (approval.status === "rejected") {
   268→      await appendLog(
   269→        runId,
   270→        "system",
   271→        `✗ Rejected by ${approval.decidedBy ?? "unknown"}.`,
   272→      );
   273→      return false;
   274→    }
   275→    if (approval.status === "expired") return false;
   276→    // Check if run was canceled externally.
   277→    const run = await db.run.findUnique({ where: { id: runId } });
   278→    if (run?.status === "canceled") return false;
   279→  }
   280→  await db.approval.update({
   281→    where: { runId },
   282→    data: { status: "expired" },
   283→  });
   284→  await appendLog(runId, "system", "Approval timed out after 24 hours.");
   285→  return false;
   286→}

   311→// Public: start an extended run

   313→
   314→export async function startRunExtended(
   315→  options: RunOptions,
   316→): Promise<{ runId: string }> {
   317→  // Ensure background timers are running (idempotent — see bootstrap.ts).
   318→  ensureBootstrapped();
   319→
   320→  const project = await db.project.findUnique({
   321→    where: { id: options.projectId },
   322→  });
   323→  if (!project) throw new Error(`Project ${options.projectId} not found`);
   324→  const workflow = getWorkflow(options.workflow);
   325→  if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   326→
   327→  const detection = JSON.parse(project.detection) as Detection;
   328→
   329→  // Concurrency control (skip for pipeline-triggered runs — they're managed
   330→  // by the pipeline engine which already serializes stages).
   331→  if (options.trigger !== "pipeline") {
   332→    const settings = await db.projectSettings.findUnique({
   333→      where: { projectId: options.projectId },
   334→    });
   335→    const group = options.concurrencyGroup ?? settings?.concurrencyGroup ?? null;
   336→
   337→    if (group) {
   338→      const inProgressInGroup = await db.run.findMany({
   339→        where: {
   340→          projectId: options.projectId,
   341→          concurrencyGroup: group,
   342→          status: { in: ["running", "queued", "waiting_approval"] },
   343→        },
   344→      });
   345→
   346→      if (inProgressInGroup.length > 0) {
   347→        if (settings?.cancelInProgress) {
   348→          // Cancel all in-progress runs in this group.
   349→          for (const run of inProgressInGroup) {
   350→            const entry = active.get(run.id);
   351→            if (entry) {
   352→              entry.canceled = true;
   353→              try {
   354→                entry.child.kill("SIGTERM");
   355→              } catch {
   356→                /* ignore */
   357→              }
   358→            }
   359→            await appendLog(
   360→              run.id,
   361→              "system",
   362→              `Run canceled: a newer run in group "${group}" was triggered.`,
   363→            );
   364→            await finishRun(run.id, "canceled", 130);
   365→          }
   366→          await sleep(100);
   367→        } else {
   368→          // Serialized mode: create the run as "queued" and wait in a
   369→          // background poll loop until all in-progress runs in the group
   370→          // reach a terminal state, then promote this run to "running".
   371→          const queuedRun = await db.run.create({
   372→            data: {
   373→              projectId: options.projectId,
   374→              workflow: options.workflow,
   375→              status: "queued",
   376→              trigger: options.trigger ?? "manual",
   377→              startedAt: new Date(),
   378→              concurrencyGroup: group,
   379→              timeoutMs: options.timeoutMs ?? null,
   380→              requiresApproval: options.requiresApproval ?? false,
   381→              label: options.label ?? null,
   382→            },
   383→          });
   384→          emit({ type: "status", runId: queuedRun.id, status: "queued" });
   385→
   386→          void (async () => {
   387→            const waitDeadline = Date.now() + 24 * 60 * 60 * 1000; // 24h
   388→            while (Date.now() < waitDeadline) {
   389→              await sleep(2000);
   390→              const stillRunning = await db.run.findFirst({
   391→                where: {
   392→                  projectId: options.projectId,
   393→                  concurrencyGroup: group,
   394→                  status: { in: ["running", "queued", "waiting_approval"] },
   395→                  id: { not: queuedRun.id },
   396→                },
   397→              });
   398→              if (!stillRunning) break;
   399→            }
   400→            await db.run.update({
   401→              where: { id: queuedRun.id },
   402→              data: { status: "running" },
   403→            });
   404→            emit({ type: "status", runId: queuedRun.id, status: "running" });
   405→            await appendLog(
   406→              queuedRun.id,
   407→              "system",
   408→              "▶ Run started (was queued in concurrency group).",
   409→            );
   410→            await executeRun(queuedRun.id, options, project.extractedPath, detection);
   411→          })();
   412→
   413→          return { runId: queuedRun.id };
   414→        }
   415→      }
   416→    } else if (settings?.concurrentCancellation ?? true) {
   417→      const canceled = await cancelInprogressRuns(options.projectId);
   418→      if (canceled.length > 0) await sleep(100);
   419→    }
   420→  }
   421→
   422→  const run = await db.run.create({
   423→    data: {
   424→      projectId: options.projectId,
   425→      workflow: options.workflow,
   426→      status: options.requiresApproval ? "waiting_approval" : "running",
   427→      trigger: options.trigger ?? "manual",
   428→      startedAt: new Date(),
   429→      matrixValues: options.matrixValues
   430→        ? JSON.stringify(options.matrixValues)
   431→        : null,
   432→      matrixIndex: options.matrixIndex,
   433→      matrixTotal: options.matrixTotal,
   434→      parentRunId: options.parentRunId,
   435→      pipelineRunId: options.pipelineRunId,
   436→      stageId: options.stageId,
   437→      reRunOfId: options.reRunOfId,
   438→      timeoutMs: options.timeoutMs ?? null,
   439→      requiresApproval: options.requiresApproval ?? false,
   440→      label: options.label ?? null,
   441→      concurrencyGroup: options.concurrencyGroup ?? null,
   442→    },
   443→  });
   444→
   445→  emit({
   446→    type: "status",
   447→    runId: run.id,
   448→    status: options.requiresApproval ? "waiting_approval" : "running",
   449→  });
   450→
   451→  void (async () => {
   452→    try {
   453→      if (options.requiresApproval) {
   454→        const approved = await waitForApproval(run.id);
   455→        if (!approved) {
   456→          await finishRun(run.id, "canceled", 130);
   457→          return;
   458→        }
   459→        await db.run.update({
   460→          where: { id: run.id },
   461→          data: { status: "running" },
   462→        });
   463→        emit({ type: "status", runId: run.id, status: "running" });
   464→      }
   465→      await executeRun(run.id, options, project.extractedPath, detection);
   466→    } catch (err) {
   467→      await appendLog(
   468→        run.id,
   469→        "system",
   470→        `Runner crashed: ${err instanceof Error ? err.message : String(err)}`,
   471→      );
   472→      await finishRun(run.id, "failed", -1);
   473→    }
   474→  })();
   475→
   476→  return { runId: run.id };
   477→}

   492→/**
   493→ * Execute a run's workflow. Used both for the primary path and for
   494→ * queued-run promotion (after the concurrency group is free).
   495→ */
   496→async function executeRun(
   497→  runId: string,
   498→  options: RunOptions,
   499→  projectRoot: string,
   500→  detection: Detection,
   501→): Promise<void> {
   502→  const workflow = getWorkflow(options.workflow);
   503→  if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   504→
   505→  const steps = workflow.build(detection);
   506→  if (!steps || steps.length === 0) {
   507→    await appendLog(
   508→      runId,
   509→      "system",
   510→      `Workflow ${options.workflow} has no steps for this project.`,
   511→    );
   512→    await finishRun(runId, "failed", 1);
   513→    return;
   514→  }
   515→
   516→  // Substitute matrix values in step commands.
   517→  const substitutedSteps = options.matrixValues
   518→    ? steps.map((s) => ({
   519→        ...s,
   520→        command: substituteMatrix(s.command, options.matrixValues!),
   521→        label: substituteMatrix(s.label, options.matrixValues!),
   522→      }))
   523→    : steps;
   524→
   525→  // Build env with secrets + matrix values.
   526→  const secretsList = options.secrets ?? [];
   527→  const extraEnv: Record<string, string> = { ...options.env };
   528→  if (options.matrixValues) {
   529→    for (const [k, v] of Object.entries(options.matrixValues)) {
   530→      extraEnv[`MATRIX_${k.toUpperCase()}`] = v;
   531→    }
   532→  }
   533→  const allSecrets =
   534→    secretsList.length > 0 ? await getAllSecrets(options.projectId) : {};
   535→
   536→  // Cache restore.
   537→  if (options.cache?.restore) {
   538→    const cacheExists = await hasCache(options.projectId, options.cache.key);
   539→    if (cacheExists) {
   540→      const restored = await restoreCache(options.projectId, options.cache.key);
   541→      if (restored.hit) {
   542→        emit({ type: "cache-hit", runId, cacheLabel: options.cache.label });
   543→        await appendLog(
   544→          runId,
   545→          "system",
   546→          `💾 Cache hit: ${options.cache.label} (${formatBytes(restored.size ?? 0)})`,
   547→        );
   548→      }
   549→    } else {
   550→      emit({ type: "cache-miss", runId, cacheLabel: options.cache.label });
   551→      await appendLog(
   552→        runId,
   553→        "system",
   554→        `💾 Cache miss: ${options.cache.label}`,
   555→      );
   556→    }
   557→  }
   558→
   559→  const startedAt = Date.now();
   560→  let lastExitCode = 0;
   561→
   562→  // AxiomState workflows (parse / bundle) bypass the shell-step loop.
   563→  if (options.workflow === "parse" || options.workflow === "bundle") {
   564→    lastExitCode = await runAxiomWorkflow(
   565→      runId,
   566→      projectRoot,
   567→      options.workflow,
   568→      options.matrixValues,
   569→    );
   570→  } else {
   571→    for (const step of substitutedSteps) {
   572→      if (active.get(runId)?.canceled) {
   573→        await finishRun(runId, "canceled", 130, Date.now() - startedAt);
   574→        return;
   575→      }
   576→      const maxAttempts = (options.retry ?? 0) + 1;
   577→      let attempt = 0;
   578→      let stepExitCode = 1;
   579→      while (attempt < maxAttempts) {
   580→        if (attempt > 0) {
   581→          emit({ type: "retry", runId, retryAttempt: attempt });
   582→          await appendLog(
   583→            runId,
   584→            "system",
   585→            `↻ Retry ${attempt}/${maxAttempts - 1} for: ${step.label}`,
   586→          );
   587→        }
   588→        await appendLog(runId, "system", `▶ ${step.label}`);
   589→        stepExitCode = await runStep(runId, step, projectRoot, {
   590→          secrets: allSecrets,
   591→          extraEnv,
   592→          timeoutMs: options.timeoutMs,
   593→        });
   594→        if (stepExitCode === 0) break;
   595→        attempt++;
   596→      }
   597→      if (stepExitCode !== 0) {
   598→        lastExitCode = stepExitCode;
   599→        await appendLog(
   600→          runId,
   601→          "system",
   602→          `✗ Step failed with exit code ${stepExitCode} after ${attempt} attempt(s).`,
   603→        );
   604→        await finishRun(runId, "failed", stepExitCode, Date.now() - startedAt);
   605→        // Cache save even on failure (partial cache is still useful).
   606→        if (options.cache?.save) {
   607→          try {
   608→            await saveCache(
   609→              options.projectId,
   610→              options.cache.key,
   611→              options.cache.label,
   612→              options.cache.paths,
   613→            );
   614→            emit({ type: "cache-saved", runId, cacheLabel: options.cache.label });
   615→            await appendLog(
   616→              runId,
   617→              "system",
   618→              `💾 Cache saved: ${options.cache.label}`,
   619→            );
   620→          } catch (err) {
   621→            await appendLog(
   622→              runId,
   623→              "system",
   624→              `Cache save failed: ${err instanceof Error ? err.message : String(err)}`,
   625→            );
   626→          }
   627→        }
   628→        return;
   629→      }
   630→      await appendLog(runId, "system", `✓ ${step.label}`);
   631→    }
   632→  }
   633→
   634→  // Test report capture (single source of truth: test-report.storeTestReport).
   635→  if (options.testReport) {
   636→    try {
   637→      const reportPath = path.join(projectRoot, options.testReport.path);
   638→      if (fs.existsSync(reportPath)) {
   639→        const content = fs.readFileSync(reportPath, "utf-8");
   640→        const { parseJUnit, parseJSONReport, parseTAP } = await import(
   641→          "./test-report"
   642→        );
   643→        const report =
   644→          options.testReport.format === "junit"
   645→            ? parseJUnit(content)
   646→            : options.testReport.format === "json"
   647→              ? parseJSONReport(content)
   648→              : parseTAP(content);
   649→        await storeTestReport(runId, report);
   650→        await appendLog(
   651→          runId,
   652→          "system",
   653→          `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`,
   654→        );
   655→      }
   656→    } catch (err) {
   657→      await appendLog(
   658→        runId,
   659→        "system",
   660→        `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`,
   661→      );
   662→    }
   663→  }
   664→
   665→  // Capture artifacts.
   666→  await captureArtifacts(runId, options.workflow, detection, projectRoot);
   667→
   668→  // Cache save on success.
   669→  if (options.cache?.save) {
   670→    try {
   671→      const { size } = await saveCache(
   672→        options.projectId,
   673→        options.cache.key,
   674→        options.cache.label,
   675→        options.cache.paths,
   676→      );
   677→      emit({ type: "cache-saved", runId, cacheLabel: options.cache.label });
   678→      await appendLog(
   679→        runId,
   680→        "system",
   681→        `💾 Cache saved: ${options.cache.label} (${formatBytes(size)})`,
   682→      );
   683→    } catch (err) {
   684→      await appendLog(
   685→        runId,
   686→        "system",
   687→        `Cache save failed: ${err instanceof Error ? err.message : String(err)}`,
   688→      );
   689→    }
   690→  }
   691→
   692→  await finishRun(runId, "success", lastExitCode, Date.now() - startedAt);
   693→}

   695→export async function cancelRun(runId: string): Promise<void> {
   696→  const entry = active.get(runId);
   697→  if (entry) {
   698→    entry.canceled = true;
   699→    try {
   700→      entry.child.kill("SIGTERM");
   701→    } catch {
   702→      /* ignore */
   703→    }
   704→  }
   705→  await appendLog(runId, "system", "Run canceled by user");
   706→  await finishRun(runId, "canceled", 130);
   707→}

   709→// ---------------------------------------------------------------------------
   710→// Step execution — delegates to the shared runChildStep primitive.
   711→// ---------------------------------------------------------------------------
   712→
   713→interface StepEnvOptions {
   714→  secrets: Record<string, string>;
   715→  extraEnv: Record<string, string>;
   716→  timeoutMs?: number;
   717→}
   718→
   719→async function runStep(
   720→  runId: string,
   721→  step: WorkflowStep,
   722→  cwd: string,
   723→  envOptions: StepEnvOptions,
   724→): Promise<number> {
   725→  if (isBlockedCommand(step.command)) {
   726→    await appendLog(
   727→      runId,
   728→      "system",
   729→      `🚫 Command blocked by security policy: ${step.command.slice(0, 80)}`,
   730→    );
   731→    return 126;
   732→  }
   733→
   734→  const baseEnv = await buildProcessEnv("", {
   735→    extraEnv: envOptions.extraEnv,
   736→    projectRoot: cwd,
   737→  });
   738→  const fullEnv = { ...baseEnv, ...envOptions.secrets };
   739→  const cancelSignal = { aborted: false };
   740→
   741→  // Track active child for cancellation / concurrent-cancellation.
   742→  // We pretend the child exists by creating a placeholder that cancelRun
   743→  // can mark canceled; runChildStep reads `cancelSignal.aborted`.
   744→  const fakeChild = new (class {
   745→    kill(_signal?: string) {
   746→      cancelSignal.aborted = true;
   747→    }
   748→  })() as unknown as import("node:child_process").ChildProcess;
   749→  active.set(runId, { child: fakeChild, canceled: false });
   750→
   751→  const result = await runChildStep({
   752→    cwd,
   753→    command: step.command,
   754→    language: "bash",
   755→    env: fullEnv,
   756→    secrets: envOptions.secrets,
   757→    timeoutMs: envOptions.timeoutMs ?? null,
   758→    signal: cancelSignal,
   759→    onLine: (stream, text) => {
   760→      void appendLog(runId, stream, text);
   761→    },
   762→  });
   763→
   764→  active.delete(runId);
   765→
   766→  if (result.timedOut) {
   767→    await appendLog(
   768→      runId,
   769→      "system",
   770→      `⏱ Step timed out after ${envOptions.timeoutMs}ms.`,
   771→    );
   772→  }
   773→  return result.exitCode;
   774→}

   776→// ---------------------------------------------------------------------------
   777→// AxiomState workflow (parse / bundle)
   778→// ---------------------------------------------------------------------------
   779→
   780→async function runAxiomWorkflow(
   781→  runId: string,
   782→  projectRoot: string,
   783→  key: "parse" | "bundle",
   784→  matrixValues?: MatrixRow,
   785→): Promise<number> {
   786→  try {
   787→    const { parseProject, writeGraph, sliceForward } = await import(
   788→      "@/lib/axiomstate/phase1"
   789→    );
   790→    const { bundleFiles } = await import("@/lib/axiomstate/phase2");
   791→    const { LSSKernel } = await import("@/lib/axiomstate/phase0/kernel");
   792→
   793→    // Place the kernel under the project's own storage directory (not under
   794→    // an unrelated project's path — fixes the original extractDir('') bug).
   795→    const kernelDir = path.join(
   796→      path.dirname(extractDir("__axiomstate__")),
   797→      `kernel-${runId}`,
   798→    );
   799→    fs.mkdirSync(kernelDir, { recursive: true });
   800→    const kernel = new LSSKernel(kernelDir);
   801→
   802→    await appendLog(
   803→      runId,
   804→      "system",
   805→      `AxiomState: parsing project at ${projectRoot}`,
   806→    );
   807→    const delta = parseProject(projectRoot);
   808→    const seq = writeGraph(kernel, delta, {
   809→      checkpoint: false,
   810→      providerName: "typescript+regex",
   811→    });
   812→    await appendLog(
   813→      runId,
   814→      "stdout",
   815→      `Parsed ${delta.nodes.length} nodes and ${delta.edges.length} edges (seq=${seq}).`,
   816→    );
   817→    if (matrixValues) {
   818→      await appendLog(runId, "system", `Matrix: ${JSON.stringify(matrixValues)}`);
   819→    }
   820→
   821→    if (key === "parse") {
   822→      for (const node of delta.nodes) {
   823→        await appendLog(
   824→          runId,
   825→          "stdout",
   826→          `  ${node.kind.padEnd(6)} ${node.id}  (deps: ${node.deps.length})`,
   827→        );
   828→      }
   829→    } else {
   830→      const indexNode = delta.nodes.find(
   831→        (n) => n.kind === "file" && /index\.(ts|js|tsx|jsx)$/.test(n.name),
   832→      );
   833→      if (!indexNode) {
   834→        await appendLog(
   835→          runId,
   836→          "stderr",
   837→          "No index.(ts|js|tsx|jsx) entry found — cannot bundle.",
   838→        );
   839→        return 1;
   840→      }
   841→      await appendLog(
   842→        runId,
   843→        "system",
   844→        `AxiomState: bundling forward slice from ${indexNode.id}`,
   845→      );
   846→      const slice = sliceForward(kernel, indexNode.id);
   847→      const fileIds = Array.from(slice.nodes.values())
   848→        .filter((n) => n.kind === "file")
   849→        .map((n) => n.id);
   850→      const bundle = bundleFiles(kernel, fileIds);
   851→      await appendLog(
   852→        runId,
   853→        "stdout",
   854→        `Bundle order (${bundle.order.length} files):`,
   855→      );
   856→      for (const id of bundle.order) await appendLog(runId, "stdout", `  • ${id}`);
   857→      if (bundle.cycles.length > 0) {
   858→        await appendLog(
   859→          runId,
   860→          "stderr",
   861→          `Cycles detected: ${bundle.cycles.length}`,
   862→        );
   863→        for (const cycle of bundle.cycles)
   864→          await appendLog(runId, "stderr", `  ↻ ${cycle.join(" → ")}`);
   865→      }
   866→      const artifactDir = runArtifactDir(runId);
   867→      const outPath = path.join(artifactDir, "bundle.js");
   868→      const parts: Buffer[] = [];
   869→      for (const entry of bundle.entries) {
   870→        parts.push(Buffer.from(`\n// --- ${entry.path} ---\n`));
   871→        parts.push(Buffer.from(entry.content));
   872→      }
   873→      fs.writeFileSync(outPath, Buffer.concat(parts));
   874→      const stat = fs.statSync(outPath);
   875→      const artifact = await db.artifact.create({
   876→        data: {
   877→          runId,
   878→          name: "bundle.js",
   879→          path: outPath,
   880→          size: stat.size,
   881→          mime: "text/javascript",
   882→        },
   883→      });
   884→      emit({
   885→        type: "artifact",
   886→        runId,
   887→        artifact: { id: artifact.id, name: artifact.name, size: artifact.size },
   888→      });
   889→      await appendLog(
   890→        runId,
   891→        "system",
   892→        `Artifact written: bundle.js (${stat.size} bytes)`,
   893→      );
   894→    }
   895→
   896→    kernel.close();
   897→    try {
   898→      fs.rmSync(kernelDir, { recursive: true, force: true });
   899→    } catch {
   900→      /* ignore */
   901→    }
   902→    return 0;
   903→  } catch (err) {
   904→    await appendLog(
   905→      runId,
   906→      "stderr",
   907→      `AxiomState error: ${err instanceof Error ? err.message : String(err)}`,
   908→    );
   909→    return 1;
   910→  }
   911→}

   913→// ---------------------------------------------------------------------------
   914→// Artifacts
   915→// ---------------------------------------------------------------------------
   916→
   917→async function captureArtifacts(
   918→  runId: string,
   919→  workflowKey: string,
   920→  detection: Detection,
   921→  projectRoot: string,
   922→): Promise<void> {
   923→  const workflow = getWorkflow(workflowKey);
   924→  if (!workflow?.producesArtifacts) return;
   925→  const specs = workflow.producesArtifacts(detection, projectRoot);
   926→  if (specs.length === 0) return;
   927→
   928→  const artifactDir = runArtifactDir(runId);
   929→  for (const spec of specs) {
   930→    // Case 1: directory glob like "dist/**/*" — zip the whole directory.
   931→    const dirMatch = spec.glob.match(/^(.+?)\/\*\*\/\*$/);
   932→    if (dirMatch) {
   933→      const dir = dirMatch[1]!;
   934→      const fullDir = path.join(projectRoot, dir);
   935→      if (!fs.existsSync(fullDir)) continue;
   936→      const zipPath = path.join(artifactDir, `${spec.name}.zip`);
   937→      await zipDirectory(fullDir, zipPath);
   938→      const stat = fs.statSync(zipPath);
   939→      const artifact = await db.artifact.create({
   940→        data: {
   941→          runId,
   942→          name: `${spec.name}.zip`,
   943→          path: zipPath,
   944→          size: stat.size,
   945→          mime: "application/zip",
   946→        },
   947→      });
   948→      emit({
   949→        type: "artifact",
   950→        runId,
   951→        artifact: { id: artifact.id, name: artifact.name, size: artifact.size },
   952→      });
   953→      await appendLog(
   954→        runId,
   955→        "system",
   956→        `Artifact captured: ${artifact.name} (${stat.size} bytes)`,
   957→      );
   958→      continue;
   959→    }
   960→
   961→    // Case 2: single file like "forge-apk-output/app-release.apk" — copy as-is.
   962→    const fullPath = path.join(projectRoot, spec.glob);
   963→    if (!fs.existsSync(fullPath)) {
   964→      await appendLog(
   965→        runId,
   966→        "system",
   967→        `Artifact not found at expected path: ${spec.glob}`,
   968→      );
   969→      continue;
   970→    }
   971→    const stat = fs.statSync(fullPath);
   972→    if (stat.isDirectory()) continue;
   973→    const destName = spec.name || path.basename(fullPath);
   974→    const destPath = path.join(artifactDir, destName);
   975→    fs.copyFileSync(fullPath, destPath);
   976→    const artifact = await db.artifact.create({
   977→      data: {
   978→        runId,
   979→        name: destName,
   980→        path: destPath,
   981→        size: stat.size,
   982→        mime: spec.mime,
   983→      },
   984→    });
   985→    emit({
   986→      type: "artifact",
   987→      runId,
   988→      artifact: { id: artifact.id, name: artifact.name, size: artifact.size },
   989→    });
   990→    await appendLog(
   991→      runId,
   992→      "system",
   993→      `Artifact captured: ${artifact.name} (${stat.size} bytes)`,
   994→    );
   995→  }
   996→}

   998→async function zipDirectory(
   999→  srcDir: string,
  1000→  outPath: string,
  1001→): Promise<void> {
  1002→  return new Promise((resolve, reject) => {
  1003→    const child = spawn("zip", ["-r", "-q", outPath, "."], { cwd: srcDir });
  1004→    child.on("close", (code) =>
  1005→      code === 0 ? resolve() : reject(new Error(`zip exited with ${code}`)),
  1006→    );
  1007→    child.on("error", reject);
  1008→  });
  1009→}
