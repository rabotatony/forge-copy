    49→type Listener = (event: RunEvent) => void;
    50→const listeners = new Map<string, Set<Listener>>();
    51→const active = new Map<string, { child: ReturnType<typeof spawn>; canceled: boolean; timeoutTimer?: ReturnType<typeof setTimeout> }>();
    52→const seqCounter = new Map<string, number>();
    53→// Per-runId promise queue for appendLog — serializes the read-increment-write
    54→// of seqCounter so concurrent calls can't grab the same seq (duplicate keys).
    55→const appendLogQueues = new Map<string, Promise<void>>();
    56→
    57→export function subscribe(runId: string, fn: Listener): () => void {
    58→  let set = listeners.get(runId);
    59→  if (!set) { set = new Set(); listeners.set(runId, set); }
    60→  set.add(fn);
    61→  return () => { set!.delete(fn); if (set!.size === 0) listeners.delete(runId); };
    62→}
    63→
    64→export function emit(event: RunEvent): void {
    65→  const set = listeners.get(event.runId);
    66→  if (set) for (const fn of set) fn(event);
    67→}

    69→// Maximum log lines per run (prevents unbounded DB growth).
    70→const MAX_LOG_LINES_PER_RUN = 10000;
    71→const logLineCounts = new Map<string, number>();
    72→
    73→export async function appendLog(runId: string, stream: 'stdout' | 'stderr' | 'system', text: string): Promise<void> {
    74→  // Serialize per-runId so concurrent calls can't grab the same seq.
    75→  const prev = appendLogQueues.get(runId) ?? Promise.resolve();
    76→  const next = prev.then(() => appendLogInner(runId, stream, text));
    77→  appendLogQueues.set(runId, next);
    78→  next.then(() => { if (appendLogQueues.get(runId) === next) appendLogQueues.delete(runId); }, () => { if (appendLogQueues.get(runId) === next) appendLogQueues.delete(runId); });
    79→  return next;
    80→}
    81→
    82→async function appendLogInner(runId: string, stream: 'stdout' | 'stderr' | 'system', text: string): Promise<void> {
    83→  // Enforce log line limit per run.
    84→  const count = logLineCounts.get(runId) ?? 0;
    85→  if (count >= MAX_LOG_LINES_PER_RUN) {
    86→    if (stream !== 'system') return;
    87→    if (count >= MAX_LOG_LINES_PER_RUN + 1) return;
    88→  }
    89→
    90→  const truncated = text.length > 10000 ? text.slice(0, 10000) + '…[truncated]' : text;
    91→
    92→  const seq = seqCounter.get(runId) ?? 0;
    93→  seqCounter.set(runId, seq + 1);
    94→  logLineCounts.set(runId, count + 1);
    95→  const ts = Date.now();
    96→  await db.logLine.create({ data: { runId, seq, stream, text: truncated } });
    97→  emit({ type: 'log', runId, log: { seq, stream, text: truncated, ts } });
    98→
    99→  if (count === MAX_LOG_LINES_PER_RUN - 1) {
   100→    const warnSeq = seq + 1;
   101→    seqCounter.set(runId, warnSeq + 1);
   102→    logLineCounts.set(runId, count + 2);
   103→    const warnText = `⚠ Log limit reached (${MAX_LOG_LINES_PER_RUN} lines). Further output suppressed.`;
   104→    await db.logLine.create({ data: { runId, seq: warnSeq, stream: 'system', text: warnText } });
   105→    emit({ type: 'log', runId, log: { seq: warnSeq, stream: 'system', text: warnText, ts: Date.now() } });
   106→  }
   107→}

   147→async function cancelInprogressRuns(projectId: string): Promise<string[]> {
   148→  // Cancel ALL non-terminal runs — not just "running".
   149→  const inProgress = await db.run.findMany({
   150→    where: { projectId, status: { in: ['running', 'queued', 'waiting_approval'] } },
   151→  });
   152→  const canceled: string[] = [];
   153→  for (const run of inProgress) {
   154→    const entry = active.get(run.id);
   155→    if (entry) {
   156→      entry.canceled = true;
   157→      try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   158→    }
   159→    await appendLog(run.id, 'system', 'Run canceled: a newer run was triggered (concurrent cancellation).');
   160→    await finishRun(run.id, 'canceled', 130);
   161→    canceled.push(run.id);
   162→  }
   163→  return canceled;
   164→}

   170→async function waitForApproval(runId: string): Promise<boolean> {
   171→  await db.approval.create({
   172→    data: { runId, status: 'pending', expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000) },
   173→  });
   174→  emit({ type: 'approval-required', runId, message: 'Manual approval required to start this run.' });
   175→  await appendLog(runId, 'system', '⏸ Manual approval required. Waiting for approval...');
   176→
   177→  // Poll for decision (every 2 seconds, up to 24 hours).
   178→  const deadline = Date.now() + 24 * 60 * 60 * 1000;
   179→  while (Date.now() < deadline) {
   180→    await sleep(2000);
   181→    const approval = await db.approval.findUnique({ where: { runId } });
   182→    if (!approval) return false;
   183→    if (approval.status === 'approved') {
   184→      await appendLog(runId, 'system', `✓ Approved by ${approval.decidedBy ?? 'unknown'}. Starting run...`);
   185→      return true;
   186→    }
   187→    if (approval.status === 'rejected') {
   188→      await appendLog(runId, 'system', `✗ Rejected by ${approval.decidedBy ?? 'unknown'}.`);
   189→      return false;
   190→    }
   191→    if (approval.status === 'expired') return false;
   192→    // Check if run was canceled externally
   193→    const run = await db.run.findUnique({ where: { id: runId } });
   194→    if (run?.status === 'canceled') return false;
   195→  }
   196→  await db.approval.update({ where: { runId }, data: { status: 'expired' } });
   197→  await appendLog(runId, 'system', 'Approval timed out after 24 hours.');
   198→  return false;
   199→}

   219→export function expandMatrix(dimensions: Array<{ key: string; values: string[] }>, exclude: MatrixRow[] = [], include: MatrixRow[] = []): MatrixRow[] {
   220→  if (dimensions.length === 0) return [{}];
   221→  const results: MatrixRow[] = [{}];
   222→  for (const dim of dimensions) {
   223→    const next: MatrixRow[] = [];
   224→    for (const row of results) {
   225→      for (const value of dim.values) {
   226→        next.push({ ...row, [dim.key]: value });
   227→      }
   228→    }
   229→    results.splice(0, results.length, ...next);
   230→  }
   231→  // Apply excludes
   232→  const filtered = results.filter(row => !exclude.some(ex => matchesRow(row, ex)));
   233→  // Apply includes
   234→  return [...filtered, ...include];
   235→}
   236→
   237→function matchesRow(row: MatrixRow, pattern: MatrixRow): boolean {
   238→  return Object.entries(pattern).every(([k, v]) => row[k] === v);
   239→}

   245→export async function startRunExtended(options: RunOptions): Promise<{ runId: string }> {
   246→  const project = await db.project.findUnique({ where: { id: options.projectId } });
   247→  if (!project) throw new Error(`Project ${options.projectId} not found`);
   248→  const workflow = getWorkflow(options.workflow);
   249→  if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   250→
   251→  const detection = JSON.parse(project.detection) as Detection;
   252→
   253→  // Concurrency control:
   254→  // 1. If the run has a concurrencyGroup, cancel or queue based on settings.
   255→  // 2. If project has concurrentCancellation enabled, cancel in-progress runs.
   256→  if (options.trigger !== 'pipeline') {
   257→    const settings = await db.projectSettings.findUnique({ where: { projectId: options.projectId } });
   258→    const group = options.concurrencyGroup ?? settings?.concurrencyGroup ?? null;
   259→
   260→    if (group) {
   261→      // Concurrency group: either cancel in-progress runs in the same group
   262→      // (if cancelInProgress) or wait for them (serialized).
   263→      const inProgressInGroup = await db.run.findMany({
   264→        where: {
   265→          projectId: options.projectId,
   266→          concurrencyGroup: group,
   267→          status: { in: ['running', 'queued', 'waiting_approval'] },
   268→        },
   269→      });
   270→
   271→      if (inProgressInGroup.length > 0) {
   272→        if (settings?.cancelInProgress) {
   273→          // Cancel all in-progress runs in this group.
   274→          for (const run of inProgressInGroup) {
   275→            const entry = active.get(run.id);
   276→            if (entry) {
   277→              entry.canceled = true;
   278→              try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   279→            }
   280→            await appendLog(run.id, 'system', `Run canceled: a newer run in group "${group}" was triggered.`);
   281→            await finishRun(run.id, 'canceled', 130);
   282→          }
   283→          await sleep(100);
   284→        } else {
   285→          // Serialized mode: create the run as "queued" and wait in a
   286→          // background poll loop until all in-progress runs in the group
   287→          // reach a terminal state. Then promote this run to "running".
   288→          // This is true queueing (GitHub Actions concurrency: queue).
   289→          const queuedRun = await db.run.create({
   290→            data: {
   291→              projectId: options.projectId,
   292→              workflow: options.workflow,
   293→              status: 'queued',
   294→              trigger: options.trigger ?? 'manual',
   295→              startedAt: new Date(),
   296→              concurrencyGroup: group,
   297→              timeoutMs: options.timeoutMs ?? null,
   298→              requiresApproval: options.requiresApproval ?? false,
   299→              label: options.label ?? null,
   300→            },
   301→          });
   302→          emit({ type: 'status', runId: queuedRun.id, status: 'queued' });
   303→
   304→          // Background: wait for in-progress runs to finish, then execute.
   305→          void (async () => {
   306→            const waitDeadline = Date.now() + 24 * 60 * 60 * 1000; // 24h max wait
   307→            while (Date.now() < waitDeadline) {
   308→              await sleep(2000);
   309→              // Check if any runs in this group are still in-progress.
   310→              const stillRunning = await db.run.findFirst({
   311→                where: {
   312→                  projectId: options.projectId,
   313→                  concurrencyGroup: group,
   314→                  status: { in: ['running', 'queued', 'waiting_approval'] },
   315→                  id: { not: queuedRun.id },
   316→                },
   317→              });
   318→              if (!stillRunning) break;
   319→            }
   320→            // Promote to running and execute.
   321→            await db.run.update({ where: { id: queuedRun.id }, data: { status: 'running' } });
   322→            emit({ type: 'status', runId: queuedRun.id, status: 'running' });
   323→            await appendLog(queuedRun.id, 'system', '▶ Run started (was queued in concurrency group).');
   324→            // Execute the workflow on this run.
   325→            await executeQueuedRun(queuedRun.id, options);
   326→          })();
   327→
   328→          return { runId: queuedRun.id };
   329→        }
   330→      }
   331→    } else if (settings?.concurrentCancellation ?? true) {
   332→      // Default: cancel all in-progress runs on this project.
   333→      const canceled = await cancelInprogressRuns(options.projectId);
   334→      if (canceled.length > 0) {
   335→        await sleep(100);
   336→      }
   337→    }
   338→  }
   339→
   340→  const run = await db.run.create({
   341→    data: {
   342→      projectId: options.projectId,
   343→      workflow: options.workflow,
   344→      status: options.requiresApproval ? 'waiting_approval' : 'running',
   345→      trigger: options.trigger ?? 'manual',
   346→      startedAt: new Date(),
   347→      matrixValues: options.matrixValues ? JSON.stringify(options.matrixValues) : null,
   348→      matrixIndex: options.matrixIndex,
   349→      matrixTotal: options.matrixTotal,
   350→      parentRunId: options.parentRunId,
   351→      pipelineRunId: options.pipelineRunId,
   352→      stageId: options.stageId,
   353→      reRunOfId: options.reRunOfId,
   354→      timeoutMs: options.timeoutMs ?? null,
   355→      requiresApproval: options.requiresApproval ?? false,
   356→      label: options.label ?? null,
   357→      concurrencyGroup: options.concurrencyGroup ?? null,
   358→    },
   359→  });
   360→
   361→  emit({ type: 'status', runId: run.id, status: options.requiresApproval ? 'waiting_approval' : 'running' });
   362→
   363→  // Fire "started" notification (best-effort).
   364→  void (async () => {
   365→    try {
   366→      const { notifyRunStarted } = await import('./notifications');
   367→      await notifyRunStarted(run.id);
   368→    } catch { /* additive */ }
   369→  })();
   370→
   371→  // Run in background. onRunStart (GitHub check-run) runs inline first
   372→  // (5s timeout) so a fast run can't finish before the check-run exists.
   373→  void (async () => {
   374→    try {
   375→      try {
   376→        const { onRunStart } = await import('./github-feedback');
   377→        await Promise.race([onRunStart(run.id), new Promise<void>((r) => setTimeout(r, 5_000))]);
   378→      } catch { /* additive */ }
   379→
   380→      // Approval gate
   381→      if (options.requiresApproval) {
   382→        const approved = await waitForApproval(run.id);
   383→        if (!approved) {
   384→          await finishRun(run.id, 'canceled', 130);
   385→          return;
   386→        }
   387→        await db.run.update({ where: { id: run.id }, data: { status: 'running' } });
   388→        emit({ type: 'status', runId: run.id, status: 'running' });
   389→      }
   390→
   391→      const steps = workflow.build(detection);
   392→      // Skip the "no steps" check for plugin workflows (parse/bundle) —
   393→      // they don't use shell steps; the engine dispatches to runAxiomWorkflow
   394→      // below. Their build() returns [] intentionally.
   395→      const isPluginWorkflow = options.workflow === 'parse' || options.workflow === 'bundle';
   396→      if (!isPluginWorkflow && (!steps || steps.length === 0)) {
   397→        await appendLog(run.id, 'system', `Workflow ${options.workflow} has no steps for this project.`);
   398→        await finishRun(run.id, 'failed', 1);
   399→        return;
   400→      }
   401→
   402→      // Substitute matrix values in step commands.
   403→      const safeSteps = steps ?? [];
   404→      const substitutedSteps = options.matrixValues
   405→        ? safeSteps.map(s => ({ ...s, command: substituteMatrix(s.command, options.matrixValues!), label: substituteMatrix(s.label, options.matrixValues!) }))
   406→        : safeSteps;
   407→
   408→      // Build env with secrets + matrix values.
   409→      const secretsList = options.secrets ?? [];
   410→      const extraEnv: Record<string, string> = { ...options.env };
   411→      if (options.matrixValues) {
   412→        for (const [k, v] of Object.entries(options.matrixValues)) {
   413→          extraEnv[`MATRIX_${k.toUpperCase()}`] = v;
   414→        }
   415→      }
   416→      const allSecrets = secretsList.length > 0 ? await getAllSecrets(options.projectId) : {};
   417→
   418→      // Cache restore.
   419→      if (options.cache?.restore) {
   420→        const cacheExists = await hasCache(options.projectId, options.cache.key);
   421→        if (cacheExists) {
   422→          const restored = await restoreCache(options.projectId, options.cache.key);
   423→          if (restored.hit) {
   424→            emit({ type: 'cache-hit', runId: run.id, cacheLabel: options.cache.label });
   425→            await appendLog(run.id, 'system', `💾 Cache hit: ${options.cache.label} (${formatBytes(restored.size ?? 0)})`);
   426→          }
   427→        } else {
   428→          emit({ type: 'cache-miss', runId: run.id, cacheLabel: options.cache.label });
   429→          await appendLog(run.id, 'system', `💾 Cache miss: ${options.cache.label}`);
   430→        }
   431→      }
   432→
   433→      // Execute.
   434→      const startedAt = Date.now();
   435→      let lastExitCode = 0;
   436→
   437→      if (options.workflow === 'parse' || options.workflow === 'bundle') {
   438→        lastExitCode = await runAxiomWorkflow(run.id, project.extractedPath, options.workflow, options.matrixValues);
   439→      } else {
   440→        for (const step of substitutedSteps) {
   441→          if (active.get(run.id)?.canceled) {
   442→            await finishRun(run.id, 'canceled', 130, Date.now() - startedAt);
   443→            return;
   444→          }
   445→          const maxAttempts = (options.retry ?? 0) + 1;
   446→          let attempt = 0;
   447→          let stepExitCode = 1;
   448→          while (attempt < maxAttempts) {
   449→            if (attempt > 0) {
   450→              emit({ type: 'retry', runId: run.id, retryAttempt: attempt });
   451→              await appendLog(run.id, 'system', `↻ Retry ${attempt}/${maxAttempts - 1} for: ${step.label}`);
   452→            }
   453→            await appendLog(run.id, 'system', `▶ ${step.label}`);
   454→            stepExitCode = await runShellStep(
   455→              run.id,
   456→              step,
   457→              project.extractedPath,
   458→              { secrets: allSecrets, extraEnv, timeoutMs: options.timeoutMs, projectId: options.projectId },
   459→            );
   460→            if (stepExitCode === 0) break;
   461→            attempt++;
   462→          }
   463→          if (stepExitCode !== 0) {
   464→            lastExitCode = stepExitCode;
   465→            await appendLog(run.id, 'system', `✗ Step failed with exit code ${stepExitCode} after ${attempt} attempt(s).`);
   466→            await finishRun(run.id, 'failed', stepExitCode, Date.now() - startedAt);
   467→            // Cache save even on failure (partial cache is still useful).
   468→            if (options.cache?.save) {
   469→              try {
   470→                await saveCache(options.projectId, options.cache.key, options.cache.label, options.cache.paths);
   471→                emit({ type: 'cache-saved', runId: run.id, cacheLabel: options.cache.label });
   472→                await appendLog(run.id, 'system', `💾 Cache saved: ${options.cache.label}`);
   473→              } catch (err) {
   474→                await appendLog(run.id, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   475→              }
   476→            }
   477→            return;
   478→          }
   479→          await appendLog(run.id, 'system', `✓ ${step.label}`);
   480→        }
   481→      }
   482→
   483→      // Test report capture.
   484→      if (options.testReport) {
   485→        try {
   486→          const reportPath = path.join(project.extractedPath, options.testReport.path);
   487→          if (fs.existsSync(reportPath)) {
   488→            const content = fs.readFileSync(reportPath, 'utf-8');
   489→            const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   490→            let report: ParsedTestReport;
   491→            if (options.testReport.format === 'junit') report = parseJUnit(content);
   492→            else if (options.testReport.format === 'json') report = parseJSONReport(content);
   493→            else report = parseTAP(content);
   494→            await db.testReport.create({
   495→              data: {
   496→                runId: run.id,
   497→                format: report.format,
   498→                total: report.total,
   499→                passed: report.passed,
   500→                failed: report.failed,
   501→                skipped: report.skipped,
   502→                duration: report.duration ?? null,
   503→                suites: JSON.stringify(report.suites),
   504→              },
   505→            });
   506→            await appendLog(run.id, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   507→          }
   508→        } catch (err) {
   509→          await appendLog(run.id, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   510→        }
   511→      }
   512→
   513→      // Capture artifacts.
   514→      await captureArtifacts(run.id, options.workflow, detection, project.extractedPath);
   515→
   516→      // Cache save on success.
   517→      if (options.cache?.save) {
   518→        try {
   519→          const { size } = await saveCache(options.projectId, options.cache.key, options.cache.label, options.cache.paths);
   520→          emit({ type: 'cache-saved', runId: run.id, cacheLabel: options.cache.label });
   521→          await appendLog(run.id, 'system', `💾 Cache saved: ${options.cache.label} (${formatBytes(size)})`);
   522→        } catch (err) {
   523→          await appendLog(run.id, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   524→        }
   525→      }
   526→
   527→      await finishRun(run.id, 'success', lastExitCode, Date.now() - startedAt);
   528→    } catch (err) {
   529→      await appendLog(run.id, 'system', `Runner crashed: ${err instanceof Error ? err.message : String(err)}`);
   530→      await finishRun(run.id, 'failed', -1);
   531→    }
   532→  })();
   533→
   534→  return { runId: run.id };
   535→}

   546→/**
   547→ * Execute a queued run after its concurrency group is free.
   548→ * Reuses the same logic as startRunExtended but skips the concurrency check
   549→ * (the run is already queued, so it's allowed to run now).
   550→ */
   551→async function executeQueuedRun(runId: string, options: RunOptions): Promise<void> {
   552→  try {
   553→    const project = await db.project.findUnique({ where: { id: options.projectId } });
   554→    if (!project) throw new Error(`Project ${options.projectId} not found`);
   555→    const workflow = getWorkflow(options.workflow);
   556→    if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   557→
   558→    const detection = JSON.parse(project.detection) as Detection;
   559→    const steps = workflow.build(detection);
   560→    const isPluginWorkflow = options.workflow === 'parse' || options.workflow === 'bundle';
   561→    if (!isPluginWorkflow && (!steps || steps.length === 0)) {
   562→      await appendLog(runId, 'system', `Workflow ${options.workflow} has no steps.`);
   563→      await finishRun(runId, 'failed', 1);
   564→      return;
   565→    }
   566→
   567→    const secretsList = options.secrets ?? [];
   568→    const extraEnv: Record<string, string> = { ...options.env };
   569→    const allSecrets = secretsList.length > 0 ? await getAllSecrets(options.projectId) : {};
   570→
   571→    const startedAt = Date.now();
   572→    let lastExitCode = 0;
   573→
   574→    for (const step of (steps ?? [])) {
   575→      if (active.get(runId)?.canceled) {
   576→        await finishRun(runId, 'canceled', 130, Date.now() - startedAt);
   577→        return;
   578→      }
   579→      await appendLog(runId, 'system', `▶ ${step.label}`);
   580→      const stepExitCode = await runShellStep(
   581→        runId,
   582→        step,
   583→        project.extractedPath,
   584→        { secrets: allSecrets, extraEnv, timeoutMs: options.timeoutMs, projectId: options.projectId },
   585→      );
   586→      if (stepExitCode !== 0) {
   587→        lastExitCode = stepExitCode;
   588→        await appendLog(runId, 'system', `✗ Step failed with exit code ${stepExitCode}.`);
   589→        // If canceled (SIGTERM killed the step), mark canceled not failed.
   590→        if (active.get(runId)?.canceled) {
   591→          await finishRun(runId, 'canceled', 130, Date.now() - startedAt);
   592→        } else {
   593→          await finishRun(runId, 'failed', stepExitCode, Date.now() - startedAt);
   594→        }
   595→        return;
   596→      }
   597→      await appendLog(runId, 'system', `✓ ${step.label}`);
   598→    }
   599→
   600→    // Capture artifacts.
   601→    await captureArtifacts(runId, options.workflow, detection, project.extractedPath);
   602→    await finishRun(runId, 'success', lastExitCode, Date.now() - startedAt);
   603→  } catch (err) {
   604→    await appendLog(runId, 'system', `Runner crashed: ${err instanceof Error ? err.message : String(err)}`);
   605→    await finishRun(runId, 'failed', -1);
   606→  }
   607→}

   609→export async function cancelRun(runId: string): Promise<void> {
   610→  const entry = active.get(runId);
   611→  if (entry) {
   612→    entry.canceled = true;
   613→    try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   614→    if (entry.timeoutTimer) clearTimeout(entry.timeoutTimer);
   615→  }
   616→  await appendLog(runId, 'system', 'Run canceled by user');
   617→  await finishRun(runId, 'canceled', 130);
   618→}

   624→interface StepEnvOptions {
   625→  secrets: Record<string, string>;
   626→  extraEnv: Record<string, string>;
   627→  timeoutMs?: number;
   628→  projectId: string; // needed to load project env vars via buildProcessEnv
   629→}

   631→async function runShellStep(
   632→  runId: string,
   633→  step: WorkflowStep,
   634→  cwd: string,
   635→  envOptions: StepEnvOptions,
   636→): Promise<number> {
   637→  return new Promise((resolve) => {
   638→    // Security: block known-dangerous commands.
   639→    const BLOCKED_PATTERNS = [
   640→      /\/etc\/(passwd|shadow|sudoers)/i,
   641→      /\/proc\/self\/(environ|cmdline)/i,
   642→      /\brm\s+-rf\s+\//i,
   643→      /\bmkfs\b/i,
   644→      /\bdd\s+if=\/dev\//i,
   645→      /:\(\)\s*\{/i,  // fork bomb
   646→    ];
   647→    for (const pattern of BLOCKED_PATTERNS) {
   648→      if (pattern.test(step.command)) {
   649→        void appendLog(runId, 'system', `🚫 Command blocked by security policy: ${step.command.slice(0, 80)}`);
   650→        resolve(126);
   651→        return;
   652→      }
   653→    }
   654→
   655→    const projectRoot = cwd;
   656→    buildProcessEnv(envOptions.projectId, { extraEnv: envOptions.extraEnv, projectRoot }).then((baseEnv) => {
   657→      const fullEnv = { ...baseEnv, ...envOptions.secrets };
   658→      const child = spawn('bash', ['-c', step.command], {
   659→        cwd,
   660→        env: fullEnv as NodeJS.ProcessEnv,
   661→      }) as import('node:child_process').ChildProcessWithoutNullStreams;
   662→      active.set(runId, { child, canceled: false });
   663→
   664→      // Timeout
   665→      let timeoutTimer: ReturnType<typeof setTimeout> | undefined;
   666→      let killTimer: ReturnType<typeof setTimeout> | undefined;
   667→      if (envOptions.timeoutMs) {
   668→        timeoutTimer = setTimeout(() => {
   669→          try { child.kill('SIGTERM'); } catch { /* ignore */ }
   670→          void appendLog(runId, 'system', `⏱ Step timed out after ${envOptions.timeoutMs}ms. Killing process...`);
   671→          // Force-kill with SIGKILL after 2 seconds if still alive.
   672→          killTimer = setTimeout(() => {
   673→            try { child.kill('SIGKILL'); } catch { /* ignore */ }
   674→            void appendLog(runId, 'system', `⏱ Force-killed (SIGKILL).`);
   675→          }, 2000);
   676→        }, envOptions.timeoutMs);
   677→        active.get(runId)!.timeoutTimer = timeoutTimer;
   678→      }
   679→
   680→      const stdout = createInterface({ input: child.stdout });
   681→      const stderr = createInterface({ input: child.stderr });
   682→      stdout.on('line', (line) => { void appendLog(runId, 'stdout', maskSecrets(line, envOptions.secrets)); });
   683→      stderr.on('line', (line) => { void appendLog(runId, 'stderr', maskSecrets(line, envOptions.secrets)); });
   684→
   685→      child.on('close', (code) => {
   686→        if (timeoutTimer) clearTimeout(timeoutTimer);
   687→        if (killTimer) clearTimeout(killTimer);
   688→        active.delete(runId);
   689→        resolve(code ?? 0);
   690→      });
   691→      child.on('error', (err) => {
   692→        if (timeoutTimer) clearTimeout(timeoutTimer);
   693→        if (killTimer) clearTimeout(killTimer);
   694→        void appendLog(runId, 'system', `spawn error: ${err.message}`);
   695→        active.delete(runId);
   696→        resolve(1);
   697→      });
   698→    });
   699→  });
   700→}

   702→// ---------------------------------------------------------------------------
   703→// AxiomState workflow (parse / bundle)
   704→// ---------------------------------------------------------------------------
   705→
   706→async function runAxiomWorkflow(
   707→  runId: string,
   708→  projectRoot: string,
   709→  key: 'parse' | 'bundle',
   710→  matrixValues?: MatrixRow,
   711→): Promise<number> {
   712→  try {
   713→    const { parseProject, writeGraph, sliceForward } = await import('@/lib/axiomstate/phase1');
   714→    const { bundleFiles } = await import('@/lib/axiomstate/phase2');
   715→    const { LSSKernel } = await import('@/lib/axiomstate/phase0/kernel');
   716→
   717→    const kernelDir = path.join(extractDir(''), '..', `kernel-${runId}`);
   718→    fs.mkdirSync(kernelDir, { recursive: true });
   719→    const kernel = new LSSKernel(kernelDir);
   720→
   721→    await appendLog(runId, 'system', `AxiomState: parsing project at ${projectRoot}`);
   722→    const delta = parseProject(projectRoot);
   723→    const seq = writeGraph(kernel, delta, { checkpoint: false, providerName: 'typescript+regex' });
   724→    await appendLog(runId, 'stdout', `Parsed ${delta.nodes.length} nodes and ${delta.edges.length} edges (seq=${seq}).`);
   725→    if (matrixValues) {
   726→      await appendLog(runId, 'system', `Matrix: ${JSON.stringify(matrixValues)}`);
   727→    }
   728→
   729→    if (key === 'parse') {
   730→      for (const node of delta.nodes) {
   731→        await appendLog(runId, 'stdout', `  ${node.kind.padEnd(6)} ${node.id}  (deps: ${node.deps.length})`);
   732→      }
   733→    } else {
   734→      const indexNode = delta.nodes.find(n => n.kind === 'file' && /index\.(ts|js|tsx|jsx)$/.test(n.name));
   735→      if (!indexNode) {
   736→        await appendLog(runId, 'stderr', 'No index.(ts|js|tsx|jsx) entry found — cannot bundle.');
   737→        return 1;
   738→      }
   739→      await appendLog(runId, 'system', `AxiomState: bundling forward slice from ${indexNode.id}`);
   740→      const slice = sliceForward(kernel, indexNode.id);
   741→      const fileIds = Array.from(slice.nodes.values()).filter(n => n.kind === 'file').map(n => n.id);
   742→      const bundle = bundleFiles(kernel, fileIds);
   743→      await appendLog(runId, 'stdout', `Bundle order (${bundle.order.length} files):`);
   744→      for (const id of bundle.order) await appendLog(runId, 'stdout', `  • ${id}`);
   745→      if (bundle.cycles.length > 0) {
   746→        await appendLog(runId, 'stderr', `Cycles detected: ${bundle.cycles.length}`);
   747→        for (const cycle of bundle.cycles) await appendLog(runId, 'stderr', `  ↻ ${cycle.join(' → ')}`);
   748→      }
   749→      const artifactDir = runArtifactDir(runId);
   750→      const outPath = path.join(artifactDir, 'bundle.js');
   751→      const parts: Buffer[] = [];
   752→      for (const entry of bundle.entries) {
   753→        parts.push(Buffer.from(`\n// --- ${entry.path} ---\n`));
   754→        parts.push(Buffer.from(entry.content));
   755→      }
   756→      fs.writeFileSync(outPath, Buffer.concat(parts));
   757→      const stat = fs.statSync(outPath);
   758→      const artifact = await db.artifact.create({
   759→        data: { runId, name: 'bundle.js', path: outPath, size: stat.size, mime: 'text/javascript' },
   760→      });
   761→      emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   762→      await appendLog(runId, 'system', `Artifact written: bundle.js (${stat.size} bytes)`);
   763→    }
   764→
   765→    kernel.close();
   766→    try { fs.rmSync(kernelDir, { recursive: true, force: true }); } catch { /* ignore */ }
   767→    return 0;
   768→  } catch (err) {
   769→    await appendLog(runId, 'stderr', `AxiomState error: ${err instanceof Error ? err.message : String(err)}`);
   770→    return 1;
   771→  }
   772→}

   774→// ---------------------------------------------------------------------------
   775→// Artifacts
   776→// ---------------------------------------------------------------------------
   777→
   778→async function captureArtifacts(
   779→  runId: string,
   780→  workflowKey: string,
   781→  detection: Detection,
   782→  projectRoot: string,
   783→): Promise<void> {
   784→  const workflow = getWorkflow(workflowKey);
   785→  if (!workflow?.producesArtifacts) return;
   786→  const specs = workflow.producesArtifacts(detection, projectRoot);
   787→  if (specs.length === 0) return;
   788→
   789→  const artifactDir = runArtifactDir(runId);
   790→  for (const spec of specs) {
   791→    // Case 1: directory glob like "dist/**/*" — zip the whole directory.
   792→    const dirMatch = spec.glob.match(/^(.+?)\/\*\*\/\*$/);
   793→    if (dirMatch) {
   794→      const dir = dirMatch[1]!;
   795→      const fullDir = path.join(projectRoot, dir);
   796→      if (!fs.existsSync(fullDir)) continue;
   797→      const zipPath = path.join(artifactDir, `${spec.name}.zip`);
   798→      await zipDirectory(fullDir, zipPath);
   799→      const stat = fs.statSync(zipPath);
   800→      const artifact = await db.artifact.create({
   801→        data: { runId, name: `${spec.name}.zip`, path: zipPath, size: stat.size, mime: 'application/zip' },
   802→      });
   803→      emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   804→      await appendLog(runId, 'system', `Artifact captured: ${artifact.name} (${stat.size} bytes)`);
   805→      continue;
   806→    }
   807→
   808→    // Case 2: single file like "forge-apk-output/app-release.apk" — copy as-is.
   809→    const fullPath = path.join(projectRoot, spec.glob);
   810→    if (!fs.existsSync(fullPath)) {
   811→      await appendLog(runId, 'system', `Artifact not found at expected path: ${spec.glob}`);
   812→      continue;
   813→    }
   814→    const stat = fs.statSync(fullPath);
   815→    if (stat.isDirectory()) continue; // skip dirs without /**/*
   816→    // Copy the file to the artifact directory, preserving the original name.
   817→    const destName = spec.name || path.basename(fullPath);
   818→    const destPath = path.join(artifactDir, destName);
   819→    fs.copyFileSync(fullPath, destPath);
   820→    const artifact = await db.artifact.create({
   821→      data: { runId, name: destName, path: destPath, size: stat.size, mime: spec.mime },
   822→    });
   823→    emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   824→    await appendLog(runId, 'system', `Artifact captured: ${artifact.name} (${stat.size} bytes)`);
   825→  }
   826→}

   828→async function zipDirectory(srcDir: string, outPath: string): Promise<void> {
   829→  return new Promise((resolve, reject) => {
   830→    const child = spawn('zip', ['-r', '-q', outPath, '.'], { cwd: srcDir });
   831→    child.on('close', (code) => code === 0 ? resolve() : reject(new Error(`zip exited with ${code}`)));
   832→    child.on('error', reject);
   833→  });
   834→}

   840→export async function finishRun(
   841→  runId: string,
   842→  status: RunStatus,
   843→  exitCode: number,
   844→  durationMs?: number,
   845→): Promise<void> {
   846→  seqCounter.delete(runId);
   847→  logLineCounts.delete(runId);
   848→
   849→  // Idempotency: if already terminal, don't overwrite (cancel/finish race).
   850→  const existing = await db.run.findUnique({ where: { id: runId }, select: { status: true } });
   851→  if (existing && existing.status !== 'running' && existing.status !== 'queued' && existing.status !== 'waiting_approval') {
   852→    emit({ type: 'done', runId, status: existing.status as RunStatus, exitCode: undefined, durationMs: undefined });
   853→    return;
   854→  }
   855→
   856→  await db.run.update({
   857→    where: { id: runId },
   858→    data: {
   859→      status,
   860→      exitCode,
   861→      finishedAt: new Date(),
   862→      durationMs: durationMs ?? 0,
   863→    },
   864→  });
   865→  emit({ type: 'done', runId, status, exitCode, durationMs });
   866→
   867→  // Fire notifications (import lazily to avoid circular deps).
   868→  try {
   869→    const { notifyRunEvent } = await import('./notifications');
   870→    await notifyRunEvent(runId, status);
   871→  } catch { /* notifications not ready yet */ }
   872→
   873→  // Best-effort: update GitHub check-run with final status + annotations.
   874→  try {
   875→    const { onRunFinish } = await import('./github-feedback');
   876→    await onRunFinish(runId, status);
   877→  } catch { /* check-runs feedback is purely additive */ }
   878→}
