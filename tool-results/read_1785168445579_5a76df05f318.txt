    34→export interface RunEvent {
    35→  type: 'status' | 'log' | 'artifact' | 'done' | 'approval-required' | 'cache-hit' | 'cache-miss' | 'cache-saved' | 'retry' | 'matrix-started' | 'matrix-completed';

    49→type Listener = (event: RunEvent) => void;
    50→const listeners = new Map<string, Set<Listener>>();
    51→const active = new Map<string, { child: ReturnType<typeof spawn>; canceled: boolean; timeoutTimer?: ReturnType<typeof setTimeout> }>();
    52→const seqCounter = new Map<string, number>();
    53→
    54→export function subscribe(runId: string, fn: Listener): () => void {
    55→  let set = listeners.get(runId);
    56→  if (!set) { set = new Set(); listeners.set(runId, set); }
    57→  set.add(fn);
    58→  return () => { set!.delete(fn); if (set!.size === 0) listeners.delete(runId); };
    59→}
    60→
    61→export function emit(event: RunEvent): void {
    62→  const set = listeners.get(event.runId);
    63→  if (set) for (const fn of set) fn(event);
    64→}

    66→// Maximum log lines per run (prevents unbounded DB growth).
    67→const MAX_LOG_LINES_PER_RUN = 10000;
    68→const logLineCounts = new Map<string, number>();
    69→
    70→export async function appendLog(runId: string, stream: 'stdout' | 'stderr' | 'system', text: string): Promise<void> {
    71→  // Enforce log line limit per run.
    72→  const count = logLineCounts.get(runId) ?? 0;
    73→  if (count >= MAX_LOG_LINES_PER_RUN) {
    74→    // Once limit reached, only allow system messages (errors, status).
    75→    if (stream !== 'system') return;
    76→    // For system messages, check if we already logged the truncation warning.
    77→    if (count >= MAX_LOG_LINES_PER_RUN + 1) return;
    78→  }
    79→
    80→  // Truncate very long lines (prevent memory issues).
    81→  const truncated = text.length > 10000 ? text.slice(0, 10000) + '…[truncated]' : text;
    82→
    83→  const seq = seqCounter.get(runId) ?? 0;
    84→  seqCounter.set(runId, seq + 1);
    85→  logLineCounts.set(runId, count + 1);
    86→  const ts = Date.now();
    87→  await db.logLine.create({ data: { runId, seq, stream, text: truncated } });
    88→  emit({ type: 'log', runId, log: { seq, stream, text: truncated, ts } });
    89→
    90→  // Log truncation warning once.
    91→  if (count === MAX_LOG_LINES_PER_RUN - 1) {
    92→    const warnSeq = seq + 1;
    93→    seqCounter.set(runId, warnSeq + 1);
    94→    logLineCounts.set(runId, count + 2);
    95→    const warnText = `⚠ Log limit reached (${MAX_LOG_LINES_PER_RUN} lines). Further output suppressed.`;
    96→    await db.logLine.create({ data: { runId, seq: warnSeq, stream: 'system', text: warnText } });
    97→    emit({ type: 'log', runId, log: { seq: warnSeq, stream: 'system', text: warnText, ts: Date.now() } });
    98→  }
    99→}

   139→async function cancelInprogressRuns(projectId: string): Promise<string[]> {
   140→  const inProgress = await db.run.findMany({
   141→    where: { projectId, status: 'running' },
   142→  });
   143→  const canceled: string[] = [];
   144→  for (const run of inProgress) {
   145→    const entry = active.get(run.id);
   146→    if (entry) {
   147→      entry.canceled = true;
   148→      try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   149→    }
   150→    await appendLog(run.id, 'system', 'Run canceled: a newer run was triggered (concurrent cancellation).');
   151→    await finishRun(run.id, 'canceled', 130);
   152→    canceled.push(run.id);
   153→  }
   154→  return canceled;
   155→}

   161→async function waitForApproval(runId: string): Promise<boolean> {
   162→  await db.approval.create({
   163→    data: { runId, status: 'pending', expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000) },
   164→  });
   165→  emit({ type: 'approval-required', runId, message: 'Manual approval required to start this run.' });
   166→  await appendLog(runId, 'system', '⏸ Manual approval required. Waiting for approval...');
   167→
   168→  // Poll for decision (every 2 seconds, up to 24 hours).
   169→  const deadline = Date.now() + 24 * 60 * 60 * 1000;
   170→  while (Date.now() < deadline) {
   171→    await sleep(2000);
   172→    const approval = await db.approval.findUnique({ where: { runId } });
   173→    if (!approval) return false;
   174→    if (approval.status === 'approved') {
   175→      await appendLog(runId, 'system', `✓ Approved by ${approval.decidedBy ?? 'unknown'}. Starting run...`);
   176→      return true;
   177→    }
   178→    if (approval.status === 'rejected') {
   179→      await appendLog(runId, 'system', `✗ Rejected by ${approval.decidedBy ?? 'unknown'}.`);
   180→      return false;
   181→    }
   182→    if (approval.status === 'expired') return false;
   183→    // Check if run was canceled externally
   184→    const run = await db.run.findUnique({ where: { id: runId } });
   185→    if (run?.status === 'canceled') return false;
   186→  }
   187→  await db.approval.update({ where: { runId }, data: { status: 'expired' } });
   188→  await appendLog(runId, 'system', 'Approval timed out after 24 hours.');
   189→  return false;
   190→}

   210→export function expandMatrix(dimensions: Array<{ key: string; values: string[] }>, exclude: MatrixRow[] = [], include: MatrixRow[] = []): MatrixRow[] {
   211→  if (dimensions.length === 0) return [{}];
   212→  const results: MatrixRow[] = [{}];
   213→  for (const dim of dimensions) {
   214→    const next: MatrixRow[] = [];
   215→    for (const row of results) {
   216→      for (const value of dim.values) {
   217→        next.push({ ...row, [dim.key]: value });
   218→      }
   219→    }
   220→    results.splice(0, results.length, ...next);
   221→  }
   222→  // Apply excludes
   223→  const filtered = results.filter(row => !exclude.some(ex => matchesRow(row, ex)));
   224→  // Apply includes
   225→  return [...filtered, ...include];
   226→}
   227→
   228→function matchesRow(row: MatrixRow, pattern: MatrixRow): boolean {
   229→  return Object.entries(pattern).every(([k, v]) => row[k] === v);
   230→}

   236→export async function startRunExtended(options: RunOptions): Promise<{ runId: string }> {
   237→  const project = await db.project.findUnique({ where: { id: options.projectId } });
   238→  if (!project) throw new Error(`Project ${options.projectId} not found`);
   239→  const workflow = getWorkflow(options.workflow);
   240→  if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   241→
   242→  const detection = JSON.parse(project.detection) as Detection;
   243→
   244→  // Concurrency control:
   245→  // 1. If the run has a concurrencyGroup, cancel or queue based on settings.
   246→  // 2. If project has concurrentCancellation enabled, cancel in-progress runs.
   247→  if (options.trigger !== 'pipeline') {
   248→    const settings = await db.projectSettings.findUnique({ where: { projectId: options.projectId } });
   249→    const group = options.concurrencyGroup ?? settings?.concurrencyGroup ?? null;
   250→
   251→    if (group) {
   252→      // Concurrency group: either cancel in-progress runs in the same group
   253→      // (if cancelInProgress) or wait for them (serialized).
   254→      const inProgressInGroup = await db.run.findMany({
   255→        where: {
   256→          projectId: options.projectId,
   257→          concurrencyGroup: group,
   258→          status: { in: ['running', 'queued', 'waiting_approval'] },
   259→        },
   260→      });
   261→
   262→      if (inProgressInGroup.length > 0) {
   263→        if (settings?.cancelInProgress) {
   264→          // Cancel all in-progress runs in this group.
   265→          for (const run of inProgressInGroup) {
   266→            const entry = active.get(run.id);
   267→            if (entry) {
   268→              entry.canceled = true;
   269→              try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   270→            }
   271→            await appendLog(run.id, 'system', `Run canceled: a newer run in group "${group}" was triggered.`);
   272→            await finishRun(run.id, 'canceled', 130);
   273→          }
   274→          await sleep(100);
   275→        } else {
   276→          // Serialized mode: create the run as "queued" and wait in a
   277→          // background poll loop until all in-progress runs in the group
   278→          // reach a terminal state. Then promote this run to "running".
   279→          // This is true queueing (GitHub Actions concurrency: queue).
   280→          const queuedRun = await db.run.create({
   281→            data: {
   282→              projectId: options.projectId,
   283→              workflow: options.workflow,
   284→              status: 'queued',
   285→              trigger: options.trigger ?? 'manual',
   286→              startedAt: new Date(),
   287→              concurrencyGroup: group,
   288→              timeoutMs: options.timeoutMs ?? null,
   289→              requiresApproval: options.requiresApproval ?? false,
   290→              label: options.label ?? null,
   291→            },
   292→          });
   293→          emit({ type: 'status', runId: queuedRun.id, status: 'queued' });
   294→
   295→          // Background: wait for in-progress runs to finish, then execute.
   296→          void (async () => {
   297→            const waitDeadline = Date.now() + 24 * 60 * 60 * 1000; // 24h max wait
   298→            while (Date.now() < waitDeadline) {
   299→              await sleep(2000);
   300→              // Check if any runs in this group are still in-progress.
   301→              const stillRunning = await db.run.findFirst({
   302→                where: {
   303→                  projectId: options.projectId,
   304→                  concurrencyGroup: group,
   305→                  status: { in: ['running', 'queued', 'waiting_approval'] },
   306→                  id: { not: queuedRun.id },
   307→                },
   308→              });
   309→              if (!stillRunning) break;
   310→            }
   311→            // Promote to running and execute.
   312→            await db.run.update({ where: { id: queuedRun.id }, data: { status: 'running' } });
   313→            emit({ type: 'status', runId: queuedRun.id, status: 'running' });
   314→            await appendLog(queuedRun.id, 'system', '▶ Run started (was queued in concurrency group).');
   315→            // Execute the workflow on this run.
   316→            await executeQueuedRun(queuedRun.id, options);
   317→          })();
   318→
   319→          return { runId: queuedRun.id };
   320→        }
   321→      }
   322→    } else if (settings?.concurrentCancellation ?? true) {
   323→      // Default: cancel all in-progress runs on this project.
   324→      const canceled = await cancelInprogressRuns(options.projectId);
   325→      if (canceled.length > 0) {
   326→        await sleep(100);
   327→      }
   328→    }
   329→  }
   330→
   331→  const run = await db.run.create({
   332→    data: {
   333→      projectId: options.projectId,
   334→      workflow: options.workflow,
   335→      status: options.requiresApproval ? 'waiting_approval' : 'running',
   336→      trigger: options.trigger ?? 'manual',
   337→      startedAt: new Date(),
   338→      matrixValues: options.matrixValues ? JSON.stringify(options.matrixValues) : null,
   339→      matrixIndex: options.matrixIndex,
   340→      matrixTotal: options.matrixTotal,
   341→      parentRunId: options.parentRunId,
   342→      pipelineRunId: options.pipelineRunId,
   343→      stageId: options.stageId,
   344→      reRunOfId: options.reRunOfId,
   345→      timeoutMs: options.timeoutMs ?? null,
   346→      requiresApproval: options.requiresApproval ?? false,
   347→      label: options.label ?? null,
   348→      concurrencyGroup: options.concurrencyGroup ?? null,
   349→    },
   350→  });
   351→
   352→  emit({ type: 'status', runId: run.id, status: options.requiresApproval ? 'waiting_approval' : 'running' });
   353→
   354→  // Run in background.
   355→  void (async () => {
   356→    try {
   357→      // Approval gate
   358→      if (options.requiresApproval) {
   359→        const approved = await waitForApproval(run.id);
   360→        if (!approved) {
   361→          await finishRun(run.id, 'canceled', 130);
   362→          return;
   363→        }
   364→        await db.run.update({ where: { id: run.id }, data: { status: 'running' } });
   365→        emit({ type: 'status', runId: run.id, status: 'running' });
   366→      }
   367→
   368→      const steps = workflow.build(detection);
   369→      if (!steps || steps.length === 0) {
   370→        await appendLog(run.id, 'system', `Workflow ${options.workflow} has no steps for this project.`);
   371→        await finishRun(run.id, 'failed', 1);
   372→        return;
   373→      }
   374→
   375→      // Substitute matrix values in step commands.
   376→      const substitutedSteps = options.matrixValues
   377→        ? steps.map(s => ({ ...s, command: substituteMatrix(s.command, options.matrixValues!), label: substituteMatrix(s.label, options.matrixValues!) }))
   378→        : steps;
   379→
   380→      // Build env with secrets + matrix values.
   381→      const secretsList = options.secrets ?? [];
   382→      const extraEnv: Record<string, string> = { ...options.env };
   383→      if (options.matrixValues) {
   384→        for (const [k, v] of Object.entries(options.matrixValues)) {
   385→          extraEnv[`MATRIX_${k.toUpperCase()}`] = v;
   386→        }
   387→      }
   388→      const allSecrets = secretsList.length > 0 ? await getAllSecrets(options.projectId) : {};
   389→
   390→      // Cache restore.
   391→      if (options.cache?.restore) {
   392→        const cacheExists = await hasCache(options.projectId, options.cache.key);
   393→        if (cacheExists) {
   394→          const restored = await restoreCache(options.projectId, options.cache.key);
   395→          if (restored.hit) {
   396→            emit({ type: 'cache-hit', runId: run.id, cacheLabel: options.cache.label });
   397→            await appendLog(run.id, 'system', `💾 Cache hit: ${options.cache.label} (${formatBytes(restored.size ?? 0)})`);
   398→          }
   399→        } else {
   400→          emit({ type: 'cache-miss', runId: run.id, cacheLabel: options.cache.label });
   401→          await appendLog(run.id, 'system', `💾 Cache miss: ${options.cache.label}`);
   402→        }
   403→      }
   404→
   405→      // Execute.
   406→      const startedAt = Date.now();
   407→      let lastExitCode = 0;
   408→
   409→      if (options.workflow === 'parse' || options.workflow === 'bundle') {
   410→        lastExitCode = await runAxiomWorkflow(run.id, project.extractedPath, options.workflow, options.matrixValues);
   411→      } else {
   412→        for (const step of substitutedSteps) {
   413→          if (active.get(run.id)?.canceled) {
   414→            await finishRun(run.id, 'canceled', 130, Date.now() - startedAt);
   415→            return;
   416→          }
   417→          const maxAttempts = (options.retry ?? 0) + 1;
   418→          let attempt = 0;
   419→          let stepExitCode = 1;
   420→          while (attempt < maxAttempts) {
   421→            if (attempt > 0) {
   422→              emit({ type: 'retry', runId: run.id, retryAttempt: attempt });
   423→              await appendLog(run.id, 'system', `↻ Retry ${attempt}/${maxAttempts - 1} for: ${step.label}`);
   424→            }
   425→            await appendLog(run.id, 'system', `▶ ${step.label}`);
   426→            stepExitCode = await runShellStep(
   427→              run.id,
   428→              step,
   429→              project.extractedPath,
   430→              { secrets: allSecrets, extraEnv, timeoutMs: options.timeoutMs },
   431→            );
   432→            if (stepExitCode === 0) break;
   433→            attempt++;
   434→          }
   435→          if (stepExitCode !== 0) {
   436→            lastExitCode = stepExitCode;
   437→            await appendLog(run.id, 'system', `✗ Step failed with exit code ${stepExitCode} after ${attempt} attempt(s).`);
   438→            await finishRun(run.id, 'failed', stepExitCode, Date.now() - startedAt);
   439→            // Cache save even on failure (partial cache is still useful).
   440→            if (options.cache?.save) {
   441→              try {
   442→                await saveCache(options.projectId, options.cache.key, options.cache.label, options.cache.paths);
   443→                emit({ type: 'cache-saved', runId: run.id, cacheLabel: options.cache.label });
   444→                await appendLog(run.id, 'system', `💾 Cache saved: ${options.cache.label}`);
   445→              } catch (err) {
   446→                await appendLog(run.id, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   447→              }
   448→            }
   449→            return;
   450→          }
   451→          await appendLog(run.id, 'system', `✓ ${step.label}`);
   452→        }
   453→      }
   454→
   455→      // Test report capture.
   456→      if (options.testReport) {
   457→        try {
   458→          const reportPath = path.join(project.extractedPath, options.testReport.path);
   459→          if (fs.existsSync(reportPath)) {
   460→            const content = fs.readFileSync(reportPath, 'utf-8');
   461→            const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   462→            let report: ParsedTestReport;
   463→            if (options.testReport.format === 'junit') report = parseJUnit(content);
   464→            else if (options.testReport.format === 'json') report = parseJSONReport(content);
   465→            else report = parseTAP(content);
   466→            await db.testReport.create({
   467→              data: {
   468→                runId: run.id,
   469→                format: report.format,
   470→                total: report.total,
   471→                passed: report.passed,
   472→                failed: report.failed,
   473→                skipped: report.skipped,
   474→                duration: report.duration ?? null,
   475→                suites: JSON.stringify(report.suites),
   476→              },
   477→            });
   478→            await appendLog(run.id, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   479→          }
   480→        } catch (err) {
   481→          await appendLog(run.id, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   482→        }
   483→      }
   484→
   485→      // Capture artifacts.
   486→      await captureArtifacts(run.id, options.workflow, detection, project.extractedPath);
   487→
   488→      // Cache save on success.
   489→      if (options.cache?.save) {
   490→        try {
   491→          const { size } = await saveCache(options.projectId, options.cache.key, options.cache.label, options.cache.paths);
   492→          emit({ type: 'cache-saved', runId: run.id, cacheLabel: options.cache.label });
   493→          await appendLog(run.id, 'system', `💾 Cache saved: ${options.cache.label} (${formatBytes(size)})`);
   494→        } catch (err) {
   495→          await appendLog(run.id, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   496→        }
   497→      }
   498→
   499→      await finishRun(run.id, 'success', lastExitCode, Date.now() - startedAt);
   500→    } catch (err) {
   501→      await appendLog(run.id, 'system', `Runner crashed: ${err instanceof Error ? err.message : String(err)}`);
   502→      await finishRun(run.id, 'failed', -1);
   503→    }
   504→  })();
   505→
   506→  return { runId: run.id };
   507→}

   519→ * Execute a queued run after its concurrency group is free.
   520→ * Reuses the same logic as startRunExtended but skips the concurrency check
   521→ * (the run is already queued, so it's allowed to run now).
   522→ */
   523→async function executeQueuedRun(runId: string, options: RunOptions): Promise<void> {
   524→  try {
   525→    const project = await db.project.findUnique({ where: { id: options.projectId } });
   526→    if (!project) throw new Error(`Project ${options.projectId} not found`);
   527→    const workflow = getWorkflow(options.workflow);
   528→    if (!workflow) throw new Error(`Unknown workflow: ${options.workflow}`);
   529→
   530→    const detection = JSON.parse(project.detection) as Detection;
   531→    const steps = workflow.build(detection);
   532→    if (!steps || steps.length === 0) {
   533→      await appendLog(runId, 'system', `Workflow ${options.workflow} has no steps.`);
   534→      await finishRun(runId, 'failed', 1);
   535→      return;
   536→    }
   537→
   538→    const secretsList = options.secrets ?? [];
   539→    const extraEnv: Record<string, string> = { ...options.env };
   540→    const allSecrets = secretsList.length > 0 ? await getAllSecrets(options.projectId) : {};
   541→
   542→    const startedAt = Date.now();
   543→    let lastExitCode = 0;
   544→
   545→    for (const step of steps) {
   546→      if (active.get(runId)?.canceled) {
   547→        await finishRun(runId, 'canceled', 130, Date.now() - startedAt);
   548→        return;
   549→      }
   550→      await appendLog(runId, 'system', `▶ ${step.label}`);
   551→      const stepExitCode = await runShellStep(
   552→        runId,
   553→        step,
   554→        project.extractedPath,
   555→        { secrets: allSecrets, extraEnv, timeoutMs: options.timeoutMs },
   556→      );
   557→      if (stepExitCode !== 0) {
   558→        lastExitCode = stepExitCode;
   559→        await appendLog(runId, 'system', `✗ Step failed with exit code ${stepExitCode}.`);
   560→        await finishRun(runId, 'failed', stepExitCode, Date.now() - startedAt);
   561→        return;
   562→      }
   563→      await appendLog(runId, 'system', `✓ ${step.label}`);
   564→    }
   565→
   566→    // Capture artifacts.
   567→    await captureArtifacts(runId, options.workflow, detection, project.extractedPath);
   568→    await finishRun(runId, 'success', lastExitCode, Date.now() - startedAt);
   569→  } catch (err) {
   570→    await appendLog(runId, 'system', `Runner crashed: ${err instanceof Error ? err.message : String(err)}`);
   571→    await finishRun(runId, 'failed', -1);
   572→  }
   573→}

   575→export async function cancelRun(runId: string): Promise<void> {
   576→  const entry = active.get(runId);
   577→  if (entry) {
   578→    entry.canceled = true;
   579→    try { entry.child.kill('SIGTERM'); } catch { /* ignore */ }
   580→    if (entry.timeoutTimer) clearTimeout(entry.timeoutTimer);
   581→  }
   582→  await appendLog(runId, 'system', 'Run canceled by user');
   583→  await finishRun(runId, 'canceled', 130);
   584→}

   596→async function runShellStep(
   597→  runId: string,
   598→  step: WorkflowStep,
   599→  cwd: string,
   600→  envOptions: StepEnvOptions,
   601→): Promise<number> {
   602→  return new Promise((resolve) => {
   603→    // Security: block known-dangerous commands.
   604→    const BLOCKED_PATTERNS = [
   605→      /\/etc\/(passwd|shadow|sudoers)/i,
   606→      /\/proc\/self\/(environ|cmdline)/i,
   607→      /\brm\s+-rf\s+\//i,
   608→      /\bmkfs\b/i,
   609→      /\bdd\s+if=\/dev\//i,
   610→      /:\(\)\s*\{/i,  // fork bomb
   611→    ];
   612→    for (const pattern of BLOCKED_PATTERNS) {
   613→      if (pattern.test(step.command)) {
   614→        void appendLog(runId, 'system', `🚫 Command blocked by security policy: ${step.command.slice(0, 80)}`);
   615→        resolve(126);
   616→        return;
   617→      }
   618→    }
   619→
   620→    const projectRoot = cwd;
   621→    buildProcessEnv('', { extraEnv: envOptions.extraEnv, projectRoot }).then((baseEnv) => {
   622→      const fullEnv = { ...baseEnv, ...envOptions.secrets };
   623→      const child = spawn('bash', ['-c', step.command], {
   624→        cwd,
   625→        env: fullEnv as NodeJS.ProcessEnv,
   626→      }) as import('node:child_process').ChildProcessWithoutNullStreams;
   627→      active.set(runId, { child, canceled: false });
   628→
   629→      // Timeout
   630→      let timeoutTimer: ReturnType<typeof setTimeout> | undefined;
   631→      let killTimer: ReturnType<typeof setTimeout> | undefined;
   632→      if (envOptions.timeoutMs) {
   633→        timeoutTimer = setTimeout(() => {
   634→          try { child.kill('SIGTERM'); } catch { /* ignore */ }
   635→          void appendLog(runId, 'system', `⏱ Step timed out after ${envOptions.timeoutMs}ms. Killing process...`);
   636→          // Force-kill with SIGKILL after 2 seconds if still alive.
   637→          killTimer = setTimeout(() => {
   638→            try { child.kill('SIGKILL'); } catch { /* ignore */ }
   639→            void appendLog(runId, 'system', `⏱ Force-killed (SIGKILL).`);
   640→          }, 2000);
   641→        }, envOptions.timeoutMs);
   642→        active.get(runId)!.timeoutTimer = timeoutTimer;
   643→      }
   644→
   645→      const stdout = createInterface({ input: child.stdout });
   646→      const stderr = createInterface({ input: child.stderr });
   647→      stdout.on('line', (line) => { void appendLog(runId, 'stdout', maskSecrets(line, envOptions.secrets)); });
   648→      stderr.on('line', (line) => { void appendLog(runId, 'stderr', maskSecrets(line, envOptions.secrets)); });
   649→
   650→      child.on('close', (code) => {
   651→        if (timeoutTimer) clearTimeout(timeoutTimer);
   652→        if (killTimer) clearTimeout(killTimer);
   653→        active.delete(runId);
   654→        resolve(code ?? 0);
   655→      });
   656→      child.on('error', (err) => {
   657→        if (timeoutTimer) clearTimeout(timeoutTimer);
   658→        if (killTimer) clearTimeout(killTimer);
   659→        void appendLog(runId, 'system', `spawn error: ${err.message}`);
   660→        active.delete(runId);
   661→        resolve(1);
   662→      });
   663→    });
   664→  });
   665→}

   667→// ---------------------------------------------------------------------------
   668→// AxiomState workflow (parse / bundle)
   669→// ---------------------------------------------------------------------------
   670→
   671→async function runAxiomWorkflow(
   672→  runId: string,
   673→  projectRoot: string,
   674→  key: 'parse' | 'bundle',
   675→  matrixValues?: MatrixRow,
   676→): Promise<number> {
   677→  try {
   678→    const { parseProject, writeGraph, sliceForward } = await import('@/lib/axiomstate/phase1');
   679→    const { bundleFiles } = await import('@/lib/axiomstate/phase2');
   680→    const { LSSKernel } = await import('@/lib/axiomstate/phase0/kernel');
   681→
   682→    const kernelDir = path.join(extractDir(''), '..', `kernel-${runId}`);
   683→    fs.mkdirSync(kernelDir, { recursive: true });
   684→    const kernel = new LSSKernel(kernelDir);
   685→
   686→    await appendLog(runId, 'system', `AxiomState: parsing project at ${projectRoot}`);
   687→    const delta = parseProject(projectRoot);
   688→    const seq = writeGraph(kernel, delta, { checkpoint: false, providerName: 'typescript+regex' });
   689→    await appendLog(runId, 'stdout', `Parsed ${delta.nodes.length} nodes and ${delta.edges.length} edges (seq=${seq}).`);
   690→    if (matrixValues) {
   691→      await appendLog(runId, 'system', `Matrix: ${JSON.stringify(matrixValues)}`);
   692→    }
   693→
   694→    if (key === 'parse') {
   695→      for (const node of delta.nodes) {
   696→        await appendLog(runId, 'stdout', `  ${node.kind.padEnd(6)} ${node.id}  (deps: ${node.deps.length})`);
   697→      }
   698→    } else {
   699→      const indexNode = delta.nodes.find(n => n.kind === 'file' && /index\.(ts|js|tsx|jsx)$/.test(n.name));
   700→      if (!indexNode) {
   701→        await appendLog(runId, 'stderr', 'No index.(ts|js|tsx|jsx) entry found — cannot bundle.');
   702→        return 1;
   703→      }
   704→      await appendLog(runId, 'system', `AxiomState: bundling forward slice from ${indexNode.id}`);
   705→      const slice = sliceForward(kernel, indexNode.id);
   706→      const fileIds = Array.from(slice.nodes.values()).filter(n => n.kind === 'file').map(n => n.id);
   707→      const bundle = bundleFiles(kernel, fileIds);
   708→      await appendLog(runId, 'stdout', `Bundle order (${bundle.order.length} files):`);
   709→      for (const id of bundle.order) await appendLog(runId, 'stdout', `  • ${id}`);
   710→      if (bundle.cycles.length > 0) {
   711→        await appendLog(runId, 'stderr', `Cycles detected: ${bundle.cycles.length}`);
   712→        for (const cycle of bundle.cycles) await appendLog(runId, 'stderr', `  ↻ ${cycle.join(' → ')}`);
   713→      }
   714→      const artifactDir = runArtifactDir(runId);
   715→      const outPath = path.join(artifactDir, 'bundle.js');
   716→      const parts: Buffer[] = [];
   717→      for (const entry of bundle.entries) {
   718→        parts.push(Buffer.from(`\n// --- ${entry.path} ---\n`));
   719→        parts.push(Buffer.from(entry.content));
   720→      }
   721→      fs.writeFileSync(outPath, Buffer.concat(parts));
   722→      const stat = fs.statSync(outPath);
   723→      const artifact = await db.artifact.create({
   724→        data: { runId, name: 'bundle.js', path: outPath, size: stat.size, mime: 'text/javascript' },
   725→      });
   726→      emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   727→      await appendLog(runId, 'system', `Artifact written: bundle.js (${stat.size} bytes)`);
   728→    }
   729→
   730→    kernel.close();
   731→    try { fs.rmSync(kernelDir, { recursive: true, force: true }); } catch { /* ignore */ }
   732→    return 0;
   733→  } catch (err) {
   734→    await appendLog(runId, 'stderr', `AxiomState error: ${err instanceof Error ? err.message : String(err)}`);
   735→    return 1;
   736→  }
   737→}

   739→// ---------------------------------------------------------------------------
   740→// Artifacts
   741→// ---------------------------------------------------------------------------
   742→
   743→async function captureArtifacts(
   744→  runId: string,
   745→  workflowKey: string,
   746→  detection: Detection,
   747→  projectRoot: string,
   748→): Promise<void> {
   749→  const workflow = getWorkflow(workflowKey);
   750→  if (!workflow?.producesArtifacts) return;
   751→  const specs = workflow.producesArtifacts(detection, projectRoot);
   752→  if (specs.length === 0) return;
   753→
   754→  const artifactDir = runArtifactDir(runId);
   755→  for (const spec of specs) {
   756→    // Case 1: directory glob like "dist/**/*" — zip the whole directory.
   757→    const dirMatch = spec.glob.match(/^(.+?)\/\*\*\/\*$/);
   758→    if (dirMatch) {
   759→      const dir = dirMatch[1]!;
   760→      const fullDir = path.join(projectRoot, dir);
   761→      if (!fs.existsSync(fullDir)) continue;
   762→      const zipPath = path.join(artifactDir, `${spec.name}.zip`);
   763→      await zipDirectory(fullDir, zipPath);
   764→      const stat = fs.statSync(zipPath);
   765→      const artifact = await db.artifact.create({
   766→        data: { runId, name: `${spec.name}.zip`, path: zipPath, size: stat.size, mime: 'application/zip' },
   767→      });
   768→      emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   769→      await appendLog(runId, 'system', `Artifact captured: ${artifact.name} (${stat.size} bytes)`);
   770→      continue;
   771→    }
   772→
   773→    // Case 2: single file like "forge-apk-output/app-release.apk" — copy as-is.
   774→    const fullPath = path.join(projectRoot, spec.glob);
   775→    if (!fs.existsSync(fullPath)) {
   776→      await appendLog(runId, 'system', `Artifact not found at expected path: ${spec.glob}`);
   777→      continue;
   778→    }
   779→    const stat = fs.statSync(fullPath);
   780→    if (stat.isDirectory()) continue; // skip dirs without /**/*
   781→    // Copy the file to the artifact directory, preserving the original name.
   782→    const destName = spec.name || path.basename(fullPath);
   783→    const destPath = path.join(artifactDir, destName);
   784→    fs.copyFileSync(fullPath, destPath);
   785→    const artifact = await db.artifact.create({
   786→      data: { runId, name: destName, path: destPath, size: stat.size, mime: spec.mime },
   787→    });
   788→    emit({ type: 'artifact', runId, artifact: { id: artifact.id, name: artifact.name, size: artifact.size } });
   789→    await appendLog(runId, 'system', `Artifact captured: ${artifact.name} (${stat.size} bytes)`);
   790→  }
   791→}

   793→async function zipDirectory(srcDir: string, outPath: string): Promise<void> {
   794→  return new Promise((resolve, reject) => {
   795→    const child = spawn('zip', ['-r', '-q', outPath, '.'], { cwd: srcDir });
   796→    child.on('close', (code) => code === 0 ? resolve() : reject(new Error(`zip exited with ${code}`)));
   797→    child.on('error', reject);
   798→  });
   799→}
