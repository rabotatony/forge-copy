     1→"use client";

     3→// ============================================================
     4→// Forge — Experiments Lab
     5→// ============================================================
     6→// A sandboxed page where hard, monitored experiments run on
     7→// Forge itself to discover development breakthroughs. Each
     8→// experiment has a hypothesis, runs in isolation with hard
     9→// timeouts + output caps, and produces a verdict:
    10→//   BREAKTHROUGH → can be promoted to a permanent workflow
    11→//   NO_CHANGE    → ran fine, no breakthrough
    12→//   REGRESSION   → made things worse
    13→//
    14→// Safety rails are enforced in the engine (engine.ts). The UI
    15→// only triggers runs and displays results — it cannot bypass
    16→// the sandbox.
    17→// ============================================================
    18→

    20→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
    21→import {
    22→  FlaskConical,
    23→  Play,
    24→  Loader2,
    25→  CheckCircle2,
    26→  XCircle,
    27→  MinusCircle,

    30→  Sparkles,
    31→  Trophy,
    32→  Clock,

    34→  ChevronRight,

    57→// ---------------------------------------------------------------------------
    58→// Types — mirror the API response shapes
    59→// ---------------------------------------------------------------------------
    60→
    61→type Verdict = "BREAKTHROUGH" | "NO_CHANGE" | "REGRESSION";
    62→type Category = "self-improvement" | "tournament" | "synthesis" | "adversarial" | "recursive" | "breakthrough";
    63→
    64→interface LatestRun {
    65→  id: string;
    66→  status: string;
    67→  verdict: Verdict | null;
    68→  verdictReason: string | null;
    69→  promoted: boolean;
    70→  startedAt: string;
    71→  completedAt: string | null;
    72→  metrics: string | null;
    73→}
    74→
    75→interface ExperimentListItem {
    76→  slug: string;
    77→  name: string;
    78→  category: Category;
    79→  hypothesis: string;
    80→  procedure: string;
    81→  dangerLevel: "safe" | "moderate" | "aggressive";
    82→  dbId: string | null;
    83→  totalRuns: number;
    84→  latestRun: LatestRun | null;
    85→}
    86→
    87→interface ExperimentsResponse {
    88→  experiments: ExperimentListItem[];
    89→  stats: { totalExperiments: number; byCategory: Record<string, number> };
    90→  verdictCounts: Record<string, number>;
    91→  promotedCount: number;
    92→}
    93→
    94→interface RunDetail {
    95→  run: {
    96→    id: string;
    97→    status: string;
    98→    verdict: Verdict | null;
    99→    verdictReason: string | null;
   100→    promoted: boolean;
   101→    promotedWorkflowId: string | null;
   102→    promotedPresetId: string | null;
   103→    startedAt: string;
   104→    completedAt: string | null;
   105→    metrics: Record<string, number | string | boolean> | null;
   106→    evidence: { summary: string; steps: Array<{ step: string; detail: unknown; t?: number }> } | null;
   107→  };
   108→  experiment: {
   109→    slug: string;
   110→    name: string;
   111→    category: Category;
   112→    hypothesis: string;
   113→    procedure: string;
   114→    dangerLevel: string;
   115→  };
   116→}

   196→// ---------------------------------------------------------------------------
   197→// Main component
   198→// ---------------------------------------------------------------------------
   199→
   200→export function ExperimentsLab() {
   201→  const queryClient = useQueryClient();
   202→  const [expandedSlug, setExpandedSlug] = useState<string | null>(null);
   203→  const [selectedRunId, setSelectedRunId] = useState<string | null>(null);
   204→
   205→  const { data, isLoading } = useQuery<ExperimentsResponse>({
   206→    queryKey: ["experiments"],
   207→    queryFn: async () => {
   208→      const r = await fetch("/api/forge/experiments");
   209→      if (!r.ok) throw new Error("Failed to load experiments");
   210→      return r.json();
   211→    },
   212→    refetchInterval: 5000,
   213→  });
   214→
   215→  const runMutation = useMutation({
   216→    mutationFn: async (slug: string) => {
   217→      const r = await fetch(`/api/forge/experiments/${slug}/run`, { method: "POST" });
   218→      if (!r.ok) {
   219→        const e = await r.json().catch(() => ({}));
   220→        throw new Error(e.error ?? "Run failed");
   221→      }
   222→      return r.json() as Promise<{ runId: string; verdict: Verdict | null; status: string }>;
   223→    },
   224→    onSuccess: (result, slug) => {
   225→      queryClient.invalidateQueries({ queryKey: ["experiments"] });
   226→      setSelectedRunId(result.runId);
   227→      setExpandedSlug(slug);
   228→      if (result.verdict === "BREAKTHROUGH") {
   229→        toast.success("Breakthrough discovered!", {
   230→          description: "The experiment found a real improvement. You can promote it.",
   231→        });
   232→      } else if (result.verdict === "REGRESSION") {
   233→        toast.error("Experiment regressed", {
   234→          description: result.status === "timeout" ? "The experiment timed out." : "See the run details.",
   235→        });
   236→      } else {
   237→        toast.info("Experiment completed", { description: "No breakthrough this time." });
   238→      }
   239→    },
   240→    onError: (err: Error) => {
   241→      toast.error("Experiment failed", { description: err.message });
   242→    },
   243→  });
   244→
   245→  const promoteMutation = useMutation({
   246→    mutationFn: async (runId: string) => {
   247→      const r = await fetch(`/api/forge/experiments/runs/${runId}?action=promote`, { method: "POST" });
   248→      if (!r.ok) {
   249→        const e = await r.json().catch(() => ({}));
   250→        throw new Error(e.error ?? "Promotion failed");
   251→      }
   252→      return r.json();
   253→    },
   254→    onSuccess: () => {
   255→      queryClient.invalidateQueries({ queryKey: ["experiments"] });
   256→      queryClient.invalidateQueries({ queryKey: ["experiment-run"] });
   257→      toast.success("Promoted to permanent workflow", {
   258→        description: "The breakthrough is now a reusable Forge workflow + preset.",
   259→      });
   260→    },
   261→    onError: (err: Error) => toast.error("Promotion failed", { description: err.message }),
   262→  });
   263→
   264→  const totalRuns = data
   265→    ? data.experiments.reduce((sum, e) => sum + e.totalRuns, 0)
   266→    : 0;
   267→  const breakthroughs = data?.verdictCounts.BREAKTHROUGH ?? 0;
   268→
   269→  return (
   270→    <div className="space-y-6">
   271→      {/* Header */}
   272→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   273→        <div className="flex items-center gap-3">
   274→          <span className="flex size-10 items-center justify-center rounded-lg bg-violet-500/10 text-violet-600 dark:text-violet-400">
   275→            <FlaskConical className="size-5" aria-hidden />
   276→          </span>
   277→          <div>
   278→            <h1 className="text-xl font-semibold tracking-tight">Experiments Lab</h1>
   279→            <p className="text-sm text-muted-foreground">
   280→              Sandboxed breakthrough discovery — Forge runs hard experiments on itself.
   281→            </p>
   282→          </div>
   283→        </div>
   284→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   285→          <Shield className="size-3.5 text-emerald-500" aria-hidden />
   286→          <span>Sandboxed · time-capped · output-limited</span>
   287→        </div>
   288→      </div>
   289→
   290→      {/* Stats row */}
   291→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   292→        <StatCard
   293→          icon={FlaskConical}
   294→          label="Experiments"
   295→          value={data?.stats.totalExperiments ?? 5}
   296→          color="text-violet-600 dark:text-violet-400"
   297→        />
   298→        <StatCard
   299→          icon={Activity}
   300→          label="Total runs"
   301→          value={totalRuns}
   302→          color="text-sky-600 dark:text-sky-400"
   303→        />
   304→        <StatCard
   305→          icon={Trophy}
   306→          label="Breakthroughs"
   307→          value={breakthroughs}
   308→          color="text-emerald-600 dark:text-emerald-400"
   309→        />
   310→        <StatCard
   311→          icon={Rocket}
   312→          label="Promoted"
   313→          value={data?.promotedCount ?? 0}
   314→          color="text-amber-600 dark:text-amber-400"
   315→        />
   316→      </div>
   317→
   318→      {/* Breakthrough rate bar */}
   319→      {data && totalRuns > 0 && (
   320→        <Card>
   321→          <CardContent className="p-4">
   322→            <div className="flex items-center justify-between text-xs text-muted-foreground">
   323→              <span className="flex items-center gap-1.5">
   324→                <TrendingUp className="size-3.5" aria-hidden />
   325→                Breakthrough rate
   326→              </span>
   327→              <span className="font-mono">
   328→                {breakthroughs}/{totalRuns} runs ({Math.round((breakthroughs / totalRuns) * 100)}%)
   329→              </span>
   330→            </div>
   331→            <Progress
   332→              value={(breakthroughs / totalRuns) * 100}
   333→              className="mt-2 h-2"
   334→            />
   335→            <div className="mt-2 flex flex-wrap gap-2 text-[11px]">
   336→              {(["BREAKTHROUGH", "NO_CHANGE", "REGRESSION"] as Verdict[]).map((v) => (
   337→                <Badge
   338→                  key={v}
   339→                  variant="outline"
   340→                  className={cn("gap-1 font-mono", VERDICT_META[v].color)}
   341→                >
   342→                  {VERDICT_META[v].label}: {data.verdictCounts[v] ?? 0}
   343→                </Badge>
   344→              ))}
   345→            </div>
   346→          </CardContent>
   347→        </Card>
   348→      )}
   349→
   350→      {/* Experiment cards */}
   351→      <div className="space-y-3">
   352→        {isLoading ? (
   353→          <Card>
   354→            <CardContent className="flex items-center justify-center p-12 text-sm text-muted-foreground">
   355→              <Loader2 className="mr-2 size-4 animate-spin" aria-hidden />
   356→              Loading experiments…
   357→            </CardContent>
   358→          </Card>
   359→        ) : (
   360→          data?.experiments.map((exp) => {
   361→            const meta = CATEGORY_META[exp.category] ?? FALLBACK_CATEGORY_META;
   362→            const Icon = meta.icon;
   363→            const isExpanded = expandedSlug === exp.slug;
   364→            const isRunning = runMutation.isPending && runMutation.variables === exp.slug;
   365→            const latest = exp.latestRun;
   366→            const verdict = latest?.verdict;
   367→            const vMeta = verdict ? VERDICT_META[verdict] : null;
   368→            const danger = DANGER_META[exp.dangerLevel];
   369→
   370→            return (
   371→              <Card key={exp.slug} className={cn("overflow-hidden transition-shadow", isExpanded && "shadow-md")}>
   372→                <CardHeader className="pb-3">
   373→                  <div className="flex items-start justify-between gap-3">
   374→                    <div className="flex min-w-0 items-start gap-3">
   375→                      <span className={cn("mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-md bg-muted", meta.color)}>
   376→                        <Icon className="size-4" aria-hidden />
   377→                      </span>
   378→                      <div className="min-w-0">
   379→                        <CardTitle className="flex flex-wrap items-center gap-2 text-base">
   380→                          <span className="truncate">{exp.name}</span>
   381→                          <Badge variant="outline" className="gap-1 text-[10px] font-normal">
   382→                            <Icon className="size-2.5" aria-hidden />
   383→                            {meta.label}
   384→                          </Badge>
   385→                          <Tooltip>
   386→                            <TooltipTrigger asChild>
   387→                              <Badge variant="outline" className={cn("gap-1 text-[10px] font-normal", danger.color)}>
   388→                                <AlertTriangle className="size-2.5" aria-hidden />
   389→                                {danger.label}
   390→                              </Badge>
   391→                            </TooltipTrigger>
   392→                            <TooltipContent>Danger level: {exp.dangerLevel}</TooltipContent>
   393→                          </Tooltip>
   394→                        </CardTitle>
   395→                        <CardDescription className="mt-1 line-clamp-2 text-xs">
   396→                          {exp.hypothesis}
   397→                        </CardDescription>
   398→                      </div>
   399→                    </div>
   400→                    <div className="flex shrink-0 items-center gap-2">
   401→                      <Button
   402→                        size="sm"
   403→                        variant="default"
   404→                        disabled={isRunning}
   405→                        onClick={() => runMutation.mutate(exp.slug)}
   406→                        className="gap-1.5"
   407→                      >
   408→                        {isRunning ? (
   409→                          <Loader2 className="size-3.5 animate-spin" aria-hidden />
   410→                        ) : (
   411→                          <Play className="size-3.5" aria-hidden />
   412→                        )}
   413→                        {isRunning ? "Running" : "Run"}
   414→                      </Button>
   415→                      <Button
   416→                        size="sm"
   417→                        variant="ghost"
   418→                        onClick={() => setExpandedSlug(isExpanded ? null : exp.slug)}
   419→                        aria-label={isExpanded ? "Collapse" : "Expand"}
   420→                      >
   421→                        {isExpanded ? <ChevronDown className="size-4" /> : <ChevronRight className="size-4" />}
   422→                      </Button>
   423→                    </div>
   424→                  </div>
   425→                </CardHeader>
   426→
   427→                {/* Latest verdict strip */}
   428→                {latest && (
   429→                  <CardContent className="pb-3 pt-0">
   430→                    <div className={cn("flex flex-wrap items-center gap-2 rounded-md border px-3 py-2 text-xs", vMeta?.bg ?? "bg-muted/30 border-border")}>
   431→                      {verdict && (
   432→                        <span className={cn("flex items-center gap-1 font-medium", vMeta?.color)}>
   433→                          {(() => {
   434→                            const VIcon = vMeta!.icon;
   435→                            return <VIcon className="size-3.5" aria-hidden />;
   436→                          })()}
   437→                          {vMeta?.label}
   438→                        </span>
   439→                      )}
   440→                      <span className="text-muted-foreground">
   441→                        {latest.verdictReason ?? `Status: ${latest.status}`}
   442→                      </span>
   443→                      <span className="ml-auto flex items-center gap-2 font-mono text-[10px] text-muted-foreground">
   444→                        {latest.promoted && (
   445→                          <Badge variant="outline" className="gap-1 text-[10px] text-emerald-600 dark:text-emerald-400">
   446→                            <Rocket className="size-2.5" aria-hidden />
   447→                            Promoted
   448→                          </Badge>
   449→                        )}
   450→                        <Clock className="size-3" aria-hidden />
   451→                        {new Date(latest.startedAt).toLocaleString()}
   452→                      </span>
   453→                    </div>
   454→                  </CardContent>
   455→                )}
   456→
   457→                {/* Expanded: procedure + runs */}
   458→                {isExpanded && (
   459→                  <CardContent className="border-t border-border pt-4">
   460→                    <div className="grid gap-4 md:grid-cols-2">
   461→                      <div>
   462→                        <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   463→                          Procedure
   464→                        </h4>
   465→                        <p className="text-xs leading-relaxed text-foreground/90">{exp.procedure}</p>
   466→                      </div>
   467→                      <div>
   468→                        <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   469→                          Recent runs
   470→                        </h4>
   471→                        <RecentRunsList slug={exp.slug} onSelect={setSelectedRunId} selectedRunId={selectedRunId} />
   472→                      </div>
   473→                    </div>
   474→                  </CardContent>
   475→                )}
   476→              </Card>
   477→            );
   478→          })
   479→        )}
   480→      </div>
   481→
   482→      {/* Run detail drawer (inline panel) */}
   483→      {selectedRunId && (
   484→        <RunDetailPanel
   485→          runId={selectedRunId}
   486→          onClose={() => setSelectedRunId(null)}
   487→          onPromote={(id) => promoteMutation.mutate(id)}
   488→          promoting={promoteMutation.isPending}
   489→        />
   490→      )}
   491→    </div>
   492→  );
   493→}

   515→// ---------------------------------------------------------------------------
   516→// Recent runs list (fetches on expand)
   517→// ---------------------------------------------------------------------------
   518→
   519→function RecentRunsList({ slug, onSelect, selectedRunId }: { slug: string; onSelect: (id: string) => void; selectedRunId: string | null }) {
   520→  const { data, isLoading } = useQuery<{ runs: LatestRun[] }>({
   521→    queryKey: ["experiment-runs", slug],
   522→    queryFn: async () => {
   523→      const r = await fetch(`/api/forge/experiments?slug=${slug}`);
   524→      if (!r.ok) throw new Error("Failed to load runs");
   525→      return r.json();
   526→    },
   527→    refetchInterval: 4000,
   528→  });
   529→
   530→  if (isLoading) {
   531→    return <div className="text-xs text-muted-foreground">Loading runs…</div>;
   532→  }
   533→  const runs = data?.runs ?? [];
   534→  if (runs.length === 0) {
   535→    return (
   536→      <div className="rounded-md border border-dashed border-border p-3 text-xs text-muted-foreground">
   537→        No runs yet. Click <span className="font-medium text-foreground">Run</span> to start this experiment.
   538→      </div>
   539→    );
   540→  }
   541→  return (
   542→    <ScrollArea className="max-h-64">
   543→      <div className="space-y-1.5 pr-2">
   544→        {runs.map((run) => {
   545→          const vMeta = run.verdict ? VERDICT_META[run.verdict] : null;
   546→          return (
   547→            <button
   548→              key={run.id}
   549→              type="button"
   550→              onClick={() => onSelect(run.id)}
   551→              className={cn(
   552→                "flex w-full items-center gap-2 rounded-md border px-2.5 py-1.5 text-left text-xs transition-colors hover:bg-accent",
   553→                selectedRunId === run.id ? "border-primary/40 bg-accent" : "border-border",
   554→              )}
   555→            >
   556→              {run.verdict ? (
   557→                <span className={cn("flex size-4 shrink-0 items-center justify-center", vMeta?.color)}>
   558→                  {(() => {
   559→                    const VIcon = vMeta!.icon;
   560→                    return <VIcon className="size-3.5" aria-hidden />;
   561→                  })()}
   562→                </span>
   563→              ) : (
   564→                <Loader2 className="size-3.5 shrink-0 animate-spin text-muted-foreground" aria-hidden />
   565→              )}
   566→              <span className="min-w-0 flex-1 truncate font-mono text-[10px] text-muted-foreground">
   567→                {new Date(run.startedAt).toLocaleTimeString()}
   568→              </span>
   569→              {run.promoted && (
   570→                <Rocket className="size-3 shrink-0 text-emerald-500" aria-hidden />
   571→              )}
   572→            </button>
   573→          );
   574→        })}
   575→      </div>

   584→function RunDetailPanel({
   585→  runId,
   586→  onClose,
   587→  onPromote,
   588→  promoting,
   589→}: {
   590→  runId: string;
   591→  onClose: () => void;
   592→  onPromote: (id: string) => void;
   593→  promoting: boolean;
   594→}) {
   595→  const { data, isLoading } = useQuery<RunDetail>({
   596→    queryKey: ["experiment-run", runId],
   597→    queryFn: async () => {
   598→      const r = await fetch(`/api/forge/experiments/runs/${runId}`);
   599→      if (!r.ok) throw new Error("Failed to load run");
   600→      return r.json();
   601→    },
   602→    refetchInterval: 3000,
   603→  });
   604→
   605→  const run = data?.run;
   606→  const exp = data?.experiment;
   607→  const verdict = run?.verdict;
   608→  const vMeta = verdict ? VERDICT_META[verdict] : null;
   609→  const canPromote = verdict === "BREAKTHROUGH" && !run?.promoted;
   610→
   611→  return (
   612→    <Card className="border-primary/30 shadow-lg">
   613→      <CardHeader className="pb-3">
   614→        <div className="flex items-center justify-between gap-2">
   615→          <div className="flex items-center gap-2">
   616→            {verdict && vMeta ? (
   617→              <span className={cn("flex size-8 items-center justify-center rounded-md", vMeta.bg, vMeta.color)}>
   618→                {(() => {
   619→                  const VIcon = vMeta.icon;
   620→                  return <VIcon className="size-4" aria-hidden />;
   621→                })()}
   622→              </span>
   623→            ) : (
   624→              <Loader2 className="size-5 animate-spin text-muted-foreground" aria-hidden />
   625→            )}
   626→            <div>
   627→              <CardTitle className="text-base">{exp?.name ?? "Run detail"}</CardTitle>
   628→              <CardDescription className="font-mono text-[10px]">{runId}</CardDescription>
   629→            </div>
   630→          </div>
   631→          <div className="flex items-center gap-2">
   632→            {canPromote && (
   633→              <Button
   634→                size="sm"
   635→                onClick={() => onPromote(runId)}
   636→                disabled={promoting}
   637→                className="gap-1.5"
   638→              >
   639→                {promoting ? <Loader2 className="size-3.5 animate-spin" /> : <Rocket className="size-3.5" />}
   640→                Promote
   641→              </Button>
   642→            )}
   643→            <Button size="sm" variant="ghost" onClick={onClose}>
   644→              Close
   645→            </Button>
   646→          </div>
   647→        </div>
   648→      </CardHeader>
   649→      <CardContent className="space-y-4">
   650→        {isLoading ? (
   651→          <div className="flex items-center gap-2 py-8 text-sm text-muted-foreground">
   652→            <Loader2 className="size-4 animate-spin" /> Loading run…
   653→          </div>
   654→        ) : (
   655→          <>
   656→            {run?.verdictReason && (
   657→              <div className={cn("rounded-md border px-3 py-2 text-sm", vMeta?.bg ?? "bg-muted/30 border-border")}>
   658→                <span className={cn("font-medium", vMeta?.color)}>{vMeta?.label}: </span>
   659→                <span className="text-foreground/90">{run.verdictReason}</span>
   660→              </div>
   661→            )}
   662→
   663→            {/* Metrics */}
   664→            {run?.metrics && (
   665→              <div>
   666→                <h4 className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   667→                  Metrics
   668→                </h4>
   669→                <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   670→                  {Object.entries(run.metrics).map(([k, v]) => (
   671→                    <div key={k} className="rounded-md border border-border bg-muted/30 px-2.5 py-1.5">
   672→                      <div className="truncate text-[10px] text-muted-foreground">{k}</div>
   673→                      <div className="font-mono text-sm font-medium tabular-nums">{String(v)}</div>
   674→                    </div>
   675→                  ))}
   676→                </div>
   677→              </div>
   678→            )}
   679→
   680→            {/* Summary */}
   681→            {run?.evidence?.summary && (
   682→              <div>
   683→                <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   684→                  Summary
   685→                </h4>
   686→                <p className="text-sm text-foreground/90">{run.evidence.summary}</p>
   687→              </div>
   688→            )}
   689→
   690→            {/* Procedure trace */}
   691→            {run?.evidence?.steps && run.evidence.steps.length > 0 && (
   692→              <div>
   693→                <h4 className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   694→                  Procedure trace ({run.evidence.steps.length} events)
