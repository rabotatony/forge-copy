    39→    39→export interface ValidationResult {
    40→    40→  valid: boolean;
    41→    41→  errors: string[];
    42→    42→}
    43→    43→
    44→    44→/**
    45→    45→ * Validate a PipelineDefinition:
    46→    46→ *   - stages must be non-empty
    47→    47→ *   - all stage IDs unique
    48→    48→ *   - all `needs` reference existing stages
    49→    49→ *   - no cycles
    50→    50→ */
    51→    51→export function validatePipelineDefinition(def: PipelineDefinition): ValidationResult {
    52→    52→  const errors: string[] = [];
    53→    53→  if (!def.stages || def.stages.length === 0) {
    54→    54→    errors.push('Pipeline must have at least one stage.');
    55→    55→    return { valid: false, errors };
    56→    56→  }
    57→    57→  const ids = new Set<string>();
    58→    58→  for (const stage of def.stages) {
    59→    59→    if (!stage.id) {
    60→    60→      errors.push(`Stage "${stage.name ?? '<unnamed>'}" is missing an id.`);
    61→    61→      continue;
    62→    62→    }
    63→    63→    if (ids.has(stage.id)) {
    64→    64→      errors.push(`Duplicate stage id: "${stage.id}".`);
    65→    65→    }
    66→    66→    ids.add(stage.id);
    67→    67→  }
    68→    68→  // Check `needs` references.
    69→    69→  for (const stage of def.stages) {
    70→    70→    if (!stage.needs) continue;
    71→    71→    for (const need of stage.needs) {
    72→    72→      if (!ids.has(need)) {
    73→    73→        errors.push(`Stage "${stage.id}" needs unknown stage "${need}".`);
    74→    74→      }
    75→    75→      if (need === stage.id) {
    76→    76→        errors.push(`Stage "${stage.id}" cannot need itself.`);
    77→    77→      }
    78→    78→    }
    79→    79→  }
    80→    80→  // Cycle detection (DFS).
    81→    81→  const cycle = detectCycle(def.stages);
    82→    82→  if (cycle) {
    83→    83→    errors.push(`Pipeline has a cycle: ${cycle.join(' → ')}.`);
    84→    84→  }
    85→    85→  return { valid: errors.length === 0, errors };
    86→    86→}
    87→    87→
    88→    88→function detectCycle(stages: PipelineStage[]): string[] | null {
    89→    89→  const adj = new Map<string, string[]>();
    90→    90→  for (const s of stages) adj.set(s.id, s.needs ?? []);
    91→    91→  const visited = new Set<string>();
    92→    92→  const path = new Set<string>();
    93→    93→  const pathList: string[] = [];
    94→    94→
    95→    95→  function dfs(node: string): string[] | null {
    96→    96→    if (path.has(node)) {
    97→    97→      // Found a cycle — return the cycle path from the first occurrence.
    98→    98→      const start = pathList.indexOf(node);
    99→    99→      return [...pathList.slice(start), node];
   100→   100→    }
   101→   101→    if (visited.has(node)) return null;
   102→   102→    visited.add(node);
   103→   103→    path.add(node);
   104→   104→    pathList.push(node);
   105→   105→    for (const dep of adj.get(node) ?? []) {
   106→   106→      const result = dfs(dep);
   107→   107→      if (result) return result;
   108→   108→    }
   109→   109→    path.delete(node);
   110→   110→    pathList.pop();
   111→   111→    return null;
   112→   112→  }
   113→   113→
   114→   114→  for (const s of stages) {
   115→   115→    const result = dfs(s.id);
   116→   116→    if (result) return result;
   117→   117→  }
   118→   118→  return null;
   119→   119→}

   121→   121→// ---------------------------------------------------------------------------
   122→   122→// Topological ordering (Kahn's algorithm — level by level)
   123→   123→// ---------------------------------------------------------------------------
   124→   124→
   125→   125→/**
   126→   126→ * Group stages into "levels" — sets of stages that can run in parallel
   127→   127→ * because all their `needs` are in earlier levels.
   128→   128→ */
   129→   129→function topologicalLevels(stages: PipelineStage[]): PipelineStage[][] {
   130→   130→  const byId = new Map(stages.map(s => [s.id, s]));
   131→   131→  const remaining = new Set(stages.map(s => s.id));
   132→   132→  const levels: PipelineStage[][] = [];
   133→   133→  while (remaining.size > 0) {
   134→   134→    const ready: PipelineStage[] = [];
   135→   135→    for (const id of remaining) {
   136→   136→      const stage = byId.get(id)!;
   137→   137→      const needs = stage.needs ?? [];
   138→   138→      if (needs.every(n => !remaining.has(n))) {
   139→   139→        ready.push(stage);
   140→   140→      }
   141→   141→    }
   142→   142→    if (ready.length === 0) {
   143→   143→      // Shouldn't happen — cycle detection should have caught this.
   144→   144→      break;
   145→   145→    }
   146→   146→    for (const s of ready) remaining.delete(s.id);
   147→   147→    levels.push(ready);
   148→   148→  }
   149→   149→  return levels;
   150→   150→}

   156→   156→interface ConditionContext {
   157→   157→  matrix: MatrixRow;
   158→   158→  // Status of all needed stages: { stageId: 'success' | 'failed' | 'skipped' | 'canceled' }
   159→   159→  neededResults: Record<string, string>;
   160→   160→}
   161→   161→
   162→   162→/**
   163→   163→ * Evaluate a stage `if` condition.
   164→   164→ *
   165→   165→ * Supported syntax:
   166→   166→ *   • `always()`                    — always true
   167→   167→ *   • `success()`                   — true if all needed stages succeeded
   168→   168→ *   • `failure()`                   — true if any needed stage failed/canceled
   169→   169→ *   • `matrix.KEY == 'value'`       — string equality
   170→   170→ *   • `matrix.KEY != 'value'`       — string inequality
   171→   171→ *   • `matrix.KEY == "value"`       — string equality (double quotes)
   172→   172→ *   • `matrix.KEY != "value"`       — string inequality (double quotes)
   173→   173→ *   • Combinations via `&&`, `||`, `!`, and parentheses.
   174→   174→ *
   175→   175→ * Returns false on any parse error (conservative — skip the stage).
   176→   176→ */
   177→   177→export function evaluateCondition(expr: string, ctx: ConditionContext): boolean {
   178→   178→  if (!expr || expr.trim() === '') return true;
   179→   179→  const allSuccess = Object.values(ctx.neededResults).every(s => s === 'success');
   180→   180→  const anyFailed = Object.values(ctx.neededResults).some(s => s === 'failed' || s === 'canceled');
   181→   181→
   182→   182→  let s = expr.trim();
   183→   183→  // Function-call replacements.
   184→   184→  s = s.replace(/\balways\s*\(\s*\)/g, 'true');
   185→   185→  s = s.replace(/\bsuccess\s*\(\s*\)/g, allSuccess ? 'true' : 'false');
   186→   186→  s = s.replace(/\bfailure\s*\(\s*\)/g, anyFailed ? 'true' : 'false');
   187→   187→  // matrix.KEY == 'value' / != 'value' (single or double quotes).
   188→   188→  s = s.replace(
   189→   189→    /\bmatrix\.(\w+)\s*(==|!=)\s*'([^']*)'/g,
   190→   190→    (_, key, op, val) => (op === '==' ? ((ctx.matrix[key] ?? '') === val) : ((ctx.matrix[key] ?? '') !== val)) ? 'true' : 'false',
   191→   191→  );
   192→   192→  s = s.replace(
   193→   193→    /\bmatrix\.(\w+)\s*(==|!=)\s*"([^"]*)"/g,
   194→   194→    (_, key, op, val) => (op === '==' ? ((ctx.matrix[key] ?? '') === val) : ((ctx.matrix[key] ?? '') !== val)) ? 'true' : 'false',
   195→   195→  );
   196→   196→  // Any remaining matrix.KEY reference → replace with its value (quoted) so
   197→   197→  // bare references work too (treated as truthy non-empty string).
   198→   198→  s = s.replace(/\bmatrix\.(\w+)/g, (_, key) => (ctx.matrix[key] ?? '') ? 'true' : 'false');
   199→   199→
   200→   200→  return evalBoolean(s);
   201→   201→}
   202→   202→
   203→   203→/**
   204→   204→ * Evaluate a reduced boolean expression containing only `true`, `false`,
   205→   205→ * `&&`, `||`, `!`, `(`, `)`, and whitespace. Hand-written recursive descent
   206→   206→ * parser — no `eval` / `Function` constructor, so no code injection risk.
   207→   207→ */
   208→   208→function evalBoolean(s: string): boolean {
   209→   209→  const tokens: string[] = [];
   210→   210→  let i = 0;
   211→   211→  while (i < s.length) {
   212→   212→    const c = s[i];
   213→   213→    if (c === ' ' || c === '\t' || c === '\n' || c === '\r') { i++; continue; }
   214→   214→    if (c === '(' || c === ')') { tokens.push(c); i++; continue; }
   215→   215→    if (c === '!') { tokens.push('!'); i++; continue; }
   216→   216→    if (c === '&' && s[i + 1] === '&') { tokens.push('&&'); i += 2; continue; }
   217→   217→    if (c === '|' && s[i + 1] === '|') { tokens.push('||'); i += 2; continue; }
   218→   218→    if (s.slice(i, i + 4) === 'true') { tokens.push('true'); i += 4; continue; }
   219→   219→    if (s.slice(i, i + 5) === 'false') { tokens.push('false'); i += 5; continue; }
   220→   220→    // Unknown token — bail.
   221→   221→    return false;
   222→   222→  }
   223→   223→  let pos = 0;
   224→   224→  const peek = (): string | undefined => tokens[pos];
   225→   225→  const consume = (): string | undefined => tokens[pos++];
   226→   226→  function parseOr(): boolean {
   227→   227→    let v = parseAnd();
   228→   228→    while (peek() === '||') { consume(); v = parseAnd() || v; }
   229→   229→    return v;
   230→   230→  }
   231→   231→  function parseAnd(): boolean {
   232→   232→    let v = parseNot();
   233→   233→    while (peek() === '&&') { consume(); v = parseNot() && v; }
   234→   234→    return v;
   235→   235→  }
   236→   236→  function parseNot(): boolean {
   237→   237→    if (peek() === '!') { consume(); return !parseNot(); }
   238→   238→    return parsePrimary();
   239→   239→  }
   240→   240→  function parsePrimary(): boolean {
   241→   241→    const t = consume();
   242→   242→    if (t === 'true') return true;
   243→   243→    if (t === 'false') return false;
   244→   244→    if (t === '(') {
   245→   245→      const v = parseOr();
   246→   246→      if (peek() === ')') consume();
   247→   247→      return v;
   248→   248→    }
   249→   249→    return false;
   250→   250→  }
   251→   251→  return parseOr();
   252→   252→}

   254→   254→// ---------------------------------------------------------------------------
   255→   255→// Matrix substitution — uses the shared `substituteMatrix` from
   256→   256→// `./matrix` so behaviour is identical to engine.ts and
   257→   257→// custom-workflow.ts. (Previously each module had its own copy.)
   258→   258→// ---------------------------------------------------------------------------
   259→   259→
   260→   260→function substituteMatrixInRecord(rec: Record<string, string> | undefined, matrix: MatrixRow): Record<string, string> {
   261→   261→  if (!rec) return {};
   262→   262→  const out: Record<string, string> = {};
   263→   263→  for (const [k, v] of Object.entries(rec)) out[k] = substituteMatrix(v, matrix);
   264→   264→  return out;
   265→   265→}

   398→   398→async function runPipelineInBackground(
   399→   399→  pipelineRunId: string,
   400→   400→  projectId: string,
   401→   401→  stages: PipelineStage[],
   402→   402→  config: PipelineDefinition['config'],
   403→   403→  trigger: string,
   404→   404→): Promise<void> {
   405→   405→  try {
   406→   406→    // Check if this is a custom-workflow pipeline (single 'custom' stage with
   407→   407→    // the workflow JSON stored in config.customWorkflow).
   408→   408→    const customWorkflow = (config as { customWorkflow?: unknown }).customWorkflow;
   409→   409→    if (customWorkflow && stages.length === 1 && stages[0]?.workflow === 'custom') {
   410→   410→      const stage = stages[0];
   411→   411→      const stageRun = await db.stageRun.create({
   412→   412→        data: {
   413→   413→          pipelineRunId,
   414→   414→          stageId: stage.id,
   415→   415→          stageName: stage.name,
   416→   416→          status: 'running',
   417→   417→          startedAt: new Date(),
   418→   418→          runIds: '[]',
   419→   419→        },
   420→   420→      });
   421→   421→      try {
   422→   422→        const { runId } = await runCustomWorkflow(projectId, customWorkflow as Parameters<typeof runCustomWorkflow>[1], {
   423→   423→          trigger: trigger as 'manual' | 'auto' | 'webhook' | 'cron' | 'pipeline',
   424→   424→          pipelineRunId,
   425→   425→          stageId: stage.id,
   426→   426→          label: stage.name,
   427→   427→        });
   428→   428→        await db.stageRun.update({
   429→   429→          where: { id: stageRun.id },
   430→   430→          data: { runIds: JSON.stringify([runId]) },
   431→   431→        });
   432→   432→        const finalStatus = await waitForRunCompletion(runId);
   433→   433→        await db.stageRun.update({
   434→   434→          where: { id: stageRun.id },
   435→   435→          data: {
   436→   436→            status: mapRunStatusToStageStatus(finalStatus),
   437→   437→            finishedAt: new Date(),
   438→   438→          },
   439→   439→        });
   440→   440→        await finishPipelineRun(pipelineRunId, finalStatus === 'success' ? 'success' : 'failed');
   441→   441→        return;
   442→   442→      } catch (err) {
   443→   443→        await db.stageRun.update({
   444→   444→          where: { id: stageRun.id },
   445→   445→          data: {
   446→   446→            status: 'failed',
   447→   447→            finishedAt: new Date(),
   448→   448→            error: err instanceof Error ? err.message : String(err),
   449→   449→          },
   450→   450→        });
   451→   451→        await finishPipelineRun(pipelineRunId, 'failed');
   452→   452→        return;
   453→   453→      }
   454→   454→    }
   455→   455→
   456→   456→    // Standard DAG execution: topologically-grouped levels, run in parallel.
   457→   457→    const levels = topologicalLevels(stages);
   458→   458→    const stageResults: Record<string, string> = {}; // stageId → 'success' | 'failed' | 'skipped' | 'canceled'
   459→   459→
   460→   460→    for (const level of levels) {
   461→   461→      // Run all stages in this level in parallel.
   462→   462→      await Promise.all(level.map(stage => runStage(stage, pipelineRunId, projectId, trigger, stageResults, config)));
   463→   463→      // If any stage in this level failed/canceled and the pipeline doesn't
   464→   464→      // allow continuation, we still proceed to the next level — individual
   465→   465→      // stages will skip themselves based on their `needs` results.
   466→   466→    }
   467→   467→
   468→   468→    // Final status: success if no stage failed/canceled; failed otherwise.
   469→   469→    const anyFailed = Object.values(stageResults).some(s => s === 'failed' || s === 'canceled');
   470→   470→    await finishPipelineRun(pipelineRunId, anyFailed ? 'failed' : 'success');
   471→   471→  } catch (err) {
   472→   472→    await db.pipelineRun.update({
   473→   473→      where: { id: pipelineRunId },
   474→   474→      data: {
   475→   475→        status: 'failed',
   476→   476→        finishedAt: new Date(),
   477→   477→      },
   478→   478→    });
   479→   479→    console.error('[forge:pipeline] background execution failed:', err);
   480→   480→  }
   481→   481→}
   482→   482→
   483→   483→async function runStage(
   484→   484→  stage: PipelineStage,
   485→   485→  pipelineRunId: string,
   486→   486→  projectId: string,
   487→   487→  trigger: string,
   488→   488→  stageResults: Record<string, string>,
   489→   489→  config: PipelineDefinition['config'],
   490→   490→): Promise<void> {
   491→   491→  const neededResults: Record<string, string> = {};
   492→   492→  for (const need of stage.needs ?? []) {
   493→   493→    neededResults[need] = stageResults[need] ?? 'skipped';
   494→   494→  }
   495→   495→
   496→   496→  // Create a StageRun row up-front (status: pending).
   497→   497→  const stageRun = await db.stageRun.create({
   498→   498→    data: {
   499→   499→      pipelineRunId,
   500→   500→      stageId: stage.id,
   501→   501→      stageName: stage.name,
   502→   502→      status: 'pending',
   503→   503→      runIds: '[]',
   504→   504→    },
   505→   505→  });
   506→   506→
   507→   507→  // 1. `if` condition.
   508→   508→  try {
   509→   509→    if (stage.if && !evaluateCondition(stage.if, { matrix: {}, neededResults })) {
   510→   510→      await db.stageRun.update({
   511→   511→        where: { id: stageRun.id },
   512→   512→        data: { status: 'skipped', finishedAt: new Date() },
   513→   513→      });
   514→   514→      stageResults[stage.id] = 'skipped';
   515→   515→      return;
   516→   516→    }
   517→   517→  } catch {
   518→   518→    // Parse error → skip conservatively.
   519→   519→    await db.stageRun.update({
   520→   520→      where: { id: stageRun.id },
   521→   521→      data: { status: 'skipped', finishedAt: new Date() },
   522→   522→    });
   523→   523→    stageResults[stage.id] = 'skipped';
   524→   524→    return;
   525→   525→  }
   526→   526→
   527→   527→  // 2. If any `needs` stage failed/canceled/skipped and this stage doesn't
   528→   528→  //    have `if: failure()` (or `if: always()`), skip.
   529→   529→  const needsFailed = Object.values(neededResults).some(s => s === 'failed' || s === 'canceled');
   530→   530→  const needsSkipped = Object.values(neededResults).some(s => s === 'skipped');
   531→   531→  const hasFailureCondition = stage.if?.includes('failure()') ?? false;
   532→   532→  const hasAlwaysCondition = stage.if?.includes('always()') ?? false;
   533→   533→  if ((needsFailed || needsSkipped) && !hasFailureCondition && !hasAlwaysCondition) {
   534→   534→    await db.stageRun.update({
   535→   535→      where: { id: stageRun.id },
   536→   536→      data: { status: 'skipped', finishedAt: new Date() },
   537→   537→    });
   538→   538→    stageResults[stage.id] = 'skipped';
   539→   539→    return;
   540→   540→  }
   541→   541→
   542→   542→  await db.stageRun.update({
   543→   543→    where: { id: stageRun.id },
   544→   544→    data: { status: 'running', startedAt: new Date() },
   545→   545→  });
   546→   546→
   547→   547→  // 3. Expand matrix (if any).
   548→   548→  let matrixRows: MatrixRow[] = [{}];
   549→   549→  if (stage.matrix?.dimensions && stage.matrix.dimensions.length > 0) {
   550→   550→    matrixRows = expandMatrix(
   551→   551→      stage.matrix.dimensions,
   552→   552→      stage.matrix.exclude ?? [],
   553→   553→      stage.matrix.include ?? [],
   554→   554→    );
   555→   555→  }
   556→   556→
   557→   557→  const runIds: string[] = [];
   558→   558→  let stageFailed = false;
   559→   559→  let stageCanceled = false;
   560→   560→
   561→   561→  // 4. Run each matrix row. Use limited concurrency (Promise.all is fine
   562→   562→  //    for typical matrix sizes — they're usually ≤ 10 rows).
   563→   563→  const results = await Promise.all(matrixRows.map(async (matrixValues, idx) => {
   564→   564→    // Per-row env / cache.key / if substitution.
   565→   565→    const rowEnv = substituteMatrixInRecord(stage.env, matrixValues);
   566→   566→    const rowCacheKey = stage.cache ? substituteMatrix(stage.cache.key, matrixValues) : undefined;
   567→   567→
   568→   568→    // Re-evaluate `if` with this matrix row (per-row skip).
   569→   569→    if (stage.if) {
   570→   570→      try {
   571→   571→        if (!evaluateCondition(stage.if, { matrix: matrixValues, neededResults })) {
   572→   572→          return { matrixValues, status: 'skipped' as const, runId: null };
   573→   573→        }
   574→   574→      } catch {
   575→   575→        return { matrixValues, status: 'skipped' as const, runId: null };
   576→   576→      }
   577→   577→    }
   578→   578→
   579→   579→    // 5. Start the run via the extended runner. Always use 'pipeline' as
   580→   580→    //    the trigger so matrix siblings don't cancel each other via the
   581→   581→    //    engine's concurrent-cancellation logic. The user-facing trigger
   582→   582→    //    (manual/webhook/cron) is preserved on the PipelineRun row.
   583→   583→    //
   584→   584→    //    RETRY POLICY: we delegate per-step retry to `startRunExtended`
   585→   585→    //    (which retries only the failing step, not the whole stage).
   586→   586→    //    The previous whole-stage retry loop here was removed because
   587→   587→    //    it caused a quadratic (retry+1)² explosion — a stage with
   588→   588→    //    retry=2 would run each step up to 9 times.
   589→   589→    const retry = stage.retry ?? config?.defaultRetry ?? 0;
   590→   590→    const timeoutMs = stage.timeoutMs ?? config?.defaultTimeoutMs ?? undefined;
   591→   591→
   592→   592→    const { runId } = await startRunExtended({
   593→   593→      projectId,
   594→   594→      workflow: stage.workflow,
   595→   595→      trigger: 'pipeline',
   596→   596→      secrets: stage.secrets,
   597→   597→      env: rowEnv,
   598→   598→      cache: stage.cache
   599→   599→        ? {
   600→   600→            key: rowCacheKey ?? stage.cache.key,
   601→   601→            label: stage.name,
   602→   602→            paths: stage.cache.paths,
   603→   603→            restore: true,
   604→   604→            save: true,
   605→   605→          }
   606→   606→        : undefined,
   607→   607→      retry,
   608→   608→      timeoutMs,
   609→   609→      matrixValues,
   610→   610→      matrixIndex: matrixRows.length > 1 ? idx : undefined,
   611→   611→      matrixTotal: matrixRows.length > 1 ? matrixRows.length : undefined,
   612→   612→      pipelineRunId,
   613→   613→      stageId: stage.id,
   614→   614→      label: matrixRows.length > 1
   615→   615→        ? `${stage.name} (${Object.entries(matrixValues).map(([k, v]) => `${k}=${v}`).join(', ')})`
   616→   616→        : stage.name,
   617→   617→      requiresApproval: stage.requiresApproval,
   618→   618→    });
   619→   619→
   620→   620→    return { matrixValues, status: 'running' as const, runId };
   621→   621→  }));
   622→   622→
   623→   623→  // 6. Wait for all started runs to complete.
   624→   624→  for (const r of results) {
   625→   625→    if (r.runId) {
   626→   626→      runIds.push(r.runId);
   627→   627→      const finalStatus = await waitForRunCompletion(r.runId);
   628→   628→      if (finalStatus === 'failed') stageFailed = true;
   629→   629→      if (finalStatus === 'canceled') stageCanceled = true;
   630→   630→    } else if (r.status === 'skipped') {
   631→   631→      // Per-row skip — counts as not-failed.
   632→   632→    }
   633→   633→  }
   634→   634→
   635→   635→  // 7. Retry policy: per-step retry is handled by `startRunExtended`
   636→   636→    //    (which retries only the failing step). The previous whole-stage
   637→   637→    //    retry loop here was removed because it caused a quadratic
   638→   638→    //    (retry+1)² explosion — see the comment in step 5 above.
   639→   639→
   640→   640→  // 8. Update the StageRun row.
   641→   641→  const finalStageStatus = stageCanceled
   642→   642→    ? 'canceled'
   643→   643→    : stageFailed
   644→   644→      ? 'failed'
   645→   645→      : 'success';
   646→   646→  await db.stageRun.update({
   647→   647→    where: { id: stageRun.id },
   648→   648→    data: {
   649→   649→      status: finalStageStatus,
   650→   650→      finishedAt: new Date(),
   651→   651→      runIds: JSON.stringify(runIds),
   652→   652→      matrixValues: matrixRows.length > 1 ? JSON.stringify(matrixRows) : null,
   653→   653→    },
   654→   654→  });
   655→   655→  stageResults[stage.id] = finalStageStatus;
   656→   656→}
   657→   657→
   658→   658→// `mapRunStatusToStageStatus` was removed — the previous identity
   659→   659→// function was dead code. Run status strings and stage status strings
   660→   660→// use the same vocabulary (running/success/failed/canceled).
   661→   661→
   662→   662→/**
   663→   663→ * Poll the DB every 1s until the run reaches a terminal status.
   664→   664→ * Returns the terminal status. Also handles approval waiting — the run
   665→   665→ * is in `waiting_approval` until approved/rejected, which we treat as a
   666→   666→ * non-terminal status here (we keep waiting).
   667→   667→ */
   668→   668→async function waitForRunCompletion(runId: string): Promise<RunStatus> {
   669→   669→  // Safety cap: 24 hours.
   670→   670→  const deadline = Date.now() + 24 * 60 * 60 * 1000;
   671→   671→  while (Date.now() < deadline) {
   672→   672→    const run = await db.run.findUnique({ where: { id: runId }, select: { status: true } });
   673→   673→    if (!run) return 'failed';
   674→   674→    const status = run.status as RunStatus;
   675→   675→    if (status === 'running' || status === 'queued' || status === 'waiting_approval') {
   676→   676→      await sleep(1000);
   677→   677→      continue;
   678→   678→    }
   679→   679→    return status;
   680→   680→  }
   681→   681→  return 'failed';
   682→   682→}
   683→   683→
   684→   684→function sleep(ms: number): Promise<void> {
   685→   685→  return new Promise(resolve => setTimeout(resolve, ms));
   686→   686→}

   688→   688→async function finishPipelineRun(pipelineRunId: string, status: 'success' | 'failed' | 'canceled'): Promise<void> {
   689→   689→  await db.pipelineRun.update({
   690→   690→    where: { id: pipelineRunId },
   691→   691→    data: { status, finishedAt: new Date() },
   692→   692→  });
   693→   693→  // Best-effort pipeline-level notification (uses the first run's project).
   694→   694→  try {
   695→   695→    const pipelineRun = await db.pipelineRun.findUnique({
   696→   696→      where: { id: pipelineRunId },
   697→   697→      include: { pipeline: true },
   698→   698→    });
   699→   699→    if (pipelineRun) {
   700→   700→      const { notifyRunEvent } = await import('./notifications');
   701→   701→      // Find any run associated with this pipeline run to use as the
   702→   702→      // notification subject. If none, skip.
   703→   703→      const firstRun = await db.run.findFirst({
   704→   704→        where: { pipelineRunId },
   705→   705→        orderBy: { startedAt: 'asc' },
   706→   706→      });
   707→   707→      if (firstRun) {
   708→   708→        // Update the run's status if it doesn't match (rare), then fire.
   709→   709→        await notifyRunEvent(firstRun.id, status);
   710→   710→      }
   711→   711→    }
   712→   712→  } catch {
   713→   713→    // Notifications are best-effort.
   714→   714→  }
   715→   715→}

   802→   802→export async function cancelPipelineRun(pipelineRunId: string): Promise<void> {
   803→   803→  const pipelineRun = await db.pipelineRun.findUnique({
   804→   804→    where: { id: pipelineRunId },
   805→   805→    include: { stageRuns: true },
   806→   806→  });
   807→   807→  if (!pipelineRun) throw new Error(`Pipeline run ${pipelineRunId} not found`);
   808→   808→
   809→   809→  // Cancel all in-progress runs in this pipeline.
   810→   810→  const runs = await db.run.findMany({
   811→   811→    where: {
   812→   812→      pipelineRunId,
   813→   813→      status: { in: ['running', 'queued', 'waiting_approval'] },
   814→   814→    },
   815→   815→  });
   816→   816→  for (const run of runs) {
   817→   817→    try {
   818→   818→      await cancelRun(run.id);
   819→   819→    } catch {
   820→   820→      // Best-effort — continue canceling others.
   821→   821→    }
   822→   822→  }
   823→   823→
   824→   824→  // Mark all pending/running stages as canceled.
   825→   825→  for (const stage of pipelineRun.stageRuns) {
   826→   826→    if (stage.status === 'pending' || stage.status === 'running') {
   827→   827→      await db.stageRun.update({
   828→   828→        where: { id: stage.id },
   829→   829→        data: { status: 'canceled', finishedAt: new Date() },
   830→   830→      });
   831→   831→    }
   832→   832→  }
   833→   833→
   834→   834→  // Set pipeline run status to canceled.
   835→   835→  await db.pipelineRun.update({
   836→   836→    where: { id: pipelineRunId },
   837→   837→    data: { status: 'canceled', finishedAt: new Date() },
   838→   838→  });
   839→   839→}
