     1→// ============================================================
     2→// Forge — user-authored custom workflows (Phase 2)
     3→// ============================================================
     4→// A CustomWorkflow is a JSON document authored by the project owner
     5→// describing a list of shell steps (with per-step env, retry, timeout,
     6→// cache, working dir, test-report capture, and continueOnError). It is
     7→// the lightweight alternative to defining a multi-stage Pipeline.
     8→//
     9→// Custom workflows share the engine's SSE event bus (via `subscribe`,
    10→// `emit`, `appendLog`, `finishRun` from `./engine`) so the existing
    11→// `/api/forge/runs/[id]/events` SSE route picks them up automatically.
    12→//
    13→// Storage: a custom workflow is stored as a Pipeline with a single
    14→// 'custom' stage and the full CustomWorkflow JSON in the pipeline's
    15→// `config.customWorkflow` field. The pipeline executor detects this
    16→// and dispatches to `runCustomWorkflow` instead of `startRunExtended`.
    17→// ============================================================

    45→// Allowed interpreter languages for custom workflow steps.
    46→const STEP_LANGUAGES: readonly CustomWorkflowStepLanguage[] = ['bash', 'node', 'python', 'ruby'] as const;
    47→
    48→/**
    49→ * Coerce an unknown value into a valid CustomWorkflowStepLanguage, or throw.
    50→ * Returns 'bash' when the value is absent (undefined/null). Throws a clear
    51→ * error when the value is present but not one of the allowed languages.
    52→ */
    53→function coerceStepLanguage(raw: unknown, stepIndex: number): CustomWorkflowStepLanguage {
    54→  if (raw === undefined || raw === null) return 'bash';
    55→  if (typeof raw !== 'string') {
    56→    throw new Error(`Step ${stepIndex} has invalid "language" (expected string, got ${typeof raw}).`);
    57→  }
    58→  if (!(STEP_LANGUAGES as readonly string[]).includes(raw)) {
    59→    throw new Error(`Step ${stepIndex} has invalid "language" "${raw}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
    60→  }
    61→  return raw as CustomWorkflowStepLanguage;
    62→}

    68→/**
    69→ * Parse a JSON string into a CustomWorkflow. Throws on invalid input.
    70→ */
    71→export function parseCustomWorkflow(json: string): CustomWorkflow {
    72→  let raw: unknown;
    73→  try {
    74→    raw = JSON.parse(json);
    75→  } catch (err) {
    76→    throw new Error(`Custom workflow is not valid JSON: ${err instanceof Error ? err.message : String(err)}`);
    77→  }
    78→  if (typeof raw !== 'object' || raw === null || Array.isArray(raw)) {
    79→    throw new Error('Custom workflow must be a JSON object.');
    80→  }
    81→  const obj = raw as Record<string, unknown>;
    82→  if (typeof obj.name !== 'string' || obj.name.trim() === '') {
    83→    throw new Error('Custom workflow must have a non-empty "name" string.');
    84→  }
    85→  if (!Array.isArray(obj.steps) || obj.steps.length === 0) {
    86→    throw new Error('Custom workflow must have a non-empty "steps" array.');
    87→  }
    88→  const steps: CustomWorkflowStep[] = [];
    89→  for (let i = 0; i < obj.steps.length; i++) {
    90→    const s = obj.steps[i] as Record<string, unknown>;
    91→    if (typeof s !== 'object' || s === null) {
    92→      throw new Error(`Step ${i} must be an object.`);
    93→    }
    94→    if (typeof s.name !== 'string' || s.name.trim() === '') {
    95→      throw new Error(`Step ${i} must have a non-empty "name".`);
    96→    }
    97→    if (typeof s.run !== 'string' || s.run.trim() === '') {
    98→      throw new Error(`Step ${i} must have a non-empty "run" command.`);
    99→    }
   100→    const step: CustomWorkflowStep = { name: s.name, run: s.run };
   101→    // Per-step interpreter language (defaults to 'bash' when absent).
   102→    const language = coerceStepLanguage(s.language, i);
   103→    if (language !== 'bash') step.language = language;
   104→    if (typeof s.workingDir === 'string') step.workingDir = s.workingDir;
   105→    if (s.env && typeof s.env === 'object') {
   106→      step.env = Object.fromEntries(
   107→        Object.entries(s.env as Record<string, unknown>).filter(([, v]) => typeof v === 'string'),
   108→      ) as Record<string, string>;
   109→    }
   110→    if (typeof s.retry === 'number') step.retry = s.retry;
   111→    if (typeof s.timeoutMs === 'number') step.timeoutMs = s.timeoutMs;
   112→    if (s.cache && typeof s.cache === 'object') {
   113→      const c = s.cache as Record<string, unknown>;
   114→      if (typeof c.key === 'string' && Array.isArray(c.paths)) {
   115→        step.cache = {
   116→          key: c.key,
   117→          paths: (c.paths as unknown[]).filter((p): p is string => typeof p === 'string'),
   118→          restore: c.restore !== false,
   119→          save: c.save !== false,
   120→        };
   121→      }
   122→    }
   123→    if (s.testReport && typeof s.testReport === 'object') {
   124→      const tr = s.testReport as Record<string, unknown>;
   125→      if (typeof tr.format === 'string' && typeof tr.path === 'string') {
   126→        step.testReport = {
   127→          format: tr.format as 'junit' | 'json' | 'tap',
   128→          path: tr.path,
   129→        };
   130→      }
   131→    }
   132→    if (typeof s.continueOnError === 'boolean') step.continueOnError = s.continueOnError;
   133→    steps.push(step);
   134→  }
   135→  const workflow: CustomWorkflow = { name: obj.name, steps };
   136→  if (typeof obj.description === 'string') workflow.description = obj.description;
   137→  if (obj.matrix && typeof obj.matrix === 'object') {
   138→    const m = obj.matrix as Record<string, unknown>;
   139→    if (Array.isArray(m.dimensions)) {
   140→      workflow.matrix = {
   141→        dimensions: (m.dimensions as unknown[])
   142→          .filter((d): d is Record<string, unknown> => typeof d === 'object' && d !== null)
   143→          .map(d => ({
   144→            key: String(d.key ?? ''),
   145→            values: Array.isArray(d.values)
   146→              ? (d.values as unknown[]).filter((v): v is string => typeof v === 'string')
   147→              : [],
   148→          }))
   149→          .filter(d => d.key !== ''),
   150→        exclude: Array.isArray(m.exclude) ? (m.exclude as MatrixRow[]) : undefined,
   151→        include: Array.isArray(m.include) ? (m.include as MatrixRow[]) : undefined,
   152→      };
   153→    }
   154→  }
   155→  if (typeof obj.retry === 'number') workflow.retry = obj.retry;
   156→  if (typeof obj.timeoutMs === 'number') workflow.timeoutMs = obj.timeoutMs;
   157→  if (typeof obj.requiresApproval === 'boolean') workflow.requiresApproval = obj.requiresApproval;
   158→  if (Array.isArray(obj.secrets)) {
   159→    workflow.secrets = (obj.secrets as unknown[]).filter((s): s is string => typeof s === 'string');
   160→  }
   161→  if (obj.env && typeof obj.env === 'object') {
   162→    workflow.env = Object.fromEntries(
   163→      Object.entries(obj.env as Record<string, unknown>).filter(([, v]) => typeof v === 'string'),
   164→    ) as Record<string, string>;
   165→  }
   166→  return workflow;
   167→}

   169→/**
   170→ * Validate a CustomWorkflow object. Returns `{ valid, errors }`.
   171→ */
   172→export function validateCustomWorkflow(workflow: CustomWorkflow): { valid: boolean; errors: string[] } {
   173→  const errors: string[] = [];
   174→  if (!workflow.name || workflow.name.trim() === '') {
   175→    errors.push('Custom workflow must have a non-empty "name".');
   176→  }
   177→  if (!workflow.steps || workflow.steps.length === 0) {
   178→    errors.push('Custom workflow must have at least one step.');
   179→  } else {
   180→    const names = new Set<string>();
   181→    for (let i = 0; i < workflow.steps.length; i++) {
   182→      const s = workflow.steps[i]!;
   183→      if (!s.name || s.name.trim() === '') {
   184→        errors.push(`Step ${i} must have a non-empty "name".`);
   185→        continue;
   186→      }
   187→      if (names.has(s.name)) {
   188→        errors.push(`Duplicate step name: "${s.name}".`);
   189→      }
   190→      names.add(s.name);
   191→      if (!s.run || s.run.trim() === '') {
   192→        errors.push(`Step "${s.name}" must have a non-empty "run" command.`);
   193→      }
   194→      if (s.language !== undefined && !(STEP_LANGUAGES as readonly string[]).includes(s.language)) {
   195→        errors.push(`Step "${s.name}" has invalid "language" "${s.language}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
   196→      }
   197→    }
   198→  }
   199→  if (workflow.matrix?.dimensions) {
   200→    for (const dim of workflow.matrix.dimensions) {
   201→      if (!dim.key || dim.key.trim() === '') {
   202→        errors.push('Matrix dimension must have a non-empty "key".');
   203→      }
   204→      if (!dim.values || dim.values.length === 0) {
   205→        errors.push(`Matrix dimension "${dim.key}" must have at least one value.`);
   206→      }
   207→    }
   208→  }
   209→  return { valid: errors.length === 0, errors };
   210→}

   216→/**
   217→ * Save a custom workflow as a Pipeline with a single 'custom' stage.
   218→ * The full CustomWorkflow JSON is stored in the pipeline's config field
   219→ * under `customWorkflow`, and the pipeline executor dispatches to
   220→ * `runCustomWorkflow` instead of `startRunExtended`.
   221→ */
   222→export async function saveCustomWorkflow(
   223→  projectId: string,
   224→  name: string,
   225→  workflow: CustomWorkflow,
   226→): Promise<{ id: string }> {
   227→  const validation = validateCustomWorkflow(workflow);
   228→  if (!validation.valid) {
   229→    throw new Error(`Invalid custom workflow: ${validation.errors.join(' ')}`);
   230→  }
   231→  // Build a minimal PipelineDefinition with a single 'custom' stage.
   232→  const stages = [
   233→    {
   234→      id: 'custom',
   235→      name: workflow.name,
   236→      workflow: 'custom',
   237→      needs: [] as string[],
   238→      ...(workflow.matrix ? { matrix: workflow.matrix } : {}),
   239→      ...(workflow.retry !== undefined ? { retry: workflow.retry } : {}),
   240→      ...(workflow.timeoutMs !== undefined ? { timeoutMs: workflow.timeoutMs } : {}),
   241→      ...(workflow.requiresApproval ? { requiresApproval: true } : {}),
   242→      ...(workflow.secrets ? { secrets: workflow.secrets } : {}),
   243→      ...(workflow.env ? { env: workflow.env } : {}),
   244→    },
   245→  ];
   246→  const config = { customWorkflow: workflow };
   247→  const pipeline = await db.pipeline.create({
   248→    data: {
   249→      projectId,
   250→      name,
   251→      stages: JSON.stringify(stages),
   252→      config: JSON.stringify(config),
   253→    },
   254→  });
   255→  return { id: pipeline.id };
   256→}

   270→/**
   271→ * Run a CustomWorkflow. Creates a Run row with `workflow: 'custom'` and
   272→ * executes the steps directly (without going through `startRunExtended`,
   273→ * which only knows about predefined workflows). Shares the engine's SSE
   274→ * event bus so the existing `/api/forge/runs/[id]/events` route picks
   275→ * it up.
   276→ *
   277→ * Returns immediately with `{ runId }`; execution continues in the
   278→ * background.
   279→ */
   280→export async function runCustomWorkflow(
   281→  projectId: string,
   282→  workflow: CustomWorkflow,
   283→  options: RunCustomWorkflowOptions = {},
   284→): Promise<{ runId: string }> {
   285→  const project = await db.project.findUnique({ where: { id: projectId } });
   286→  if (!project) throw new Error(`Project ${projectId} not found`);
   287→
   288→  const run = await db.run.create({
   289→    data: {
   290→      projectId,
   291→      workflow: 'custom',
   292→      status: options.matrixValues ? 'running' : 'running',
   293→      trigger: options.trigger ?? 'manual',
   294→      startedAt: new Date(),
   295→      matrixValues: options.matrixValues ? JSON.stringify(options.matrixValues) : null,
   296→      pipelineRunId: options.pipelineRunId,
   297→      stageId: options.stageId,
   298→      label: options.label ?? workflow.name,
   299→    },
   300→  });
   301→
   302→  emit({ type: 'status', runId: run.id, status: 'running' });
   303→
   304→  // Fire-and-forget execution.
   305→  void executeCustomSteps(run.id, project.extractedPath, workflow, projectId, options);
   306→
   307→  return { runId: run.id };
   308→}
   309→
   310→async function executeCustomSteps(
   311→  runId: string,
   312→  projectRoot: string,
   313→  workflow: CustomWorkflow,
   314→  projectId: string,
   315→  options: RunCustomWorkflowOptions,
   316→): Promise<void> {
   317→  const startedAt = Date.now();
   318→  try {
   319→    // Resolve secrets (workflow-level + per-step).
   320→    const secretKeys = new Set<string>(workflow.secrets ?? []);
   321→    for (const step of workflow.steps) {
   322→      // Per-step secrets aren't in CustomWorkflowStep; only workflow-level.
   323→      void step;
   324→    }
   325→    const allSecrets = secretKeys.size > 0
   326→      ? await getSecrets(projectId, Array.from(secretKeys))
   327→      : {};
   328→
   329→    // Merge workflow-level env + matrix-derived env.
   330→    const workflowEnv: Record<string, string> = { ...(workflow.env ?? {}) };
   331→    if (options.matrixValues) {
   332→      for (const [k, v] of Object.entries(options.matrixValues)) {
   333→        workflowEnv[`MATRIX_${k.toUpperCase()}`] = v;
   334→      }
   335→    }
   336→
   337→    // Run each step.
   338→    let lastExitCode = 0;
   339→    for (const step of workflow.steps) {
   340→      // Matrix substitution in step command + env.
   341→      const matrix = options.matrixValues ?? {};
   342→      const stepCommand = substituteMatrix(step.run, matrix);
   343→      const stepEnv: Record<string, string> = {
   344→        ...workflowEnv,
   345→        ...substituteMatrixInRecord(step.env ?? {}, matrix),
   346→      };
   347→
   348→      // Cache restore (per-step).
   349→      if (step.cache?.restore) {
   350→        const hit = await hasCache(projectId, step.cache.key);
   351→        if (hit) {
   352→          const restored = await restoreCache(projectId, step.cache.key);
   353→          if (restored.hit) {
   354→            await appendLog(runId, 'system', `💾 Cache hit: ${step.cache.key} (${formatBytes(restored.size ?? 0)})`);
   355→          }
   356→        } else {
   357→          await appendLog(runId, 'system', `💾 Cache miss: ${step.cache.key}`);
   358→        }
   359→      }
   360→
   361→      // Run with per-step retry.
   362→      const maxAttempts = (step.retry ?? 0) + 1;
   363→      let attempt = 0;
   364→      let stepExit = 1;
   365→      while (attempt < maxAttempts) {
   366→        if (attempt > 0) {
   367→          emit({ type: 'retry', runId, retryAttempt: attempt });
   368→          await appendLog(runId, 'system', `↻ Retry ${attempt}/${maxAttempts - 1} for: ${step.name}`);
   369→        }
   370→        await appendLog(runId, 'system', `▶ ${step.name}`);
   371→        // Pass the step with `run` already matrix-substituted so the
   372→        // executeStepCommand helper can write the script body to a temp
   373→        // file when language is node/python/ruby.
   374→        stepExit = await executeStepCommand(
   375→          runId,
   376→          { ...step, run: stepCommand },
   377→          step.workingDir ? path.join(projectRoot, step.workingDir) : projectRoot,
   378→          { secrets: allSecrets, extraEnv: stepEnv, timeoutMs: step.timeoutMs },
   379→        );
   380→        if (stepExit === 0) break;
   381→        attempt++;
   382→      }
   383→
   384→      // Cache save (per-step) — even on failure (partial cache is useful).
   385→      if (step.cache?.save) {
   386→        try {
   387→          const { size } = await saveCache(projectId, step.cache.key, step.cache.key, step.cache.paths);
   388→          await appendLog(runId, 'system', `💾 Cache saved: ${step.cache.key} (${formatBytes(size)})`);
   389→        } catch (err) {
   390→          await appendLog(runId, 'system', `Cache save failed: ${err instanceof Error ? err.message : String(err)}`);
   391→        }
   392→      }
   393→
   394→      // Test report capture (per-step).
   395→      if (step.testReport) {
   396→        try {
   397→          const reportPath = path.join(projectRoot, step.testReport.path);
   398→          if (fs.existsSync(reportPath)) {
   399→            const content = fs.readFileSync(reportPath, 'utf-8');
   400→            const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   401→            let report;
   402→            if (step.testReport.format === 'junit') report = parseJUnit(content);
   403→            else if (step.testReport.format === 'json') report = parseJSONReport(content);
   404→            else report = parseTAP(content);
   405→            await db.testReport.create({
   406→              data: {
   407→                runId,
   408→                format: report.format,
   409→                total: report.total,
   410→                passed: report.passed,
   411→                failed: report.failed,
   412→                skipped: report.skipped,
   413→                duration: report.duration ?? null,
   414→                suites: JSON.stringify(report.suites),
   415→              },
   416→            });
   417→            await appendLog(runId, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   418→          }
   419→        } catch (err) {
   420→          await appendLog(runId, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   421→        }
   422→      }
   423→
   424→      if (stepExit !== 0) {
   425→        lastExitCode = stepExit;
   426→        if (step.continueOnError) {
   427→          await appendLog(runId, 'system', `⚠ Step "${step.name}" failed with exit code ${stepExit} (continueOnError=true).`);
   428→          continue;
   429→        }
   430→        await appendLog(runId, 'system', `✗ Step "${step.name}" failed with exit code ${stepExit} after ${attempt} attempt(s).`);
   431→        await finishRun(runId, 'failed', stepExit, Date.now() - startedAt);
   432→        return;
   433→      }
   434→      await appendLog(runId, 'system', `✓ ${step.name}`);
   435→    }
   436→
   437→    // Workflow-level test report capture (after all steps complete).
   438→    if (workflow.testReport) {
   439→      try {
   440→        const reportPath = path.join(projectRoot, workflow.testReport.path);
   441→        if (fs.existsSync(reportPath)) {
   442→          const content = fs.readFileSync(reportPath, 'utf-8');
   443→          const { parseJUnit, parseJSONReport, parseTAP } = await import('./test-report');
   444→          let report;
   445→          if (workflow.testReport.format === 'junit') report = parseJUnit(content);
   446→          else if (workflow.testReport.format === 'json') report = parseJSONReport(content);
   447→          else report = parseTAP(content);
   448→          await db.testReport.create({
   449→            data: {
   450→              runId,
   451→              format: report.format,
   452→              total: report.total,
   453→              passed: report.passed,
   454→              failed: report.failed,
   455→              skipped: report.skipped,
   456→              duration: report.duration ?? null,
   457→              suites: JSON.stringify(report.suites),
   458→            },
   459→          });
   460→          await appendLog(runId, 'system', `📊 Test report: ${report.passed} passed, ${report.failed} failed, ${report.skipped} skipped (${report.total} total).`);
   461→        }
   462→      } catch (err) {
   463→        await appendLog(runId, 'system', `Test report parsing failed: ${err instanceof Error ? err.message : String(err)}`);
   464→      }
   465→    }
   466→
   467→    await finishRun(runId, 'success', lastExitCode, Date.now() - startedAt);
   468→  } catch (err) {
   469→    await appendLog(runId, 'system', `Custom workflow crashed: ${err instanceof Error ? err.message : String(err)}`);
   470→    await finishRun(runId, 'failed', -1, Date.now() - startedAt);
   471→  }
   472→}

   474→// ---------------------------------------------------------------------------
   475→// Step execution (with secrets masking + per-step timeout)

   477→
   478→interface StepEnvOptions {
   479→  secrets: Record<string, string>;
   480→  extraEnv: Record<string, string>;
   481→  timeoutMs?: number;
   482→}
   483→
   484→// Track active child processes so cancellation could be wired up later.
   485→const active = new Map<string, { child: ReturnType<typeof spawn>; canceled: boolean; timeoutTimer?: ReturnType<typeof setTimeout> }>();
   486→
   487→/**
   488→ * Execute a single custom-workflow step. The interpreter is chosen by
   489→ * `step.language`:
   490→ *   - 'bash'   → spawn `bash -c <step.run>` (UNCHANGED from legacy behavior)
   491→ *   - 'node'   → write step.run to <cwd>/.forge-step-<uuid>.mjs, spawn `node <file>`
   492→ *   - 'python' → write step.run to <cwd>/.forge-step-<uuid>.py,  spawn `python3 <file>`
   493→ *   - 'ruby'   → write step.run to <cwd>/.forge-step-<uuid>.rb,  spawn `ruby <file>`
   494→ *
   495→ * The temp file is placed in the step's working dir (cwd) so relative
   496→ * requires/imports resolve correctly. The temp file is deleted best-effort
   497→ * after the process closes (success or error).
   498→ *
   499→ * ALL existing behavior is preserved: BLOCKED_PATTERNS security check
   500→ * (applied to the command text regardless of language), per-step timeout
   501→ * (SIGTERM then SIGKILL after 2s), `active` map tracking, stdout/stderr
   502→ * line streaming with createInterface + appendLog + maskSecrets, env
   503→ * building via buildProcessEnv, and the Promise<number> return type.
   504→ */
   505→async function executeStepCommand(
   506→  runId: string,
   507→  step: CustomWorkflowStep,
   508→  cwd: string,
   509→  envOptions: StepEnvOptions,
   510→): Promise<number> {
   511→  // Defensive: the workflow object may have been loaded from the DB without
   512→  // going through parseCustomWorkflow, so step.language could in theory be
   513→  // anything. Reject unknown languages explicitly rather than falling
   514→  // through to a default interpreter.
   515→  const rawLanguage = step.language ?? 'bash';
   516→  if (!(STEP_LANGUAGES as readonly string[]).includes(rawLanguage)) {
   517→    void appendLog(runId, 'system', `✗ Unknown step language "${String(step.language)}" for step "${step.name}". Allowed: ${STEP_LANGUAGES.join(', ')}.`);
   518→    return Promise.resolve(1);
   519→  }
   520→  const language: CustomWorkflowStepLanguage = rawLanguage;
   521→  const command = step.run;
   522→  return new Promise((resolve) => {
   523→    void (async () => {
   524→      // Security: block known-dangerous commands that read system files.
   525→      // Applied to ALL languages so a ruby/python/node step can't bypass it.
   526→      const BLOCKED_PATTERNS = [
   527→        /\/etc\/(passwd|shadow|sudoers)/i,
   528→        /\/proc\/self\/(environ|cmdline)/i,
   529→        /\brm\s+-rf\s+\//i,
   530→        /\bmkfs\b/i,
   531→        /\bdd\s+if=\/dev\//i,
   532→        /:\(\)\s*\{/i,  // fork bomb
   533→      ];
   534→      for (const pattern of BLOCKED_PATTERNS) {
   535→        if (pattern.test(command)) {
   536→          await appendLog(runId, 'system', `🚫 Command blocked by security policy: ${command.slice(0, 80)}`);
   537→          resolve(126); // 126 = command not executable (security)
   538→          return;
   539→        }
   540→      }
   541→
   542→      // Surface which interpreter is being used so users can see it in logs.
   543→      await appendLog(runId, 'system', `▶ Running ${language} step: ${step.name}`);
   544→
   545→      const baseEnv = await buildProcessEnv('', {
   546→        extraEnv: envOptions.extraEnv,
   547→        projectRoot: cwd,
   548→      });
   549→      const fullEnv = { ...baseEnv, ...envOptions.secrets };
   550→
   551→      // For bash: spawn directly with -c (legacy behavior, unchanged).
   552→      // For node/python/ruby: write the script to a temp file in cwd so
   553→      // relative imports work, then spawn the interpreter with the file.
   554→      let tempFile: string | undefined;
   555→      let spawnCmd: string;
   556→      let spawnArgs: string[];
   557→      if (language === 'bash') {
   558→        spawnCmd = 'bash';
   559→        spawnArgs = ['-c', command];
   560→      } else if (language === 'node') {
   561→        tempFile = path.join(cwd, `.forge-step-${randomUUID()}.mjs`);
   562→        try {
   563→          await fs.promises.writeFile(tempFile, command, 'utf-8');
   564→        } catch (err) {
   565→          await appendLog(runId, 'system', `Failed to write temp script: ${err instanceof Error ? err.message : String(err)}`);
   566→          resolve(1);
   567→          return;
   568→        }
   569→        spawnCmd = 'node';
   570→        spawnArgs = [tempFile];
   571→      } else if (language === 'python') {
   572→        tempFile = path.join(cwd, `.forge-step-${randomUUID()}.py`);
   573→        try {
   574→          await fs.promises.writeFile(tempFile, command, 'utf-8');
   575→        } catch (err) {
   576→          await appendLog(runId, 'system', `Failed to write temp script: ${err instanceof Error ? err.message : String(err)}`);
   577→          resolve(1);
   578→          return;
   579→        }
   580→        spawnCmd = 'python3';
   581→        spawnArgs = [tempFile];
   582→      } else {
   583→        // language === 'ruby'
   584→        tempFile = path.join(cwd, `.forge-step-${randomUUID()}.rb`);
   585→        try {
   586→          await fs.promises.writeFile(tempFile, command, 'utf-8');
   587→        } catch (err) {
   588→          await appendLog(runId, 'system', `Failed to write temp script: ${err instanceof Error ? err.message : String(err)}`);
   589→          resolve(1);
   590→          return;
   591→        }
   592→        spawnCmd = 'ruby';
   593→        spawnArgs = [tempFile];
   594→      }
   595→
   596→      // Best-effort cleanup helper. Called from close/error/throw paths.
   597→      const cleanupTempFile = () => {
   598→        if (tempFile) {
   599→          fs.promises.unlink(tempFile).catch(() => { /* ignore */ });
   600→        }
   601→      };
   602→
   603→      let child: import('node:child_process').ChildProcessWithoutNullStreams;
   604→      try {
   605→        child = spawn(spawnCmd, spawnArgs, {
   606→          cwd,
   607→          env: fullEnv as NodeJS.ProcessEnv,
   608→        }) as import('node:child_process').ChildProcessWithoutNullStreams;
   609→      } catch (err) {
   610→        cleanupTempFile();
   611→        await appendLog(runId, 'system', `spawn error: ${err instanceof Error ? err.message : String(err)}`);
   612→        resolve(1);
   613→        return;
   614→      }
   615→      active.set(runId, { child, canceled: false });
   616→
   617→      let timeoutTimer: ReturnType<typeof setTimeout> | undefined;
   618→      let killTimer: ReturnType<typeof setTimeout> | undefined;
   619→      if (envOptions.timeoutMs) {
   620→        timeoutTimer = setTimeout(() => {
   621→          try { child.kill('SIGTERM'); } catch { /* ignore */ }
   622→          void appendLog(runId, 'system', `⏱ Step timed out after ${envOptions.timeoutMs}ms. Killing process...`);
   623→          // Force-kill with SIGKILL after 2 seconds if still alive.
   624→          killTimer = setTimeout(() => {
   625→            try { child.kill('SIGKILL'); } catch { /* ignore */ }
   626→            void appendLog(runId, 'system', `⏱ Force-killed (SIGKILL).`);
   627→          }, 2000);
   628→        }, envOptions.timeoutMs);
   629→        active.get(runId)!.timeoutTimer = timeoutTimer;
   630→      }
   631→
   632→      const stdout = createInterface({ input: child.stdout });
   633→      const stderr = createInterface({ input: child.stderr });
   634→      stdout.on('line', (line) => { void appendLog(runId, 'stdout', maskSecrets(line, envOptions.secrets)); });
   635→      stderr.on('line', (line) => { void appendLog(runId, 'stderr', maskSecrets(line, envOptions.secrets)); });
   636→
   637→      child.on('close', (code) => {
   638→        if (timeoutTimer) clearTimeout(timeoutTimer);
   639→        if (killTimer) clearTimeout(killTimer);
   640→        active.delete(runId);
   641→        cleanupTempFile();
   642→        resolve(code ?? 0);
   643→      });
   644→      child.on('error', (err) => {
   645→        if (timeoutTimer) clearTimeout(timeoutTimer);
   646→        if (killTimer) clearTimeout(killTimer);
   647→        void appendLog(runId, 'system', `spawn error: ${err.message}`);
   648→        active.delete(runId);
   649→        cleanupTempFile();
   650→        resolve(1);
   651→      });
   652→    })();
   653→  });
   654→}

   660→function substituteMatrix(text: string, matrix: MatrixRow): string {
   661→  return text.replace(/\$\{\{\s*matrix\.(\w+)\s*\}\}/g, (_, key) => matrix[key] ?? '');
   662→}

   664→function substituteMatrixInRecord(rec: Record<string, string>, matrix: MatrixRow): Record<string, string> {
   665→  const out: Record<string, string> = {};
   666→  for (const [k, v] of Object.entries(rec)) out[k] = substituteMatrix(v, matrix);
   667→  return out;
   668→}

   670→function formatBytes(bytes: number): string {
   671→  if (bytes < 1024) return `${bytes} B`;
   672→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   673→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   674→}
