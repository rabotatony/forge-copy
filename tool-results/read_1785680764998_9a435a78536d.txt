     1→// ============================================================
     2→// Forge — workflow catalog
     3→// ============================================================
     4→// Workflows are predefined CI jobs (like GitHub Actions workflows).
     5→// Each workflow knows how to build its command list for a given
     6→// project root and detection.
     7→//
     8→// Phase 2 extensions:
     9→//   • Optional `secrets`, `cache`, `testReport`, `requiresApproval`,
    10→//     `defaultRetry`, `defaultTimeoutMs` fields per workflow — these
    11→//     are read by the API layer and passed into `startRunExtended`.
    12→//   • Optional `applies(detection, projectRoot)` predicate for
    13→//     file-existence checks (e.g. Dockerfile) that `build(detection)`
    14→//     alone can't perform.
    15→//   • `WorkflowStep` extended with `env`, `retry`, `timeoutMs`,
    16→//     `workingDir` for per-step overrides.
    17→// ============================================================
    18→
    19→import * as fs from 'node:fs';
    20→import * as path from 'node:path';
    21→import type { Detection, ProjectKind } from './detector';
    22→
    23→export interface WorkflowStep {
    24→  // Human-readable label shown in the UI.
    25→  label: string;
    26→  // Shell command (run with `bash -c` on unix, or via child_process spawn).
    27→  command: string;
    28→  // Working directory (always the project root, but kept for flexibility).
    29→  cwd?: 'project';
    30→  // Optional: which stream to mark the output (default: auto from process).
    31→  stream?: 'stdout' | 'stderr' | 'system';
    32→  // Optional: extra env vars for this step only (merged on top of run env).
    33→  env?: Record<string, string>;
    34→  // Optional: per-step retry count (overrides the run-level retry).
    35→  retry?: number;
    36→  // Optional: per-step timeout in ms (overrides the run-level timeout).
    37→  timeoutMs?: number;
    38→  // Optional: working directory relative to project root.
    39→  workingDir?: string;
    40→}
    41→
    42→export interface WorkflowCacheConfig {
    43→  label: string;
    44→  // Paths (relative to project root) to save/restore.
    45→  paths: string[];
    46→  // Built-in cache-key generator to use.
    47→  keyGenerator: 'node' | 'cargo' | 'go' | 'python' | 'none';
    48→}
    49→
    50→export interface WorkflowTestReportConfig {
    51→  format: 'junit' | 'json' | 'tap';
    52→  // Path (relative to project root) of the test report file produced.
    53→  path: string;
    54→}
    55→
    56→export interface Workflow {
    57→  key: string;
    58→  name: string;
    59→  description: string;
    60→  // Icon name from lucide-react.
    61→  icon: string;
    62→  // Which project kinds this workflow supports.
    63→  kinds: ProjectKind[];
    64→  // Builder returns a list of steps; or null if the workflow doesn't
    65→  // apply to this detection (e.g. no `build` script in package.json).
    66→  build: (detection: Detection) => WorkflowStep[] | null;
    67→  // Whether to capture artifacts after the run.
    68→  producesArtifacts?: (detection: Detection, projectRoot: string) => ArtifactSpec[];
    69→
    70→  // --- Phase 2 optional fields ---
    71→  // File-existence / applicability predicate. If present and returns
    72→  // false, the workflow is excluded from `workflowsForKind` results.
    73→  // Use this for things `build(detection)` can't check on its own
    74→  // (e.g. Dockerfile presence).
    75→  applies?: (detection: Detection, projectRoot: string) => boolean;
    76→  // Secret keys to inject as env vars when running this workflow.
    77→  secrets?: string[];
    78→  // Cache configuration (the caller computes the actual cache key via
    79→  // the corresponding `*CacheKey` helper from `./cache`).
    80→  cache?: WorkflowCacheConfig;
    81→  // Test report capture: if set, the runner parses the file at `path`
    82→  // after the run and stores a TestReport row.
    83→  testReport?: WorkflowTestReportConfig;
    84→  // Whether this workflow requires manual approval before running.
    85→  requiresApproval?: boolean;
    86→  // Default per-step retry count for this workflow.
    87→  defaultRetry?: number;
    88→  // Default per-run timeout in ms for this workflow.
    89→  defaultTimeoutMs?: number;
    90→  // Plugin flag: when true, this workflow is implemented by a
    91→  // `WorkflowPlugin` registered in `./workflow-plugins.ts` (e.g. the
    92→  // AxiomState `parse` / `bundle` workflows). The engine checks the
    93→  // plugin registry FIRST and, if a plugin is registered for this
    94→  // workflow's key, dispatches to `plugin.execute()` instead of
    95→  // running shell steps. `build()` should return `[]` (empty) — the
    96→  // steps are ignored when a plugin is registered.
    97→  plugin?: boolean;
    98→}
    99→
   100→export interface ArtifactSpec {
   101→  // Logical name shown in the UI.
   102→  name: string;
   103→  // Glob pattern relative to projectRoot.
   104→  glob: string;
   105→  mime: string;
   106→}
   107→
   108→// ---------------------------------------------------------------------------
   109→// Helpers
   110→// ---------------------------------------------------------------------------
   111→
   112→function fileExists(projectRoot: string, file: string): boolean {
   113→  try {
   114→    return fs.existsSync(path.join(projectRoot, file));
   115→  } catch {
   116→    return false;
   117→  }
   118→}
   119→
   120→function hasNodeScript(detection: Detection, name: string): boolean {
   121→  return detection.type === 'node' && !!detection.scripts?.[name];
   122→}
   123→
   124→function projectName(detection: Detection): string {
   125→  if (detection.type === 'node' && detection.packageName) {
   126→    // Strip scope (e.g. @forge/app -> app) for docker tag use.
   127→    return detection.packageName.replace(/^@[^/]+\//, '');
   128→  }
   129→  if (detection.type === 'rust' && detection.crateName) return detection.crateName;
   130→  if (detection.type === 'go' && detection.moduleName) {
   131→    return detection.moduleName.split('/').pop() ?? 'app';
   132→  }
   133→  if (detection.type === 'python' && detection.projectName) return detection.projectName;
   134→  return 'app';
   135→}

   722→export function getWorkflow(key: string): Workflow | undefined {
   723→  return ALL_WORKFLOWS.find(w => w.key === key);

   726→/**
   727→ * List workflows applicable to the given project kind + detection.
   728→ * If `projectRoot` is provided, the optional `applies(detection, projectRoot)`
   729→ * predicate is consulted (e.g. for Dockerfile checks). If omitted, workflows
   730→ * without `applies` are listed; workflows with `applies` are listed too
   731→ * (with the predicate skipped, treating it as true).
   732→ */
   733→export function workflowsForKind(
   734→  kind: ProjectKind,
   735→  detection: Detection,
   736→  projectRoot?: string,
   737→): Workflow[] {
   738→  return ALL_WORKFLOWS.filter(w => {
   739→    if (!w.kinds.includes(kind)) return false;
   740→    // `build` must return non-null for this detection.
   741→    if (w.build(detection) === null) return false;
   742→    // `applies` predicate (only when projectRoot is available).
   743→    if (w.applies && projectRoot) {
   744→      try {
   745→        if (!w.applies(detection, projectRoot)) return false;
   746→      } catch {
   747→        // If the predicate throws (e.g. fs error), be conservative and skip.
   748→        return false;
   749→      }
   750→    }
   751→    return true;
   752→  });
   753→}
