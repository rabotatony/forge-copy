  5670→function median(values: number[]): number {
  5671→  if (values.length === 0) return 0;
  5672→  const sorted = [...values].sort((a, b) => a - b);
  5673→  const mid = Math.floor(sorted.length / 2);
  5674→  return sorted.length % 2 !== 0
  5675→    ? sorted[mid]
  5676→    : Math.round((sorted[mid - 1] + sorted[mid]) / 2);
  5677→}

  5679→/**
  5680→ * Estimate cyclomatic complexity by counting branching keywords:
  5681→ * if, elif, for, while, and, or, except. Each adds 1 to the base complexity of 1.

  5683→function measureComplexity(code: string): number {

  5691→/**
  5692→ * Estimate the maximum nesting depth of a Python source string by counting
  5693→ * indentation levels. Each 4-space indent (or tab) at the start of a line
  5694→ * that is deeper than the previous non-blank line's indent counts as a level.
  5695→ * Returns the deepest indentation seen, in units of 4 spaces.
  5696→ */
  5697→function measureMaxNesting(code: string): number {
  5698→  let maxDepth = 0;
  5699→  for (const line of code.split('\n')) {
  5700→    if (!line.trim()) continue;
  5701→    const leading = line.match(/^(\t|    )*/)?.[0] ?? '';
  5702→    // Convert tabs to 4-space units, then count groups of 4.
  5703→    const spaces = leading.replace(/\t/g, '    ');
  5704→    const depth = Math.floor(spaces.length / 4);
  5705→    if (depth > maxDepth) maxDepth = depth;
  5706→  }
  5707→  return maxDepth;
  5708→}

  5774→interface CIFailure { type: string; file?: string; line?: number; message: string; }
  5775→function parseCILog(log: string): { failures: CIFailure[]; total: number } {
  5776→  const failures: CIFailure[] = [];
  5777→  const patterns = [
  5778→    { regex: /File "([^"]+)", line (\d+).*?\n\s*(.*?)(?=\n  File |\nTraceback|\Z)/gs, type: 'runtime-error' },
  5779→    { regex: /error TS\d+: (.+?)\n\s+at (.+?):(\d+)/g, type: 'compile-error' },
  5780→    { regex: /FAILED (\S+) - (.+)/g, type: 'test-failure' },
  5781→    { regex: /##\[error\](.+)/g, type: 'ci-error' },
  5782→    { regex: /Error:\s+(.+)/g, type: 'generic-error' },
  5783→  ];
  5784→  for (const p of patterns) {
  5785→    let m; while ((m = p.regex.exec(log)) !== null) {
